Crea il hook `src/hooks/useSidebarSubCategories.ts`:

```typescript
// src/hooks/useSidebarSubCategories.ts
import { useState, useCallback, useEffect } from "react";
import { useLocation } from "react-router-dom";
import {
  URL_TO_SUBCATEGORY,
  DEFAULT_SUBCATEGORY_STATE,
} from "@/lib/sidebarSubCategories";

const STORAGE_KEY = "sidebar_subcategories_state";

function loadState(): Record<string, boolean> {
  try {
    const stored = localStorage.getItem(STORAGE_KEY);
    if (stored) return { ...DEFAULT_SUBCATEGORY_STATE, ...JSON.parse(stored) };
  } catch {}
  return { ...DEFAULT_SUBCATEGORY_STATE };
}

function saveState(state: Record<string, boolean>) {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
  } catch {}
}

export function useSidebarSubCategories() {
  const location = useLocation();

  const [state, setState] = useState<Record<string, boolean>>(() => {
    const s = loadState();
    // Auto-espandi la sub-categoria della route corrente al primo render
    const active = getActiveSubcategory(location.pathname);
    if (active) s[active] = true;
    return s;
  });

  // Auto-espandi quando cambia la route
  useEffect(() => {
    const active = getActiveSubcategory(location.pathname);
    if (active && !state[active]) {
      setState((prev) => {
        const next = { ...prev, [active]: true };
        saveState(next);
        return next;
      });
    }
  }, [location.pathname]);

  const toggle = useCallback((id: string) => {
    setState((prev) => {
      const next = { ...prev, [id]: !prev[id] };
      saveState(next);
      return next;
    });
  }, []);

  const isOpen = useCallback(
    (id: string) => state[id] ?? true,
    [state]
  );

  return { toggle, isOpen };
}

function getActiveSubcategory(pathname: string): string | null {
  // Cerca prima match esatto, poi match per prefisso (più specifico prima)
  const sorted = Object.entries(URL_TO_SUBCATEGORY).sort(
    (a, b) => b[0].length - a[0].length
  );
  for (const [url, section] of sorted) {
    if (pathname === url || pathname.startsWith(url + "/")) {
      return section;
    }
  }
  return null;
}
```

Crea il file `src/components/layouts/SubCategorySection.tsx`:

```typescript
// src/components/layouts/SubCategorySection.tsx
import { ChevronDown } from "lucide-react";
import { cn } from "@/lib/utils";

interface SubCategorySectionProps {
  label: string;
  isOpen: boolean;
  onToggle: () => void;
  children: React.ReactNode;
}

export function SubCategorySection({
  label,
  isOpen,
  onToggle,
  children,
}: SubCategorySectionProps) {
  return (
    <div className="mb-0.5">
      {/* Header sub-categoria */}
      <button
        onClick={onToggle}
        className={cn(
          "w-full flex items-center justify-between",
          "px-3 py-1.5",
          "text-[10px] font-semibold uppercase tracking-[0.08em]",
          "text-muted-foreground/70 hover:text-muted-foreground",
          "transition-colors duration-150 cursor-pointer select-none",
          "rounded-md hover:bg-muted/40"
        )}
      >
        <span>{label}</span>
        <ChevronDown
          className={cn(
            "h-3 w-3 transition-transform duration-200 shrink-0",
            isOpen ? "rotate-0" : "-rotate-90"
          )}
        />
      </button>

      {/* Voci animate */}
      <div
        className={cn(
          "overflow-hidden transition-all duration-200 ease-in-out",
          isOpen ? "max-h-[400px] opacity-100" : "max-h-0 opacity-0"
        )}
      >
        {children}
      </div>
    </div>
  );
}
```

PERCHÉ QUESTO STILE:
- Label 10px uppercase muted: funge da etichetta discreta, non compete con le voci
- Nessuno sfondo sulla sub-categoria stessa → il bordo sinistro sulla voce attiva
  è il punto focale, non il contenitore
- ChevronDown che ruota a -90° quando chiuso (invece di scomparire)
- Hover sottile: bg-muted/40 arrotondato

NOTE IMPORTANTI
============================================================

- Il filtro `.filter((item) => (URL_TO_SUBCATEGORY[item.url] || "") === "gi_operazioni")`
  usa il mapping da sidebarSubCategories.ts. Se una voce non ha mapping,
  NON appare in nessuna sub-categoria — non scompare, ma va aggiunto il mapping.

- DOPO l'implementazione: controlla che ogni voce di visibleInternalItems e
  visibleMarketingItems sia mappata in URL_TO_SUBCATEGORY. Le voci non mappate
  non saranno visibili. Adatta i path alle URL reali in sidebarConfig.ts.

- La voce Cruscotto rimane standalone sopra Gestione Interna, NON dentro una sub-categoria.

- Il Collapsible esterno (Gestione Interna / Marketing) rimane invariato —
  collassa tutto il blocco. Le SubCategorySection sono un secondo livello dentro di esso.

- I badge BETA e i permission check (filterNavItems) continuano a funzionare
  perché i filtri operano su visibleInternalItems/visibleMarketingItems già filtrati.

- NON modificare sidebarConfig.ts, NavLink.tsx, né le route esistenti.
