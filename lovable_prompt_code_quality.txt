- src/pages/azienda/settings/SettingsUserDetail.tsx (righe ~127-136)

Crea il file: src/utils/permissionSync.ts

```typescript
/**
 * Sincronizza i flag legacy can_view_marketing / can_edit_marketing
 * dai permessi granulari. Da chiamare PRIMA di ogni salvataggio permessi.
 */
export function syncLegacyMarketingFlags(
  perms: Record<string, boolean>
): Record<string, boolean> {
  return {
    ...perms,
    can_view_marketing:
      perms.can_view_marketing_contacts ||
      perms.can_view_opportunities ||
      perms.can_view_marketing_activities ||
      perms.can_view_email_campaigns ||
      perms.can_view_whatsapp ||
      perms.can_view_automations ||
      false,
    can_edit_marketing:
      perms.can_edit_marketing_contacts ||
      perms.can_edit_opportunities ||
      perms.can_edit_marketing_activities ||
      perms.can_edit_email_campaigns ||
      perms.can_send_whatsapp ||
      perms.can_manage_automations ||
      false,
  };
}
```

Poi in ciascuno dei tre file sostituisci il blocco inline con:
```typescript

// ...
```
