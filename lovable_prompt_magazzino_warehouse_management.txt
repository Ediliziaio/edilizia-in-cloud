-- Rimuove il vecchio CHECK constraint se esiste, aggiunge il nuovo
DO $$
BEGIN
  -- Tenta di rimuovere il vecchio constraint (ignora errore se non esiste)
  ALTER TABLE order_items
    DROP CONSTRAINT IF EXISTS order_items_status_check;

  -- Aggiunge il nuovo constraint con stati estesi
  ALTER TABLE order_items
    ADD CONSTRAINT order_items_status_check
    CHECK (status IN (
      'da_ordinare',   -- da ordinare al fornitore
      'ordinato',      -- ordine inviato al fornitore
      'in_arrivo',     -- confermato in spedizione / data consegna fissata
      'in_magazzino',  -- fisicamente in magazzino
      'prenotato',     -- in magazzino ma riservato per questa commessa
      'installato'     -- montato/posato in cantiere
    ));
EXCEPTION WHEN OTHERS THEN NULL;
END $$;

-- ============================================================
-- 7. TRIGGER — scarico automatico a posa completata
-- ============================================================
-- Funzione: quando un order_item passa a 'installato' e ha stock_item_id,
-- crea automaticamente un movimento scarico_automatico
CREATE OR REPLACE FUNCTION trigger_auto_deduct_stock()
RETURNS TRIGGER AS $$
DECLARE
  v_company_id UUID;
BEGIN
  -- Solo se lo status cambia a 'installato' e c'è un articolo collegato
  IF NEW.status = 'installato'
     AND OLD.status <> 'installato'
     AND NEW.stock_item_id IS NOT NULL
     AND NEW.auto_deducted = FALSE
  THEN
    -- Ottieni company_id dall'ordine
    SELECT company_id INTO v_company_id
    FROM orders WHERE id = NEW.order_id;

    -- Crea il movimento scarico automatico
    INSERT INTO warehouse_movements (
      stock_item_id,
      order_item_id,
      order_id,
      movement_type,
      quantity,
      notes,
      performed_by,
      company_id,
      lot_number
    ) VALUES (
      NEW.stock_item_id,
      NEW.id,
      NEW.order_id,
      'scarico_automatico',
      NEW.quantity,
      'Scarico automatico a posa completata',
      auth.uid(),
      v_company_id,
      NEW.lot_number
    );

    -- Aggiorna la giacenza
    UPDATE warehouse_stock
    SET
      quantity = GREATEST(0, quantity - NEW.quantity),
      quantity_reserved = GREATEST(0, quantity_reserved - NEW.quantity_reserved),
      updated_at = NOW()
    WHERE id = NEW.stock_item_id;

    -- Segna come già scaricato per evitare doppi scarichi
    NEW.auto_deducted := TRUE;
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Crea il trigger sull'aggiornamento status order_items
DROP TRIGGER IF EXISTS auto_deduct_on_install ON order_items;
CREATE TRIGGER auto_deduct_on_install
  BEFORE UPDATE OF status ON order_items
  FOR EACH ROW
  EXECUTE FUNCTION trigger_auto_deduct_stock();

-- ============================================================
-- 8. TRIGGER — aggiorna quantity_reserved su prenotazione/rilascio
-- ============================================================
CREATE OR REPLACE FUNCTION trigger_update_stock_reservation()
RETURNS TRIGGER AS $$
BEGIN
  -- Quando stock_item_id viene assegnato o status diventa 'prenotato'
  IF NEW.stock_item_id IS NOT NULL THEN
    IF NEW.status = 'prenotato' AND (OLD.status <> 'prenotato' OR OLD.stock_item_id IS NULL) THEN
      -- Prenotazione: incrementa quantity_reserved
      UPDATE warehouse_stock
      SET quantity_reserved = quantity_reserved + NEW.quantity,
          updated_at = NOW()
      WHERE id = NEW.stock_item_id;

    ELSIF OLD.status = 'prenotato' AND NEW.status <> 'prenotato' AND OLD.stock_item_id IS NOT NULL THEN
      -- Rilascio prenotazione: decrementa quantity_reserved
      UPDATE warehouse_stock
      SET quantity_reserved = GREATEST(0, quantity_reserved - OLD.quantity),
          updated_at = NOW()
      WHERE id = OLD.stock_item_id;
    END IF;
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS update_reservation_on_status_change ON order_items;
CREATE TRIGGER update_reservation_on_status_change
  AFTER UPDATE OF status, stock_item_id ON order_items
  FOR EACH ROW
  EXECUTE FUNCTION trigger_update_stock_reservation();

-- 9. RPC: ordini bloccati per mancanza materiali
-- ============================================================
CREATE OR REPLACE FUNCTION get_blocked_orders(p_company_id UUID)
RETURNS TABLE (
  order_id UUID,
  order_code TEXT,
  customer_name TEXT,
  expected_date DATE,
  work_start_date DATE,
  blocking_items_count BIGINT,
  missing_items JSONB,
  urgency_level TEXT   -- 'critica' | 'alta' | 'normale'
) AS $$
BEGIN
  RETURN QUERY
  WITH blocking AS (
    SELECT
      oi.order_id,
      COUNT(oi.id) AS cnt,
      jsonb_agg(jsonb_build_object(
        'item_id', oi.id,
        'name', oi.name,
        'status', oi.status,
        'quantity', oi.quantity,
        'delivery_date', oi.delivery_date
      ) ORDER BY oi.status) AS items
    FROM order_items oi
    WHERE oi.status IN ('da_ordinare', 'ordinato', 'in_arrivo')
    GROUP BY oi.order_id
    HAVING COUNT(oi.id) > 0
  )
  SELECT
    o.id AS order_id,
    o.order_code,
    p.first_name || ' ' || p.last_name AS customer_name,
    o.expected_date,
    o.work_start_date,
    b.cnt AS blocking_items_count,
    b.items AS missing_items,
    CASE
      WHEN o.work_start_date <= CURRENT_DATE THEN 'critica'
      WHEN o.work_start_date <= CURRENT_DATE + INTERVAL '7 days' THEN 'alta'
      ELSE 'normale'
    END AS urgency_level
  FROM orders o
  INNER JOIN blocking b ON b.order_id = o.id
  LEFT JOIN profiles p ON p.id = o.customer_id
  WHERE o.company_id = p_company_id
  ORDER BY
    CASE
      WHEN o.work_start_date <= CURRENT_DATE THEN 0
      WHEN o.work_start_date <= CURRENT_DATE + INTERVAL '7 days' THEN 1
      ELSE 2
    END,
    o.work_start_date NULLS LAST;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================================
-- 10. RPC: alert sottoscorta
-- ============================================================
CREATE OR REPLACE FUNCTION get_low_stock_alerts(p_company_id UUID)
RETURNS TABLE (
  stock_item_id UUID,
  name TEXT,
  quantity INTEGER,
  quantity_reserved INTEGER,
  quantity_available INTEGER,
  min_stock_level INTEGER,
  deficit INTEGER,
  supplier_name TEXT,
  section_name TEXT
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    ws.id AS stock_item_id,
    ws.name,
    ws.quantity,
    ws.quantity_reserved,
    ws.quantity_available,
    ws.min_stock_level,
    (ws.min_stock_level - ws.quantity_available) AS deficit,
    s.name AS supplier_name,
    wsc.name AS section_name
  FROM warehouse_stock ws
  LEFT JOIN suppliers s ON s.id = ws.supplier_id
  LEFT JOIN warehouse_sections wsc ON wsc.id = ws.section_id
  WHERE ws.company_id = p_company_id
    AND ws.min_stock_level > 0
    AND ws.quantity_available < ws.min_stock_level
  ORDER BY (ws.min_stock_level - ws.quantity_available) DESC;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================================
-- 11. RPC: storico movimenti per commessa
-- ============================================================
CREATE OR REPLACE FUNCTION get_order_materials_history(
  p_company_id UUID,
  p_order_id UUID
)
RETURNS TABLE (
  movement_id UUID,
  stock_item_name TEXT,
  order_item_name TEXT,
  movement_type TEXT,
  quantity INTEGER,
  unit_cost NUMERIC,
  lot_number TEXT,
  notes TEXT,
  performed_by_name TEXT,
  created_at TIMESTAMPTZ
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    wm.id AS movement_id,
    ws.name AS stock_item_name,
    oi.name AS order_item_name,
    wm.movement_type,
    wm.quantity,
    wm.unit_cost,
    wm.lot_number,
    wm.notes,
    p.first_name || ' ' || p.last_name AS performed_by_name,
    wm.created_at
  FROM warehouse_movements wm
  LEFT JOIN warehouse_stock ws ON ws.id = wm.stock_item_id
  LEFT JOIN order_items oi ON oi.id = wm.order_item_id
  LEFT JOIN profiles p ON p.id = wm.performed_by
  WHERE wm.order_id = p_order_id
     OR wm.order_item_id IN (
       SELECT id FROM order_items WHERE order_id = p_order_id
     )
  ORDER BY wm.created_at DESC;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

        quantity: Number(row.quantity),
        unit_cost: row.unit_cost ? Number(row.unit_cost) : null,
      }));
    },
    staleTime: 1000 * 60 * 2,
  });
}

export function useLotBatches(companyId: string | null, stockItemId?: string) {
  return useQuery({
    queryKey: magazzinoKeys.lotBatches(companyId ?? '', stockItemId),

        .from('warehouse_lot_batches')

      if (stockItemId) {
        query = query.eq('stock_item_id', stockItemId);

export function useWarehouseStats(companyId: string | null) {
  return useQuery({
    queryKey: magazzinoKeys.stats(companyId ?? ''),
    enabled: !!companyId,
    queryFn: async (): Promise<WarehouseStats> => {
      const [stockRes, lowStockRes, blockedRes, pendingRes] = await Promise.all([
        supabase
          .from('warehouse_stock')
          .select('id, quantity, unit_cost, quantity_reserved, min_stock_level')
          .eq('company_id', companyId!),
        supabase
          .rpc('get_low_stock_alerts', { p_company_id: companyId! }),
        supabase
          .rpc('get_blocked_orders', { p_company_id: companyId! }),
        supabase
          .from('order_items')
          .select('id, status', { count: 'exact', head: true })
          .in('status', ['ordinato', 'in_arrivo']),
      ]);

      const stock = stockRes.data ?? [];
      const totalStockValue = stock.reduce(
        (s, item) => s + (Number(item.quantity) * Number(item.unit_cost ?? 0)),
        0
      );
      const reservedItems = stock.filter((i) => Number(i.quantity_reserved) > 0).length;

      return {
        totalStockItems: stock.length,
        totalStockValue,
        lowStockCount: (lowStockRes.data ?? []).length,
        blockedOrdersCount: (blockedRes.data ?? []).length,
        reservedQuantityItems: reservedItems,
        pendingDeliveriesCount: pendingRes.count ?? 0,
      };
    },
    staleTime: 1000 * 60 * 5,
  });
}

export function useApplyAuditAdjustmentMutation() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async ({
      auditId,
      stockItemId,
      actualQuantity,
      difference,
      companyId,
    }: {
      auditId: string;
      stockItemId: string;
      actualQuantity: number;
      difference: number;
      companyId: string;
    }) => {
      // Crea movimento di rettifica
      const { error: movError } = await supabase
        .from('warehouse_movements')
        .insert({
          stock_item_id: stockItemId,
          movement_type: 'rettifica',
          quantity: Math.abs(difference),
          notes: `Rettifica inventariale — differenza: ${difference > 0 ? '+' : ''}${difference}`,
          performed_by: (await supabase.auth.getUser()).data.user?.id,
          company_id: companyId,
        });
      if (movError) throw movError;

      // Aggiorna la giacenza alla quantità reale
      const { error: stockError } = await supabase
        .from('warehouse_stock')
        .update({
          quantity: actualQuantity,
          updated_at: new Date().toISOString(),
        })
        .eq('id', stockItemId);
      if (stockError) throw stockError;

      // Segna l'audit come applicato
      const { error: auditError } = await supabase
        .from('inventory_audits')
        .update({
          adjustment_applied: true,
          applied_at: new Date().toISOString(),
        })
        .eq('id', auditId);
      if (auditError) throw auditError;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: magazzinoKeys.all });
      queryClient.invalidateQueries({ queryKey: ['warehouse'] });
    },
  });
}

    await createAudit.mutateAsync({
      companyId,
      stockItemId: stockItem.id,
      systemQuantity: stockItem.quantity,
      actualQuantity: parsedActual,
      notes: notes || undefined,
      auditedBy: userId,
    });

    setActualQuantity('');
    setNotes('');
    onOpenChange(false);
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-w-md">
        <DialogHeader>
          <DialogTitle className="flex items-center gap-2">
            <ClipboardCheck className="h-5 w-5 text-primary" />
            Inventario fisico
          </DialogTitle>
          <DialogDescription>
            Conta il materiale fisicamente e inserisci la quantità reale.
            Se c'è differenza, potrai applicare la rettifica.
          </DialogDescription>
        </DialogHeader>

        <div className="space-y-4 py-2">
          <div className="p-3 rounded-lg bg-muted/50 border">
            <div className="text-sm font-medium">{stockItem.name}</div>
            <div className="text-sm text-muted-foreground mt-1">
              Quantità nel sistema: <strong>{stockItem.quantity}</strong>
            </div>
          </div>

          <div>
            <label className="text-sm font-medium">
              Quantità conteggiata fisicamente *
            </label>
            <Input
              type="number"
              min="0"
              placeholder="Inserisci il conteggio reale..."
              value={actualQuantity}
              onChange={(e) => setActualQuantity(e.target.value)}
              className="mt-1"
              autoFocus
            />
          </div>

          {isValid && difference !== null && (
            <div className={`p-3 rounded-lg border ${
              difference === 0
                ? 'bg-green-50 border-green-200'
                : difference > 0
                ? 'bg-blue-50 border-blue-200'
                : 'bg-red-50 border-red-200'
            }`}>
              <div className="flex items-center gap-2">
                {difference === 0 ? (
                  <><Check className="h-4 w-4 text-green-600" />
                  <span className="text-sm font-medium text-green-700">
                    Nessuna differenza — giacenza corretta
                  </span></>
                ) : (
                  <><AlertTriangle className="h-4 w-4 text-orange-600" />
                  <span className="text-sm font-medium">
                    Differenza: {difference > 0 ? '+' : ''}{difference} pz
                    {difference > 0 ? ' (eccedenza)' : ' (ammanchi)'}
                  </span></>
                )}
              </div>
            </div>
          )}

          <div>
            <label className="text-sm font-medium">Note (opzionale)</label>
            <Textarea
              placeholder="Es: Materiale consumato non registrato, rotture, furto..."
              value={notes}
              onChange={(e) => setNotes(e.target.value)}
              className="mt-1 min-h-[60px] resize-none"
            />
          </div>
        </div>

        <DialogFooter>
          <Button variant="outline" onClick={() => onOpenChange(false)}>
            Annulla
          </Button>
          <Button
            disabled={!isValid || createAudit.isPending}
            onClick={handleSubmit}
          >
            {createAudit.isPending ? 'Salvataggio...' : 'Registra inventario'}
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}

export function InventoryAuditsList({
