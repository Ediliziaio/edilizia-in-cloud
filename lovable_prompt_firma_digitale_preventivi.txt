```sql
-- Migration: firma_digitale_preventivi
-- Aggiunge colonne firma a `quotes` se non esistono già,
-- crea policy pubblica per QuoteSignPage e bucket per PDF firmati.

-- 1. Aggiungi colonne firma (safe: usa IF NOT EXISTS)
ALTER TABLE quotes
  ADD COLUMN IF NOT EXISTS signature_token   TEXT UNIQUE,
  ADD COLUMN IF NOT EXISTS signature_link    TEXT,
  ADD COLUMN IF NOT EXISTS sent_at           TIMESTAMPTZ,
  ADD COLUMN IF NOT EXISTS viewed_at         TIMESTAMPTZ,
  ADD COLUMN IF NOT EXISTS signed_at         TIMESTAMPTZ,
  ADD COLUMN IF NOT EXISTS signed_by_name    TEXT,
  ADD COLUMN IF NOT EXISTS signed_by_ip      TEXT,
  ADD COLUMN IF NOT EXISTS refused_at        TIMESTAMPTZ,
  ADD COLUMN IF NOT EXISTS refused_reason    TEXT,
  ADD COLUMN IF NOT EXISTS expires_at        TIMESTAMPTZ;

-- Indice per ricerca veloce tramite token
CREATE INDEX IF NOT EXISTS idx_quotes_signature_token ON quotes(signature_token)
  WHERE signature_token IS NOT NULL;

-- 2. Policy Supabase RLS — lettura pubblica tramite token
-- La pagina /offerta/:token NON usa autenticazione.
-- Espone solo le colonne necessarie, mai dati sensibili aziendali.
CREATE POLICY IF NOT EXISTS "public_read_quote_by_token"
ON quotes FOR SELECT
TO anon
USING (
  signature_token IS NOT NULL
  AND expires_at > now()
  AND status IN ('inviata', 'accettata', 'rifiutata')
);

-- 3. Policy pubblica per quote_items (lettura via token, join tramite edge function)
-- La edge function usa service_role quindi non serve policy anon su quote_items.
-- Lascia la lettura alle edge functions con service role key.

-- 4. Funzione per generare numero preventivo progressivo (se non esiste)
CREATE OR REPLACE FUNCTION generate_quote_number(p_company_id uuid)
RETURNS text
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_year   int  := EXTRACT(YEAR FROM now());
  v_count  int;
BEGIN
  SELECT COALESCE(COUNT(*), 0) + 1 INTO v_count
  FROM quotes
  WHERE company_id = p_company_id
    AND EXTRACT(YEAR FROM created_at) = v_year;
  RETURN 'OFF-' || v_year || '-' || LPAD(v_count::text, 3, '0');
END;
$$;
GRANT EXECUTE ON FUNCTION generate_quote_number(uuid) TO authenticated;

-- 5. Storage bucket per PDF firmati (separato da quote-pdfs)
INSERT INTO storage.buckets (id, name, public)
VALUES ('quote-signed-pdfs', 'quote-signed-pdfs', false)
ON CONFLICT (id) DO NOTHING;

CREATE POLICY IF NOT EXISTS "company_members_read_signed_pdfs"
ON storage.objects FOR SELECT TO authenticated
USING (
  bucket_id = 'quote-signed-pdfs'
  AND (storage.foldername(name))[1] = (
    SELECT company_id::text FROM profiles WHERE id = auth.uid()
  )
);

-- Service role può scrivere (usato dall'edge function quote-sign)
CREATE POLICY IF NOT EXISTS "service_role_write_signed_pdfs"
ON storage.objects FOR INSERT TO service_role
WITH CHECK (bucket_id = 'quote-signed-pdfs');

-- 6. Funzione helper per notifiche (se non esiste già)
-- Se create_notification RPC non esiste nel tuo schema, rimuovi la chiamata
-- dalla edge function quote-sign o crea questa funzione base:
CREATE OR REPLACE FUNCTION create_notification(
  p_company_id  uuid,
  p_type        text,
  p_title       text,
  p_body        text,
  p_link        text DEFAULT NULL
) RETURNS void
LANGUAGE plpgsql SECURITY DEFINER AS $$
BEGIN
  INSERT INTO notifications(company_id, type, title, body, link, read, created_at)
  VALUES (p_company_id, p_type, p_title, p_body, p_link, false, now())
  ON CONFLICT DO NOTHING;
EXCEPTION WHEN OTHERS THEN NULL; -- silenzia se notifications non esiste
END;
$$;
GRANT EXECUTE ON FUNCTION create_notification(uuid,text,text,text,text) TO service_role;
```

```typescript
import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from "jsr:@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: corsHeaders });

    Deno.env.get("SUPABASE_URL")!,
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
  );

  // Verifica autenticazione
  const authHeader = req.headers.get("authorization") ?? "";
  const { data: { user }, error: authErr } = await supabase.auth.getUser(
    authHeader.replace("Bearer ", "")
  );
  if (authErr || !user) {
    return new Response(JSON.stringify({ error: "Non autorizzato" }), {
      status: 401, headers: { ...corsHeaders, "Content-Type": "application/json" },
    });

    await req.json();

    return new Response(JSON.stringify({ error: "quote_id e recipient_email obbligatori" }), {
      status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" },
    });

    .from("quotes")

      id, company_id, quote_number, title, total, valid_until, status,

    `)
    .eq("id", quote_id)
    .maybeSingle();

      status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  // Verifica membership utente → azienda
  const { data: member } = await supabase
    .from("company_members")
    .select("id")
    .eq("company_id", quote.company_id)
    .eq("user_id", user.id)
    .maybeSingle();

    return new Response(JSON.stringify({ error: "Accesso negato" }), {
      status: 403, headers: { ...corsHeaders, "Content-Type": "application/json" },

  const { error: updateErr } = await supabase
    .from("quotes")
    .update({
      signature_token: token,
      signature_link:  signatureLink,
      sent_at:         new Date().toISOString(),
      expires_at:      expiresAt,
      status:          "inviata",
    })
    .eq("id", quote_id);

      status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" },

      style: "currency", currency: "EUR",

    const greeting = recipient_name ? `Gentile ${recipient_name},` : "Gentile Cliente,";

      : "";
```

```typescript
// supabase/functions/quote-sign/index.ts
// POST { token, action: 'view'|'sign'|'refuse', signer_name?, refuse_reason? }
// Pubblica — no autenticazione. Autenticazione tramite token univoco.

import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from "jsr:@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: corsHeaders });
  if (req.method !== "POST") {
    return new Response("Method not allowed", { status: 405, headers: corsHeaders });
  }

  const supabase = createClient(
    Deno.env.get("SUPABASE_URL")!,
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
  );

  const { token, action, signer_name, refuse_reason } = await req.json();

  // Validazione input
  if (!token || !action) {
    return new Response(JSON.stringify({ error: "token e action obbligatori" }), {
      status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  // Validazione formato token (UUID)
  const UUID_REGEX = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
  if (!UUID_REGEX.test(token)) {
    return new Response(JSON.stringify({ error: "Token non valido" }), {
      status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  if (!["view", "sign", "refuse"].includes(action)) {
    return new Response(JSON.stringify({ error: "action deve essere view, sign o refuse" }), {
      status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  // Carica la quote tramite token
  const { data: quote, error: quoteErr } = await supabase
    .from("quotes")
    .select(`
      id, company_id, quote_number, title, total, status,
      expires_at, signature_token, signed_at, signed_by_name,
      refused_at, valid_until, contact_id,
      quote_items(id, name, description, quantity, unit, unit_price, discount_percentage, total_price, position),
      companies!inner(id, name, email, phone, address, logo_url, vat_number)
    `)
    .eq("signature_token", token)
    .maybeSingle();

  if (quoteErr || !quote) {
    return new Response(JSON.stringify({ error: "Token non trovato" }), {
      status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  // Verifica scadenza
  if (quote.expires_at && new Date(quote.expires_at) < new Date()) {
    if (quote.status === "inviata") {
      await supabase.from("quotes").update({ status: "scaduta" }).eq("id", quote.id);
    }
    return new Response(JSON.stringify({ error: "expired", status: "scaduta" }), {
      status: 410, headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  // IP del client
  const clientIp =
    req.headers.get("x-forwarded-for") ??
    req.headers.get("cf-connecting-ip") ??
    "unknown";

  // ─────────────────────────────────────────────
  // ACTION: view
  // ─────────────────────────────────────────────
  if (action === "view") {
    // Traccia la prima visualizzazione
    if (!quote.viewed_at) {
      await supabase
        .from("quotes")
        .update({ viewed_at: new Date().toISOString() })
        .eq("id", quote.id);
    }

    // Restituisce i dati della quote (esclude dati sensibili interni)
    return new Response(
      JSON.stringify({
        id:           quote.id,
        quote_number: quote.quote_number,
        title:        quote.title,
        total:        quote.total,
        status:       quote.status,
        expires_at:   quote.expires_at,
        valid_until:  quote.valid_until,
        signed_at:    quote.signed_at,
        signed_by:    quote.signed_by_name,
        refused_at:   quote.refused_at,
        items:        (quote.quote_items as any[])?.sort((a, b) => a.position - b.position) ?? [],
        company: {
          name:       (quote.companies as any)?.name,
          email:      (quote.companies as any)?.email,
          phone:      (quote.companies as any)?.phone,
          address:    (quote.companies as any)?.address,
          logo_url:   (quote.companies as any)?.logo_url,
          vat_number: (quote.companies as any)?.vat_number,
        },
      }),
      { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  }

  // ─────────────────────────────────────────────
  // ACTION: sign
  // ─────────────────────────────────────────────
  if (action === "sign") {
    if (!signer_name?.trim()) {
      return new Response(JSON.stringify({ error: "signer_name obbligatorio" }), {
        status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    if (quote.status === "accettata") {
      return new Response(JSON.stringify({ error: "Preventivo già firmato" }), {
        status: 409, headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    if (quote.status !== "inviata") {
      return new Response(
        JSON.stringify({ error: `Impossibile firmare: stato attuale "${quote.status}"` }),
        { status: 422, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    const signedAt = new Date().toISOString();

    await supabase.from("quotes").update({
      status:         "accettata",
      signed_at:      signedAt,
      signed_by_name: signer_name.trim(),
      signed_by_ip:   clientIp,
    }).eq("id", quote.id);

    // Notifica all'azienda
    try {
      await supabase.rpc("create_notification", {
        p_company_id: quote.company_id,
        p_type:       "quote",
        p_title:      "Preventivo firmato!",
        p_body:       `${signer_name.trim()} ha accettato l'offerta ${quote.quote_number}`,
        p_link:       `/azienda/marketing/preventivi/${quote.id}`,
      });
    } catch (_) { /* notifiche opzionali */ }

    return new Response(
      JSON.stringify({ success: true, signed_at: signedAt, signer_name: signer_name.trim() }),
      { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  }

  // ─────────────────────────────────────────────
  // ACTION: refuse
  // ─────────────────────────────────────────────
  if (action === "refuse") {
    if (quote.status === "rifiutata") {
      return new Response(JSON.stringify({ error: "Già rifiutato" }), {
        status: 409, headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    if (quote.status === "accettata") {
      return new Response(JSON.stringify({ error: "Non puoi rifiutare un preventivo già firmato" }), {
        status: 422, headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    await supabase.from("quotes").update({
      status:         "rifiutata",
      refused_at:     new Date().toISOString(),
      refused_reason: refuse_reason?.trim() ?? null,
    }).eq("id", quote.id);

    // Notifica all'azienda
    try {
      await supabase.rpc("create_notification", {
        p_company_id: quote.company_id,
        p_type:       "quote",
        p_title:      "Preventivo rifiutato",
        p_body:       `L'offerta ${quote.quote_number} è stata rifiutata${refuse_reason ? `: ${refuse_reason}` : ""}`,
        p_link:       `/azienda/marketing/preventivi/${quote.id}`,
      });
    } catch (_) { /* notifiche opzionali */ }

    return new Response(
      JSON.stringify({ success: true }),
      { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  }

  return new Response(JSON.stringify({ error: "Action non gestita" }), {
    status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" },
  });
});
```

```typescript
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { supabase } from "@/lib/supabaseClient";
import { toast } from "sonner";

interface SendParams {
  quoteId:        string;
  recipientEmail: string;
  recipientName?: string;
  customMessage?: string;
  expiresDays?:   number;
}

export function useSignatureActions(companyId: string) {
  const queryClient = useQueryClient();

  const sendForSignature = useMutation({
    mutationFn: async (params: SendParams) => {
      const { data: { session } } = await supabase.auth.getSession();
      if (!session) throw new Error("Non autenticato");

      const res = await fetch(
        `${import.meta.env.VITE_SUPABASE_URL}/functions/v1/send-quote-signature`,
        {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${session.access_token}`,
            "apikey": import.meta.env.VITE_SUPABASE_ANON_KEY,
          },
          body: JSON.stringify({
            quote_id:       params.quoteId,
            recipient_email: params.recipientEmail,
            recipient_name:  params.recipientName,
            custom_message:  params.customMessage,
            expires_days:    params.expiresDays ?? 30,
          }),
        }
      );

      const body = await res.json();
      if (!res.ok) throw new Error(body.error ?? "Errore durante l'invio");
      return body as { success: boolean; signature_link: string; token: string };
    },
    onSuccess: (_, variables) => {
      toast.success("Preventivo inviato per firma!");
      queryClient.invalidateQueries({ queryKey: ["quote", variables.quoteId] });
      queryClient.invalidateQueries({ queryKey: ["quotes", companyId] });
    },
    onError: (err: Error) => toast.error(err.message),
  });

  return { sendForSignature };
}
```

Crea src/components/marketing/preventivi/QuoteSignatureStatusCard.tsx:

```tsx
import React from "react";
import { CheckCircle, Clock, XCircle, Send, Eye } from "lucide-react";
import { Badge } from "@/components/ui/badge";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";

interface Quote {
  status:         string;
  signature_link?: string;
  sent_at?:        string;
  viewed_at?:      string;
  signed_at?:      string;
  signed_by_name?: string;
  refused_at?:     string;
  refused_reason?: string;
  expires_at?:     string;
}

function fmt(iso?: string) {
  if (!iso) return "—";
  return new Date(iso).toLocaleDateString("it-IT", { day: "2-digit", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit" });
}

export function QuoteSignatureStatusCard({ quote }: { quote: Quote }) {
  if (!quote.sent_at && !quote.signed_at) return null;

  const isExpired = quote.expires_at && new Date(quote.expires_at) < new Date()
    && quote.status !== "accettata" && quote.status !== "rifiutata";

  return (
    <Card className="border-0 shadow-sm">
      <CardHeader className="pb-2">
        <CardTitle className="text-sm font-semibold text-gray-600 flex items-center gap-2">
          <Send className="w-4 h-4" /> Stato Firma Digitale
        </CardTitle>
      </CardHeader>
      <CardContent className="space-y-2">

        {/* Timeline */}
        <div className="space-y-2 text-sm">

          {quote.sent_at && (
            <div className="flex items-start gap-3">
              <div className="w-5 h-5 rounded-full bg-blue-100 flex items-center justify-center flex-shrink-0 mt-0.5">
                <Send className="w-3 h-3 text-blue-600" />
              </div>
              <div>
                <p className="font-medium text-gray-700">Link inviato</p>
                <p className="text-gray-400 text-xs">{fmt(quote.sent_at)}</p>
                {isExpired && (
                  <Badge variant="outline" className="text-amber-600 border-amber-300 text-xs mt-1">
                    Link scaduto
                  </Badge>
                )}
              </div>
            </div>
          )}

          {quote.viewed_at && (
            <div className="flex items-start gap-3">
              <div className="w-5 h-5 rounded-full bg-purple-100 flex items-center justify-center flex-shrink-0 mt-0.5">
                <Eye className="w-3 h-3 text-purple-600" />
              </div>
              <div>
                <p className="font-medium text-gray-700">Visualizzato dal cliente</p>
                <p className="text-gray-400 text-xs">{fmt(quote.viewed_at)}</p>
              </div>
            </div>
          )}

          {quote.signed_at && (
            <div className="flex items-start gap-3">
              <div className="w-5 h-5 rounded-full bg-green-100 flex items-center justify-center flex-shrink-0 mt-0.5">
                <CheckCircle className="w-3 h-3 text-green-600" />
              </div>
              <div>
                <p className="font-medium text-green-700">
                  Firmato da <strong>{quote.signed_by_name}</strong>
                </p>
                <p className="text-gray-400 text-xs">{fmt(quote.signed_at)}</p>
              </div>
            </div>
          )}

          {quote.refused_at && (
            <div className="flex items-start gap-3">
              <div className="w-5 h-5 rounded-full bg-red-100 flex items-center justify-center flex-shrink-0 mt-0.5">
                <XCircle className="w-3 h-3 text-red-500" />
              </div>
              <div>
                <p className="font-medium text-red-600">Offerta rifiutata</p>
                <p className="text-gray-400 text-xs">{fmt(quote.refused_at)}</p>
                {quote.refused_reason && (
                  <p className="text-gray-500 text-xs mt-0.5 italic">"{quote.refused_reason}"</p>
                )}
              </div>
            </div>
          )}

          {/* Pending: inviato ma non ancora azione */}
          {quote.sent_at && !quote.signed_at && !quote.refused_at && !isExpired && (
            <div className="flex items-start gap-3">
              <div className="w-5 h-5 rounded-full bg-amber-100 flex items-center justify-center flex-shrink-0 mt-0.5">
                <Clock className="w-3 h-3 text-amber-500" />
              </div>
              <div>
                <p className="font-medium text-amber-700">In attesa di firma</p>
                {quote.expires_at && (
                  <p className="text-gray-400 text-xs">Scade il {fmt(quote.expires_at)}</p>
                )}
              </div>
            </div>
          )}

        </div>

        {/* Link diretto (copiabile) */}
        {quote.signature_link && !quote.signed_at && !quote.refused_at && !isExpired && (
          <div className="pt-2 border-t border-gray-100">
            <p className="text-xs text-gray-400 mb-1">Link firma cliente</p>
            <div className="flex items-center gap-2">
              <p className="text-xs text-gray-500 truncate flex-1 font-mono">
                {quote.signature_link}
              </p>
              <button
                onClick={() => navigator.clipboard.writeText(quote.signature_link!)}
                className="text-xs text-blue-600 hover:text-blue-800 flex-shrink-0"
              >
                Copia
              </button>
            </div>
          </div>
        )}

      </CardContent>
    </Card>
  );
}
```

NOTE IMPORTANTI PER L'IMPLEMENTAZIONE
================================================================================

1. ORDINE DI ESECUZIONE (obbligatorio):
   FD1 → FD2 → FD3 → FD4 → FD5
   Non saltare passi. Le edge functions dipendono dal DB (FD1).

2. VARIABILI D'AMBIENTE necessarie in Supabase → Edge Functions → Secrets:
   - RESEND_API_KEY      → per invio email (obbligatorio per FD2)
   - SITE_URL            → es. https://app.ediliziacloud.it (obbligatorio per FD2)
   - SUPABASE_URL        → auto-disponibile nelle edge functions
   - SUPABASE_SERVICE_ROLE_KEY → auto-disponibile nelle edge functions

3. VARIABILI VITE nel frontend (file .env.local):
   - VITE_SUPABASE_URL
   - VITE_SUPABASE_ANON_KEY
   Queste vengono usate in QuoteSignPage.tsx per chiamare le edge functions.

4. ROUTE PUBBLICA: la route /offerta/:token DEVE stare fuori
   dall'AuthGuard/ProtectedRoute in App.tsx.
   Aggiungila prima del blocco delle route protette.

5. FIRMA ELETTRONICA SEMPLICE (art. 3 Reg. EU 910/2014):
   Il sistema registra: nome firmatario + IP + data/ora UTC.
   Sufficiente per la validità legale in Italia come firma elettronica semplice.
   Non serve firma grafica/canvas per questo tipo di firma.

6. STATES BADGE nella lista preventivi:
   - bozza    → Badge grigio
   - inviata  → Badge blu (in attesa di firma)
   - accettata → Badge verde
   - rifiutata → Badge rosso
   - scaduta  → Badge ambra

7. SCADENZA AUTOMATICA: aggiorna anche il cron job check-lifecycle-events
   per marcare come 'scaduta' le quotes con valid_until < today AND status='inviata'.
