Crea questa migrazione Supabase per Scadenzario e Prima Nota.
NON toccare le tabelle esistenti `orders`, `order_items`, `suppliers`, `company_costs`.
Usa quelle tramite JOIN/VIEW nelle query.

```sql
-- ════════════════════════════════════════════════════════════════
-- TABELLA 1: prima_nota_entries
-- Registro cronologico di tutte le movimentazioni finanziarie effettive.
-- È il "libro mastro semplificato": ogni riga è una transazione reale
-- (non previsionale). Il saldo si calcola come running total.
-- ════════════════════════════════════════════════════════════════
CREATE TABLE prima_nota_entries (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,

  -- Tipo e direzione
  direction TEXT NOT NULL CHECK (direction IN ('entrata', 'uscita')),
  category TEXT NOT NULL,
  -- Categorie entrata: 'incasso_ordine' | 'acconto_ordine' | 'altro_entrata'
  -- Categorie uscita:  'pagamento_fornitore' | 'costo_aziendale' | 'stipendio' |
  --                    'provvigione' | 'tasse' | 'altro_uscita'

  description TEXT NOT NULL,
  amount NUMERIC(12, 2) NOT NULL CHECK (amount > 0),
  entry_date DATE NOT NULL,

  -- Collegamento opzionale a entità esistenti
  order_id UUID REFERENCES orders(id) ON DELETE SET NULL,
  order_item_id UUID REFERENCES order_items(id) ON DELETE SET NULL,
  supplier_id UUID REFERENCES suppliers(id) ON DELETE SET NULL,
  company_cost_id UUID REFERENCES company_costs(id) ON DELETE SET NULL,

  -- Metodo di pagamento
  payment_method TEXT,
  -- 'bonifico' | 'contanti' | 'carta' | 'riba' | 'assegno' | 'altro'

  -- Conto bancario / cassa (facoltativo, testo libero per ora)
  account_label TEXT,  -- es. "Conto BancaXYZ", "Cassa"

  -- Note libere
  notes TEXT,

  -- Allegato (ricevuta, estratto conto)
  attachment_url TEXT,

  -- Creato manualmente o generato automaticamente
  is_auto_generated BOOLEAN NOT NULL DEFAULT false,
  auto_source TEXT,  -- 'order_payment' | 'supplier_payment' | 'company_cost'

  created_by UUID REFERENCES auth.users(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- ════════════════════════════════════════════════════════════════
-- TABELLA 2: scadenze
-- Scadenze finanziarie future pianificate (non ancora pagate/incassate).
-- Aggrega in modo unificato: rate da incassare dai clienti +
-- pagamenti da fare ai fornitori + costi aziendali ricorrenti.
-- Quando una scadenza viene saldata → si crea una prima_nota_entry.
-- ════════════════════════════════════════════════════════════════
CREATE TABLE scadenze (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,

  -- Tipo di scadenza
  type TEXT NOT NULL CHECK (type IN (
    'incasso_cliente',     -- rata da ricevere da un cliente
    'pagamento_fornitore', -- rata da pagare a un fornitore
    'costo_aziendale',     -- costo fisso/variabile aziendale
    'scadenza_fiscale',    -- IVA, F24, dichiarazione ecc.
    'altro'
  )),
  direction TEXT GENERATED ALWAYS AS (
    CASE
      WHEN type IN ('incasso_cliente') THEN 'entrata'
      ELSE 'uscita'
    END
  ) STORED,

  description TEXT NOT NULL,
  amount NUMERIC(12, 2) NOT NULL CHECK (amount > 0),
  due_date DATE NOT NULL,

  -- Stato
  status TEXT NOT NULL DEFAULT 'pending'
    CHECK (status IN ('pending', 'paid', 'overdue', 'cancelled')),
  paid_date DATE,
  paid_amount NUMERIC(12, 2),  -- può differire da amount (pagamento parziale)

  -- Collegamento a entità esistenti
  order_id UUID REFERENCES orders(id) ON DELETE SET NULL,
  order_item_id UUID REFERENCES order_items(id) ON DELETE SET NULL,
  supplier_id UUID REFERENCES suppliers(id) ON DELETE SET NULL,
  company_cost_id UUID REFERENCES company_costs(id) ON DELETE SET NULL,
  prima_nota_entry_id UUID REFERENCES prima_nota_entries(id) ON DELETE SET NULL,

  -- Metadati
  notes TEXT,
  is_auto_generated BOOLEAN NOT NULL DEFAULT false,
  auto_source TEXT,

  -- Alert: giorni di preavviso per notifica
  alert_days_before INT DEFAULT 7,
  alert_sent_at TIMESTAMPTZ,

  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- ════════════════════════════════════════════════════════════════
-- INDICI
-- ════════════════════════════════════════════════════════════════
CREATE INDEX idx_prima_nota_company ON prima_nota_entries(company_id);
CREATE INDEX idx_prima_nota_date ON prima_nota_entries(company_id, entry_date DESC);
CREATE INDEX idx_prima_nota_direction ON prima_nota_entries(company_id, direction);
CREATE INDEX idx_prima_nota_order ON prima_nota_entries(order_id) WHERE order_id IS NOT NULL;

CREATE INDEX idx_scadenze_company ON scadenze(company_id);
CREATE INDEX idx_scadenze_due_date ON scadenze(company_id, due_date);
CREATE INDEX idx_scadenze_status ON scadenze(company_id, status);
CREATE INDEX idx_scadenze_type ON scadenze(company_id, type);

-- ════════════════════════════════════════════════════════════════
-- FUNZIONE: mark_scadenza_paid(scadenza_id, paid_date, paid_amount, payment_method, notes)
-- Segna una scadenza come pagata e crea automaticamente la prima nota entry.
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION mark_scadenza_paid(
  p_scadenza_id UUID,
  p_paid_date DATE DEFAULT CURRENT_DATE,
  p_paid_amount NUMERIC DEFAULT NULL,
  p_payment_method TEXT DEFAULT 'bonifico',
  p_notes TEXT DEFAULT NULL,
  p_account_label TEXT DEFAULT NULL
)
RETURNS UUID LANGUAGE plpgsql AS $$
DECLARE
  v_sc scadenze%ROWTYPE;
  v_entry_id UUID;
  v_amount NUMERIC;
BEGIN
  SELECT * INTO v_sc FROM scadenze WHERE id = p_scadenza_id;
  IF NOT FOUND THEN RAISE EXCEPTION 'Scadenza non trovata'; END IF;

  v_amount := COALESCE(p_paid_amount, v_sc.amount);

  -- Crea prima nota entry
  INSERT INTO prima_nota_entries (
    company_id, direction, category, description, amount, entry_date,
    order_id, order_item_id, supplier_id, company_cost_id,
    payment_method, account_label, notes, is_auto_generated, auto_source
  ) VALUES (
    v_sc.company_id,
    v_sc.direction,
    CASE v_sc.type
      WHEN 'incasso_cliente'     THEN 'incasso_ordine'
      WHEN 'pagamento_fornitore' THEN 'pagamento_fornitore'
      WHEN 'costo_aziendale'     THEN 'costo_aziendale'
      WHEN 'scadenza_fiscale'    THEN 'tasse'
      ELSE 'altro_' || v_sc.direction
    END,
    v_sc.description,
    v_amount,
    p_paid_date,
    v_sc.order_id,
    v_sc.order_item_id,
    v_sc.supplier_id,
    v_sc.company_cost_id,
    p_payment_method,
    p_account_label,
    p_notes,
    true,
    'scadenza_paid'
  ) RETURNING id INTO v_entry_id;

  -- Aggiorna scadenza
  UPDATE scadenze SET
    status = 'paid',
    paid_date = p_paid_date,
    paid_amount = v_amount,
    prima_nota_entry_id = v_entry_id,
    updated_at = now()
  WHERE id = p_scadenza_id;

  RETURN v_entry_id;
END;
$$;

-- ════════════════════════════════════════════════════════════════
-- FUNZIONE: get_prima_nota_saldo(company_id, as_of_date)
-- Calcola entrate totali, uscite totali e saldo netto fino a una data.
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION get_prima_nota_saldo(
  p_company_id UUID,
  p_as_of DATE DEFAULT CURRENT_DATE,
  p_from_date DATE DEFAULT NULL
)
RETURNS TABLE (
  total_entrate NUMERIC,
  total_uscite NUMERIC,
  saldo NUMERIC,
  entry_count BIGINT
) LANGUAGE sql STABLE AS $$
  SELECT
    COALESCE(SUM(CASE WHEN direction = 'entrata' THEN amount ELSE 0 END), 0) AS total_entrate,
    COALESCE(SUM(CASE WHEN direction = 'uscita'  THEN amount ELSE 0 END), 0) AS total_uscite,
    COALESCE(SUM(CASE WHEN direction = 'entrata' THEN amount ELSE -amount END), 0) AS saldo,
    COUNT(*) AS entry_count
  FROM prima_nota_entries
  WHERE
    company_id = p_company_id
    AND entry_date <= p_as_of
    AND (p_from_date IS NULL OR entry_date >= p_from_date);
$$;

-- ════════════════════════════════════════════════════════════════
-- FUNZIONE: get_scadenzario_summary(company_id, days_ahead)
-- Per il Cruscotto: prossime scadenze con urgency grouping.
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION get_scadenzario_summary(
  p_company_id UUID,
  p_days_ahead INT DEFAULT 30
)
RETURNS TABLE (
  scadute_count BIGINT,
  scadute_amount NUMERIC,
  questa_settimana_count BIGINT,
  questa_settimana_amount NUMERIC,
  questo_mese_count BIGINT,
  questo_mese_amount NUMERIC,
  entrate_previste NUMERIC,
  uscite_previste NUMERIC
) LANGUAGE sql STABLE AS $$
  SELECT
    COUNT(*) FILTER (WHERE status = 'pending' AND due_date < CURRENT_DATE) AS scadute_count,
    COALESCE(SUM(amount) FILTER (WHERE status = 'pending' AND due_date < CURRENT_DATE), 0) AS scadute_amount,
    COUNT(*) FILTER (WHERE status = 'pending' AND due_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 7) AS questa_settimana_count,
    COALESCE(SUM(amount) FILTER (WHERE status = 'pending' AND due_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 7), 0) AS questa_settimana_amount,
    COUNT(*) FILTER (WHERE status = 'pending' AND due_date BETWEEN CURRENT_DATE AND CURRENT_DATE + p_days_ahead) AS questo_mese_count,
    COALESCE(SUM(amount) FILTER (WHERE status = 'pending' AND due_date BETWEEN CURRENT_DATE AND CURRENT_DATE + p_days_ahead), 0) AS questo_mese_amount,
    COALESCE(SUM(amount) FILTER (WHERE status = 'pending' AND direction = 'entrata' AND due_date BETWEEN CURRENT_DATE AND CURRENT_DATE + p_days_ahead), 0) AS entrate_previste,
    COALESCE(SUM(amount) FILTER (WHERE status = 'pending' AND direction = 'uscita' AND due_date BETWEEN CURRENT_DATE AND CURRENT_DATE + p_days_ahead), 0) AS uscite_previste
  FROM scadenze
  WHERE company_id = p_company_id;
$$;

-- ════════════════════════════════════════════════════════════════
-- TRIGGER: auto-aggiorna status a 'overdue' su lettura (via funzione)
-- Non serve un trigger continuo; invece usa un job cron o la funzione
-- check_overdue_scadenze() chiamata dalla edge function check-due-dates
-- (già esistente nel progetto).
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION check_overdue_scadenze(p_company_id UUID DEFAULT NULL)
RETURNS INT LANGUAGE plpgsql AS $$
DECLARE
  updated_count INT;
BEGIN
  UPDATE scadenze
  SET status = 'overdue', updated_at = now()
  WHERE
    status = 'pending'
    AND due_date < CURRENT_DATE
    AND (p_company_id IS NULL OR company_id = p_company_id);
  GET DIAGNOSTICS updated_count = ROW_COUNT;
  RETURN updated_count;
END;
$$;
```

```sql
-- ════════════════════════════════════════════════════════════════
-- ALTER TABLE suppliers: aggiungi campi operativi mancanti

  ADD COLUMN IF NOT EXISTS bank_name TEXT;

-- ════════════════════════════════════════════════════════════════
-- TABELLA: purchase_orders (Ordini di Acquisto)
-- Un OdA è il documento formale inviato a un fornitore per acquistare
-- materiali/servizi. Può nascere:
--   a) da un ordine cliente (collegato via order_id)
--   b) autonomamente (riordino magazzino, acquisto attrezzature, ecc.)
-- ════════════════════════════════════════════════════════════════
CREATE TABLE purchase_orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,

  -- Numero OdA auto-generato: ODA-2025-0001
  oda_number TEXT,

  -- Fornitore (obbligatorio)
  supplier_id UUID NOT NULL REFERENCES suppliers(id) ON DELETE RESTRICT,

  -- Collegamento opzionale a ordine cliente
  order_id UUID REFERENCES orders(id) ON DELETE SET NULL,

  -- Stato workflow
  status TEXT NOT NULL DEFAULT 'bozza'
    CHECK (status IN (
      'bozza',       -- creato ma non inviato
      'inviato',     -- inviato al fornitore
      'confermato',  -- fornitore ha confermato
      'parziale',    -- ricevuto parzialmente
      'ricevuto',    -- tutto ricevuto
      'annullato'    -- annullato
    )),

  -- Date chiave
  issue_date DATE NOT NULL DEFAULT CURRENT_DATE,
  expected_delivery_date DATE,
  actual_delivery_date DATE,

  -- Totali (calcolati da trigger)
  subtotal NUMERIC(12, 2) NOT NULL DEFAULT 0,
  vat_amount NUMERIC(12, 2) NOT NULL DEFAULT 0,
  total_amount NUMERIC(12, 2) NOT NULL DEFAULT 0,

  -- Condizioni di pagamento
  payment_method TEXT,
  payment_terms TEXT,  -- es. "60 gg data fattura"

  -- Note e riferimenti
  notes TEXT,
  supplier_reference TEXT,  -- codice ordine del fornitore
  delivery_address TEXT,

  -- Chi ha creato/modificato
  created_by UUID REFERENCES auth.users(id) ON DELETE SET NULL,
  sent_at TIMESTAMPTZ,
  confirmed_at TIMESTAMPTZ,

  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- ════════════════════════════════════════════════════════════════
-- TABELLA: purchase_order_items (Righe dell'OdA)
-- ════════════════════════════════════════════════════════════════
CREATE TABLE purchase_order_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  purchase_order_id UUID NOT NULL REFERENCES purchase_orders(id) ON DELETE CASCADE,
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,

  -- Collegamento opzionale all'articolo dell'ordine cliente
  order_item_id UUID REFERENCES order_items(id) ON DELETE SET NULL,

  -- Descrizione
  name TEXT NOT NULL,
  description TEXT,
  quantity NUMERIC(10, 3) NOT NULL DEFAULT 1,
  unit TEXT DEFAULT 'pz',  -- 'pz' | 'm2' | 'ml' | 'kg' | 'h'
  unit_price NUMERIC(12, 4) NOT NULL DEFAULT 0,
  vat_rate NUMERIC(5, 2) NOT NULL DEFAULT 22,
  discount_percent NUMERIC(5, 2) DEFAULT 0,

  -- Calcolati
  line_total NUMERIC(12, 2) GENERATED ALWAYS AS (
    ROUND(quantity * unit_price * (1 - COALESCE(discount_percent, 0) / 100), 2)
  ) STORED,
  vat_line NUMERIC(12, 2) GENERATED ALWAYS AS (
    ROUND(quantity * unit_price * (1 - COALESCE(discount_percent, 0) / 100) * vat_rate / 100, 2)
  ) STORED,

  -- Ricezione parziale
  quantity_received NUMERIC(10, 3) DEFAULT 0,
  received_at TIMESTAMPTZ,

  position INT NOT NULL DEFAULT 0,
  notes TEXT,

  created_at TIMESTAMPTZ DEFAULT now()
);

-- ════════════════════════════════════════════════════════════════
-- TRIGGER: ricalcola totali su purchase_orders
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION recalculate_purchase_order_totals()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  UPDATE purchase_orders
  SET
    subtotal = (
      SELECT COALESCE(SUM(line_total), 0) FROM purchase_order_items
      WHERE purchase_order_id = COALESCE(NEW.purchase_order_id, OLD.purchase_order_id)
    ),
    vat_amount = (
      SELECT COALESCE(SUM(vat_line), 0) FROM purchase_order_items
      WHERE purchase_order_id = COALESCE(NEW.purchase_order_id, OLD.purchase_order_id)
    ),
    total_amount = (
      SELECT COALESCE(SUM(line_total + vat_line), 0) FROM purchase_order_items
      WHERE purchase_order_id = COALESCE(NEW.purchase_order_id, OLD.purchase_order_id)
    ),
    updated_at = now()
  WHERE id = COALESCE(NEW.purchase_order_id, OLD.purchase_order_id);
  RETURN COALESCE(NEW, OLD);
END;
$$;

CREATE TRIGGER trg_purchase_order_totals
AFTER INSERT OR UPDATE OR DELETE ON purchase_order_items
FOR EACH ROW EXECUTE FUNCTION recalculate_purchase_order_totals();

-- ════════════════════════════════════════════════════════════════
-- FUNZIONE: generate_oda_number(company_id)
-- Genera il prossimo numero OdA: ODA-YYYY-NNNN
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION generate_oda_number(p_company_id UUID)
RETURNS TEXT LANGUAGE plpgsql AS $$
DECLARE
  v_year TEXT := to_char(CURRENT_DATE, 'YYYY');
  v_count INT;
BEGIN
  PERFORM pg_advisory_xact_lock(hashtext(p_company_id::text || v_year || '_oda'));
  SELECT COUNT(*) + 1 INTO v_count
  FROM purchase_orders
  WHERE company_id = p_company_id
    AND to_char(issue_date, 'YYYY') = v_year;
  RETURN 'ODA-' || v_year || '-' || lpad(v_count::text, 4, '0');
END;
$$;

-- ════════════════════════════════════════════════════════════════
-- TRIGGER: assegna oda_number automaticamente all'INSERT
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION trg_assign_oda_number()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.oda_number IS NULL THEN
    NEW.oda_number := generate_oda_number(NEW.company_id);
  END IF;
  RETURN NEW;
END;
$$;

CREATE TRIGGER before_insert_purchase_order
BEFORE INSERT ON purchase_orders
FOR EACH ROW EXECUTE FUNCTION trg_assign_oda_number();

-- ════════════════════════════════════════════════════════════════
-- FUNZIONE: create_oda_from_order(order_id, supplier_id, item_ids[])
-- Crea un OdA a partire da articoli di un ordine cliente.
-- Gli articoli selezionati devono avere status 'da_ordinare'.
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION create_oda_from_order(
  p_order_id UUID,
  p_supplier_id UUID,
  p_item_ids UUID[],
  p_expected_delivery_date DATE DEFAULT NULL,
  p_notes TEXT DEFAULT NULL
)
RETURNS UUID LANGUAGE plpgsql AS $$
DECLARE
  v_company_id UUID;
  v_oda_id UUID;
  v_item order_items%ROWTYPE;
  v_supplier suppliers%ROWTYPE;
BEGIN
  -- Recupera company_id dall'ordine
  SELECT company_id INTO v_company_id FROM orders WHERE id = p_order_id;
  IF NOT FOUND THEN RAISE EXCEPTION 'Ordine non trovato'; END IF;

  -- Recupera fornitore
  SELECT * INTO v_supplier FROM suppliers WHERE id = p_supplier_id;
  IF NOT FOUND THEN RAISE EXCEPTION 'Fornitore non trovato'; END IF;

  -- Crea OdA
  INSERT INTO purchase_orders (
    company_id, supplier_id, order_id,
    status, issue_date, expected_delivery_date,
    payment_method, payment_terms, notes
  ) VALUES (
    v_company_id, p_supplier_id, p_order_id,
    'bozza', CURRENT_DATE, p_expected_delivery_date,
    v_supplier.payment_method, NULL, p_notes
  ) RETURNING id INTO v_oda_id;

  -- Aggiungi righe dagli order_items selezionati
  FOR v_item IN
    SELECT * FROM order_items
    WHERE id = ANY(p_item_ids) AND order_id = p_order_id
  LOOP
    INSERT INTO purchase_order_items (
      purchase_order_id, company_id, order_item_id,
      name, description, quantity, unit_price, vat_rate, position
    ) VALUES (
      v_oda_id, v_company_id, v_item.id,
      v_item.name, v_item.description, v_item.quantity,
      COALESCE(v_item.purchase_price, 0), COALESCE(v_item.vat_rate, 22),

    );

    -- Aggiorna status articolo a 'ordinato'
    UPDATE order_items SET status = 'ordinato' WHERE id = v_item.id;
  END LOOP;

  RETURN v_oda_id;
END;
$$;

  USING (company_id = (SELECT company_id FROM users WHERE id = auth.uid()));
```

### Hook: src/hooks/useScadenzario.ts

```typescript
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { supabase } from "@/lib/supabase";
import { useCompany } from "@/hooks/useCompany";
import { addDays, startOfDay } from "date-fns";

export function useScadenzario(filters?: {
  dateFrom?: Date;
  dateTo?: Date;
  type?: string;
  status?: string;
  search?: string;
}) {
  const { company } = useCompany();

  const { data: scadenze = [], isLoading } = useQuery({
    queryKey: ["scadenze", company?.id, filters],
    queryFn: async () => {
      let query = supabase
        .from("scadenze")
        .select(`
          *,
          orders(id, order_code, customer_name),
          suppliers(id, name),
          prima_nota_entries(id, entry_date)
        `)
        .eq("company_id", company!.id)
        .order("due_date", { ascending: true });

      if (filters?.dateFrom) query = query.gte("due_date", filters.dateFrom.toISOString().split("T")[0]);
      if (filters?.dateTo) query = query.lte("due_date", filters.dateTo.toISOString().split("T")[0]);
      if (filters?.type && filters.type !== "all") query = query.eq("type", filters.type);
      if (filters?.status && filters.status !== "all") query = query.eq("status", filters.status);
      if (filters?.search) query = query.ilike("description", `%${filters.search}%`);

      const { data, error } = await query;
      if (error) throw error;
      return data || [];
    },
    enabled: !!company?.id,
  });

  const { data: summary } = useQuery({
    queryKey: ["scadenzario-summary", company?.id],
    queryFn: async () => {
      const { data, error } = await supabase.rpc("get_scadenzario_summary", {
        p_company_id: company!.id,
        p_days_ahead: 30,
      });
      if (error) throw error;
      return data?.[0] || null;
    },
    enabled: !!company?.id,
  });

  const markPaidMutation = useMutation({
    mutationFn: async (params: {
      scadenzaId: string;
      paidDate: string;
      paidAmount: number;
      paymentMethod: string;
      notes?: string;
      accountLabel?: string;
    }) => {
      const { data, error } = await supabase.rpc("mark_scadenza_paid", {
        p_scadenza_id: params.scadenzaId,
        p_paid_date: params.paidDate,
        p_paid_amount: params.paidAmount,
        p_payment_method: params.paymentMethod,
        p_notes: params.notes || null,
        p_account_label: params.accountLabel || null,
      });
      if (error) throw error;
      return data;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["scadenze"] });
      queryClient.invalidateQueries({ queryKey: ["prima-nota"] });
      queryClient.invalidateQueries({ queryKey: ["scadenzario-summary"] });
    },
  });

  const queryClient = useQueryClient();
  return { scadenze, isLoading, summary, markPaidMutation };
}
```

```typescript
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { supabase } from "@/lib/supabase";
import { useCompany } from "@/hooks/useCompany";
import { startOfMonth, endOfMonth } from "date-fns";

export function usePrimaNota(filters?: {
  dateFrom?: Date;
  dateTo?: Date;
  direction?: string;
  category?: string;
  search?: string;
}) {
  const { company } = useCompany();
  const queryClient = useQueryClient();

  const { data: entries = [], isLoading } = useQuery({
    queryKey: ["prima-nota", company?.id, filters],
    queryFn: async () => {
      let query = supabase
        .from("prima_nota_entries")
        .select(`
          *,
          orders(id, order_code, customer_name),
          suppliers(id, name)
        `)
        .eq("company_id", company!.id)
        .order("entry_date", { ascending: false })
        .order("created_at", { ascending: false });

      const from = filters?.dateFrom || startOfMonth(new Date());
      const to = filters?.dateTo || endOfMonth(new Date());
      query = query.gte("entry_date", from.toISOString().split("T")[0]);
      query = query.lte("entry_date", to.toISOString().split("T")[0]);

      if (filters?.direction && filters.direction !== "all")
        query = query.eq("direction", filters.direction);
      if (filters?.category && filters.category !== "all")
        query = query.eq("category", filters.category);
      if (filters?.search)
        query = query.ilike("description", `%${filters.search}%`);

      const { data, error } = await query;
      if (error) throw error;
      return data || [];
    },
    enabled: !!company?.id,
  });

  const { data: saldo } = useQuery({
    queryKey: ["prima-nota-saldo", company?.id, filters?.dateFrom, filters?.dateTo],
    queryFn: async () => {
      const { data, error } = await supabase.rpc("get_prima_nota_saldo", {
        p_company_id: company!.id,
        p_as_of: (filters?.dateTo || endOfMonth(new Date())).toISOString().split("T")[0],
        p_from_date: (filters?.dateFrom || startOfMonth(new Date())).toISOString().split("T")[0],
      });
      if (error) throw error;
      return data?.[0] || { total_entrate: 0, total_uscite: 0, saldo: 0 };
    },
    enabled: !!company?.id,
  });

  const createMutation = useMutation({
    mutationFn: async (entry: Omit<typeof entries[0], "id" | "created_at" | "updated_at">) => {
      const { data, error } = await supabase
        .from("prima_nota_entries")
        .insert({ ...entry, company_id: company!.id })
        .select()
        .single();
      if (error) throw error;
      return data;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["prima-nota"] });
      queryClient.invalidateQueries({ queryKey: ["prima-nota-saldo"] });
    },
  });

  return { entries, isLoading, saldo, createMutation };
}
```

```typescript
import { useQuery } from "@tanstack/react-query";
import { supabase } from "@/lib/supabase";
import { useCompany } from "@/hooks/useCompany";

        .from("suppliers")
```
