ALTER TABLE public.audit_logs
  ADD COLUMN IF NOT EXISTS admin_user_id UUID REFERENCES auth.users(id),
  ADD COLUMN IF NOT EXISTS search_vector TSVECTOR;

-- Popola admin_user_id dai dati esistenti se c'è un campo simile
-- (adatta il nome della colonna se diverso nel tuo schema)
UPDATE public.audit_logs
  SET admin_user_id = (
    SELECT id FROM auth.users
    WHERE email = audit_logs.admin_email
    LIMIT 1
  )
WHERE admin_user_id IS NULL
  AND admin_email IS NOT NULL;

-- Crea indici per performance delle query filtrate
CREATE INDEX IF NOT EXISTS idx_audit_logs_created_at
  ON public.audit_logs (created_at DESC);

CREATE INDEX IF NOT EXISTS idx_audit_logs_admin_user_id
  ON public.audit_logs (admin_user_id, created_at DESC);

CREATE INDEX IF NOT EXISTS idx_audit_logs_action
  ON public.audit_logs (action, created_at DESC);

-- Funzione RPC per audit log paginato con filtri server-side
CREATE OR REPLACE FUNCTION public.get_audit_logs_paginated(
  p_limit        INT      DEFAULT 50,
  p_offset       INT      DEFAULT 0,
  p_action       TEXT     DEFAULT NULL,
  p_admin_id     UUID     DEFAULT NULL,
  p_date_from    DATE     DEFAULT NULL,
  p_date_to      DATE     DEFAULT NULL,
  p_search       TEXT     DEFAULT NULL
)
RETURNS TABLE (
  id            UUID,
  action        TEXT,
  admin_email   TEXT,
  admin_name    TEXT,
  admin_user_id UUID,
  details       JSONB,
  created_at    TIMESTAMPTZ,
  total_count   BIGINT
)
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
  RETURN QUERY
  SELECT
    al.id,
    al.action,
    al.admin_email,
    al.admin_name,
    al.admin_user_id,
    al.details,
    al.created_at,
    COUNT(*) OVER() AS total_count
  FROM public.audit_logs al
  WHERE
    (p_action   IS NULL OR al.action = p_action)
    AND (p_admin_id  IS NULL OR al.admin_user_id = p_admin_id)
    AND (p_date_from IS NULL OR al.created_at >= p_date_from::TIMESTAMPTZ)
    AND (p_date_to   IS NULL OR al.created_at <  (p_date_to + INTERVAL '1 day')::TIMESTAMPTZ)
    AND (p_search    IS NULL OR
         al.admin_email ILIKE '%' || p_search || '%' OR
         al.admin_name  ILIKE '%' || p_search || '%' OR
         al.action      ILIKE '%' || p_search || '%' OR
         al.details::TEXT ILIKE '%' || p_search || '%')
  ORDER BY al.created_at DESC
  LIMIT p_limit
  OFFSET p_offset;
END;
$$;

GRANT EXECUTE ON FUNCTION public.get_audit_logs_paginated TO authenticated;
```

export function useAuditLogs(
  filters: AuditLogFilters,
  page: number,
  pageSize: number = 50
) {
  const offset = (page - 1) * pageSize;

  return useQuery({
    queryKey: ["audit-logs", filters, page, pageSize],
    staleTime: 30_000,
    refetchInterval: 60_000,   // auto-refresh ogni 60s
    queryFn: async (): Promise<AuditLogPage> => {
      const { data, error } = await supabase.rpc("get_audit_logs_paginated", {
        p_limit:     pageSize,
        p_offset:    offset,
        p_action:    filters.action   ?? null,
        p_admin_id:  filters.adminId  ?? null,
        p_date_from: filters.dateFrom ?? null,
        p_date_to:   filters.dateTo   ?? null,
        p_search:    filters.search   || null,
      });

      if (error) throw error;
      if (!data || data.length === 0) return { entries: [], totalCount: 0 };

      const totalCount = Number(data[0].total_count ?? 0);
      const entries: AuditLogEntry[] = data.map((row: any) => ({
        id:          row.id,
        action:      row.action,
        adminEmail:  row.admin_email  ?? "",
        adminName:   row.admin_name   ?? "",
        adminUserId: row.admin_user_id ?? null,
        details:     row.details      ?? null,
        createdAt:   row.created_at,
      }));

      return { entries, totalCount };
    },
  });
}

export function exportToCsv(
  filename: string,
  headers: string[],
  rows: (string | number | null | undefined)[][]
): void {
  const escape = (val: string | number | null | undefined): string => {
    if (val == null) return "";
    const str = String(val).replace(/"/g, '""');
    return str.includes(",") || str.includes("\n") || str.includes('"')
      ? `"${str}"`
      : str;
  };

  const csvContent = [
    headers.map(escape).join(","),
    ...rows.map((row) => row.map(escape).join(",")),
  ].join("\n");

  const blob = new Blob(["\uFEFF" + csvContent], { type: "text/csv;charset=utf-8;" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = `${filename}.csv`;
  a.click();
  URL.revokeObjectURL(url);
}

              const p = start + i;
              return p <= totalPages ? (
                <Button
                  key={p}
                  variant={p === page ? "default" : "outline"}
                  size="icon"
                  className="h-8 w-8 text-xs"
                  onClick={() => setPage(p)}
                >
                  {p}
                </Button>
              ) : null;
            })}
            <Button
              variant="outline"
              size="icon"
              className="h-8 w-8"
              disabled={page >= totalPages}
              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
            >
              <ChevronRight className="h-4 w-4" />
            </Button>
          </div>
        </div>
      )}

      {/* Detail sheet */}
      <DetailSheet
        entry={selectedEntry}
        onClose={() => setSelectedEntry(null)}
      />
    </div>
  );
}
```
