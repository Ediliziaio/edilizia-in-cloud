import { useState } from "react"
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query"
import { supabase } from "@/integrations/supabase/client"
import { Button } from "@/components/ui/button"
import { Badge } from "@/components/ui/badge"
import { Card, CardContent } from "@/components/ui/card"
import { Avatar, AvatarFallback } from "@/components/ui/avatar"
import {
  Table, TableBody, TableCell, TableHead, TableHeader, TableRow
} from "@/components/ui/table"
import {
  DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuTrigger
} from "@/components/ui/dropdown-menu"
import { AlertDialog, AlertDialogAction, AlertDialogCancel,
  AlertDialogContent, AlertDialogDescription, AlertDialogFooter,
  AlertDialogHeader, AlertDialogTitle } from "@/components/ui/alert-dialog"
import { Shield, MoreVertical, Plus, KeyRound, Trash2,
  Settings2, UserCheck } from "lucide-react"
import { toast } from "sonner"
import { CreatePlatformUserDialog } from "./CreatePlatformUserDialog"
import { PlatformPermissionsDialog } from "./PlatformPermissionsDialog"
import { ResetPasswordDialog } from "./ResetPasswordDialog"
import { PLATFORM_ROLE_DESCRIPTIONS } from "@/types/auth"
import { formatDate } from "@/lib/formatters"

// Fetch team piattaforma (super_admin + platform_* roles)
async function fetchPlatformTeam() {
  const result = await supabase.functions.invoke("manage-platform-users", {
    body: { action: "list" }
  })
  if (result.error) throw result.error
  return result.data.users ?? []
}

export function PlatformTeamTab() {
  const queryClient = useQueryClient()
  const [createOpen, setCreateOpen] = useState(false)
  const [permissionsUser, setPermissionsUser] = useState<any>(null)
  const [resetUser, setResetUser] = useState<any>(null)
  const [deleteUser, setDeleteUser] = useState<any>(null)

  const { data: users = [], isLoading } = useQuery({
    queryKey: ["platform-team"],
    queryFn: fetchPlatformTeam,
    staleTime: 2 * 60 * 1000,
  })

  const deleteMutation = useMutation({
    mutationFn: async (userId: string) => {
      const result = await supabase.functions.invoke("manage-platform-users", {
        body: { action: "delete", userId }
      })
      if (result.error) throw result.error
    },
    onSuccess: () => {
      toast.success("Utente rimosso dal team")
      queryClient.invalidateQueries({ queryKey: ["platform-team"] })
      setDeleteUser(null)
    },
    onError: (e: any) => toast.error(e.message ?? "Errore nella rimozione"),
  })

  const getPlatformRoleInfo = (role: string) => {
    return PLATFORM_ROLE_DESCRIPTIONS[role as keyof typeof PLATFORM_ROLE_DESCRIPTIONS]
      ?? PLATFORM_ROLE_DESCRIPTIONS.custom
  }

  if (isLoading) {
    return <div className="py-8 text-center text-muted-foreground">Caricamento team...</div>
  }

  return (
    <div className="space-y-4">
      {/* Header con stat cards */}
      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
        {Object.entries(PLATFORM_ROLE_DESCRIPTIONS)
          .filter(([key]) => key !== 'custom')
          .map(([role, info]) => {
            const count = users.filter((u: any) => u.platform_role === role).length
            return (
              <Card key={role} className="border">
                <CardContent className="pt-4 pb-3">
                  <p className="text-xs text-muted-foreground">{info.label}</p>
                  <p className="text-2xl font-bold">{count}</p>
                </CardContent>
              </Card>
            )
          })}
      </div>

      {/* Bottone Aggiungi + Tabella */}
      <div className="flex justify-between items-center">
        <h2 className="text-sm font-medium text-muted-foreground">
          {users.length} member{users.length !== 1 ? 'i' : 'o'} del team
        </h2>
        <Button onClick={() => setCreateOpen(true)} size="sm" className="gap-1.5">
          <Plus className="h-4 w-4" />
          Aggiungi Membro
        </Button>
      </div>

      <div className="rounded-lg border overflow-hidden">
        <Table>
          <TableHeader>
            <TableRow>
              <TableHead>Membro</TableHead>
              <TableHead>Ruolo</TableHead>
              <TableHead>Permessi</TableHead>
              <TableHead>Aggiunto il</TableHead>
              <TableHead className="w-12" />
            </TableRow>
          </TableHeader>
          <TableBody>
            {users.length === 0 ? (
              <TableRow>
                <TableCell colSpan={5} className="text-center py-8 text-muted-foreground">
                  Nessun membro del team. Aggiungi il primo membro.
                </TableCell>
              </TableRow>
            ) : (
              users.map((user: any) => {
                const roleInfo = getPlatformRoleInfo(user.platform_role ?? 'custom')
                const isSuperAdmin = user.platform_role === 'super_admin'
                return (
                  <TableRow key={user.id}>
                    <TableCell>
                      <div className="flex items-center gap-3">
                        <Avatar className="h-8 w-8">
                          <AvatarFallback className="text-xs">
                            {user.first_name?.[0]}{user.last_name?.[0]}
                          </AvatarFallback>
                        </Avatar>
                        <div>
                          <p className="font-medium text-sm">
                            {user.first_name} {user.last_name}
                            {isSuperAdmin && (
                              <Shield className="h-3.5 w-3.5 inline ml-1 text-violet-600" />
                            )}
                          </p>
                          <p className="text-xs text-muted-foreground">{user.email}</p>
                        </div>
                      </div>
                    </TableCell>
                    <TableCell>
                      <Badge
                        variant="outline"
                        className={`text-xs ${roleInfo.color}`}
                      >
                        {roleInfo.label}
                      </Badge>
                    </TableCell>
                    <TableCell>
                      <div className="flex flex-wrap gap-1 max-w-[280px]">
                        {user.permissions && Object.entries(user.permissions)
                          .filter(([k, v]) => v === true && k.startsWith('can_'))
                          .map(([key]) => (
                            <Badge key={key} variant="secondary" className="text-[10px] px-1.5 py-0">
                              {key.replace('can_manage_', '').replace('can_view_', '📊 ')}
                            </Badge>
                          ))}
                      </div>
                    </TableCell>
                    <TableCell className="text-sm text-muted-foreground">
                      {formatDate(user.created_at)}
                    </TableCell>
                    <TableCell>
                      <DropdownMenu>
                        <DropdownMenuTrigger asChild>
                          <Button variant="ghost" size="icon" className="h-8 w-8">
                            <MoreVertical className="h-4 w-4" />
                          </Button>
                        </DropdownMenuTrigger>
                        <DropdownMenuContent align="end">
                          <DropdownMenuItem onClick={() => setPermissionsUser(user)}>
                            <Settings2 className="h-4 w-4 mr-2" />
                            Modifica permessi
                          </DropdownMenuItem>
                          <DropdownMenuItem onClick={() => setResetUser(user)}>
                            <KeyRound className="h-4 w-4 mr-2" />
                            Reset password
                          </DropdownMenuItem>
                          <DropdownMenuItem
                            className="text-destructive"
                            onClick={() => setDeleteUser(user)}
                          >
                            <Trash2 className="h-4 w-4 mr-2" />
                            Rimuovi dal team
                          </DropdownMenuItem>
                        </DropdownMenuContent>
                      </DropdownMenu>
                    </TableCell>
                  </TableRow>
                )
              })
            )}
          </TableBody>
        </Table>
      </div>

      {/* Dialogs */}
      <CreatePlatformUserDialog
        open={createOpen}
        onOpenChange={setCreateOpen}
        onCreated={() => queryClient.invalidateQueries({ queryKey: ["platform-team"] })}
      />

      {permissionsUser && (
        <PlatformPermissionsDialog
          user={permissionsUser}
          open={!!permissionsUser}
          onOpenChange={(o) => !o && setPermissionsUser(null)}
          onSaved={() => queryClient.invalidateQueries({ queryKey: ["platform-team"] })}
        />
      )}

      {resetUser && (
        <ResetPasswordDialog
          user={resetUser}
          open={!!resetUser}
          onOpenChange={(o) => !o && setResetUser(null)}
        />
      )}

      {deleteUser && (
        <AlertDialog open={!!deleteUser} onOpenChange={(o) => !o && setDeleteUser(null)}>
          <AlertDialogContent>
            <AlertDialogHeader>
              <AlertDialogTitle>Rimuovi membro dal team</AlertDialogTitle>
              <AlertDialogDescription>
                Stai per rimuovere <strong>{deleteUser.first_name} {deleteUser.last_name}</strong>
                {' '}({deleteUser.email}) dal team della piattaforma.
                Non potrà più accedere all'area admin.
              </AlertDialogDescription>
            </AlertDialogHeader>
            <AlertDialogFooter>
              <AlertDialogCancel>Annulla</AlertDialogCancel>
              <AlertDialogAction
                className="bg-destructive text-destructive-foreground"
                onClick={() => deleteMutation.mutate(deleteUser.id)}
              >
                Rimuovi
              </AlertDialogAction>
            </AlertDialogFooter>
          </AlertDialogContent>
        </AlertDialog>
      )}
    </div>
  )
}

function MultiCompanySwitcher() {
  const { multiCompanyAccesses, selectedMultiCompany, switchMultiCompany } = useAuth()

  if (!multiCompanyAccesses || multiCompanyAccesses.length <= 1) return null

  return (
    <DropdownMenu>
      <DropdownMenuTrigger asChild>
        <Button variant="outline" size="sm" className="gap-2 h-8 max-w-[200px]">
          {selectedMultiCompany?.logo_url ? (
            <img src={selectedMultiCompany.logo_url} className="h-4 w-4 rounded" />
          ) : (
            <div className="h-4 w-4 rounded bg-primary/20 flex items-center justify-center text-[8px] font-bold">
              {selectedMultiCompany?.name?.[0]}
            </div>
          )}
          <span className="truncate text-xs">{selectedMultiCompany?.name ?? 'Seleziona azienda'}</span>
          <ChevronDown className="h-3 w-3 ml-auto flex-shrink-0" />
        </Button>
      </DropdownMenuTrigger>
      <DropdownMenuContent align="start" className="w-64">
        <DropdownMenuLabel className="text-xs text-muted-foreground">
          Le tue aziende ({multiCompanyAccesses.length})
        </DropdownMenuLabel>
        <DropdownMenuSeparator />
        {multiCompanyAccesses.map((access) => (
          <DropdownMenuItem
            key={access.company_id}
            onClick={() => switchMultiCompany(access.company_id)}
            className={`gap-2 ${access.company_id === selectedMultiCompany?.id ? 'bg-accent' : ''}`}
          >
            <div className="h-6 w-6 rounded bg-primary/10 flex items-center justify-center text-xs font-bold flex-shrink-0">
              {(access.company as any)?.name?.[0]}
            </div>
            <div className="flex-1 min-w-0">
              <p className="text-sm truncate">{(access.company as any)?.name}</p>
              <p className="text-xs text-muted-foreground">
                {access.access_role === 'company_admin' ? 'Amministratore' : 'Staff'}
              </p>
            </div>
            {access.company_id === selectedMultiCompany?.id && (
              <Check className="h-4 w-4 text-primary flex-shrink-0" />
            )}
          </DropdownMenuItem>
        ))}
      </DropdownMenuContent>
    </DropdownMenu>
  )
}
