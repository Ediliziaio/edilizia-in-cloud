-- Tabella inviti per nuovi super admin (link one-time)
CREATE TABLE IF NOT EXISTS public.admin_invites (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email       TEXT NOT NULL,
  token       TEXT NOT NULL UNIQUE DEFAULT gen_random_uuid()::text,
  invited_by  UUID REFERENCES auth.users(id),
  permissions JSONB NOT NULL DEFAULT '{
    "can_manage_companies": false,
    "can_manage_plans": false,
    "can_manage_tickets": false,
    "can_manage_referrals": false,
    "can_manage_admins": false,
    "can_view_platform_stats": true
  }',
  accepted_at  TIMESTAMPTZ,
  expires_at   TIMESTAMPTZ NOT NULL DEFAULT now() + INTERVAL '7 days',
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now()
);

ALTER TABLE public.admin_invites ENABLE ROW LEVEL SECURITY;

-- Solo service_role gestisce gli inviti
CREATE POLICY "service_role_admin_invites"
  ON public.admin_invites FOR ALL
  USING (auth.role() = 'service_role');

-- Colonna per tracciare ultimo accesso
ALTER TABLE public.profiles
  ADD COLUMN IF NOT EXISTS last_login_at TIMESTAMPTZ;

// ─── HOOK: LISTA ADMIN ────────────────────────────────────────────────────────

export function useAdminTeam() {
  return useQuery({
    queryKey: ["admin-team"],
    staleTime: 60_000,
    queryFn: async (): Promise<AdminMember[]> => {
      // Recupera tutti i super_admin dalla tabella user_roles
      const { data: roles, error: rolesError } = await supabase
        .from("user_roles")
        .select("user_id")
        .eq("role", "super_admin");
      if (rolesError) throw rolesError;

      const userIds = (roles ?? []).map((r: any) => r.user_id);
      if (userIds.length === 0) return [];

      // Recupera profili
      const { data: profiles, error: profilesError } = await supabase
        .from("profiles")
        .select("id, first_name, last_name, avatar_url, last_login_at, created_at")
        .in("id", userIds);
      if (profilesError) throw profilesError;

      // Recupera permessi
      const { data: perms } = await supabase
        .from("super_admin_permissions")
        .select("*")
        .in("user_id", userIds);

      // Recupera sessioni attive
      const { data: sessions } = await supabase
        .from("admin_sessions")
        .select("user_id")
        .in("user_id", userIds);

      const permMap = new Map((perms ?? []).map((p: any) => [p.user_id, p]));
      const sessionCount = (sessions ?? []).reduce((acc: Map<string, number>, s: any) => {
        acc.set(s.user_id, (acc.get(s.user_id) ?? 0) + 1);
        return acc;
      }, new Map<string, number>());

      // Recupera email da auth (solo service_role — usiamo Edge Function per email)
      // In alternativa, l'email è in profiles se la copiamo lì. Per semplicità,
      // recuperiamo l'email dalla tabella profiles se disponibile, altrimenti placeholder
      return (profiles ?? []).map((p: any): AdminMember => {
        const perm = permMap.get(p.id);
        return {
          id: p.id,
          firstName: p.first_name ?? "",
          lastName:  p.last_name ?? "",
          email:     p.email ?? "",     // se c'è la colonna email in profiles
          avatarUrl: p.avatar_url ?? null,
          lastLoginAt: p.last_login_at ?? null,
          createdAt: p.created_at,
          activeSessions: sessionCount.get(p.id) ?? 0,
          permissions: perm
            ? {
                can_manage_companies:    perm.can_manage_companies,
                can_manage_plans:        perm.can_manage_plans,
                can_manage_tickets:      perm.can_manage_tickets,
                can_manage_referrals:    perm.can_manage_referrals,
                can_manage_admins:       perm.can_manage_admins,
                can_view_platform_stats: perm.can_view_platform_stats,
                allowed_company_ids:     perm.allowed_company_ids ?? null,
              }
            : null,
        };
      });
    },
  });
}

// ─── HOOK: AGGIORNA PERMESSI (inline) ────────────────────────────────────────

export function useUpdateAdminPermission() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async ({
      userId,
      key,
      value,
    }: {
      userId: string;
      key: keyof AdminPermissions;
      value: boolean;
    }) => {
      const { error } = await supabase.functions.invoke("update-admin-permissions", {
        body: { target_user_id: userId, permissions: { [key]: value } },
      });
      if (error) throw error;
    },
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ["admin-team"] }),
    onError: (err: any) =>
      toast.error("Errore aggiornamento permesso", { description: err.message }),
  });
}

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

serve(async (req) => {
  if (req.method === "OPTIONS") return new Response(null, { headers: corsHeaders });

  try {
    const authHeader = req.headers.get("Authorization");
    if (!authHeader) throw new Error("Missing auth");

    const supabase = createClient(
      Deno.env.get("SUPABASE_URL") ?? "",
      Deno.env.get("SUPABASE_ANON_KEY") ?? "",
      { global: { headers: { Authorization: authHeader } } }
    );
    const supabaseAdmin = createClient(
      Deno.env.get("SUPABASE_URL") ?? "",
      Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? ""
    );

    const { data: { user } } = await supabase.auth.getUser();
    if (!user) throw new Error("Unauthorized");

    // Verifica che chi invita sia super_admin con can_manage_admins
    const { data: perm } = await supabaseAdmin
      .from("super_admin_permissions")
      .select("can_manage_admins")
      .eq("user_id", user.id)
      .single();
    if (!(perm?.can_manage_admins ?? true)) throw new Error("Missing permission");

    const { email, permissions } = await req.json();
    if (!email) throw new Error("Email required");

    // Crea il record di invito
    const token = crypto.randomUUID();
    const { error: insertError } = await supabaseAdmin
      .from("admin_invites")
      .insert({
        email,
        token,
        invited_by: user.id,
        permissions,
        expires_at: new Date(Date.now() + 7 * 86400000).toISOString(),
      });
    if (insertError) throw insertError;

    // Costruisci link di invito
    const inviteUrl =
      `${Deno.env.get("SITE_URL") ?? "https://app.example.com"}/admin/accept-invite?token=${token}`;

    // Invia email tramite provider configurato
    // (riusa la logica del send-dunning-email o simile)
    // Per semplicità inviamo con Resend se la chiave è configurata
    const resendKey = Deno.env.get("RESEND_API_KEY");
    if (resendKey) {
      await fetch("https://api.resend.com/emails", {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${resendKey}`,
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          from: "no-reply@ediliziaincloud.it",
          to: [email],
          subject: "Sei stato invitato come Super Admin",
          html: `
            <h2>Invito Super Admin</h2>
            <p>Sei stato invitato ad accedere al pannello di amministrazione.</p>
            <p>Clicca il link qui sotto per accettare (valido 7 giorni):</p>
            <a href="${inviteUrl}" style="display:inline-block;padding:12px 24px;background:#0f172a;color:white;border-radius:8px;text-decoration:none;">
              Accetta invito
            </a>
            <p style="margin-top:16px;color:#64748b;font-size:14px;">
              Se non ti aspettavi questo invito, ignora questa email.
            </p>
          `,
        }),
      });
    }

    return new Response(
      JSON.stringify({ ok: true, inviteUrl }),
      { headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  } catch (err: any) {
    return new Response(
      JSON.stringify({ error: err.message }),
      { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  }
});

  const { data: members, isLoading } = useAdminTeam();
  const [search, setSearch] = useState("");
  const [showInvite, setShowInvite] = useState(false);
  const [toDelete, setToDelete] = useState<AdminMember | null>(null);
  const [resetTarget, setResetTarget] = useState<{ id: string; name: string } | null>(null);

  const { user } = useAuth();

  const filtered = (members ?? []).filter(
    (m) =>
      search === "" ||
      `${m.firstName} ${m.lastName}`.toLowerCase().includes(search.toLowerCase()) ||
      m.email.toLowerCase().includes(search.toLowerCase())
  );

  return (
    <div className="space-y-6">
      {/* Header */}
      <div className="flex items-center justify-between">
        <div>
          <h3 className="text-lg font-semibold">Team Super Admin</h3>
          <p className="text-sm text-muted-foreground mt-1">
            Gestisci gli amministratori e i loro permessi.
          </p>
        </div>
        <Button onClick={() => setShowInvite(true)} className="gap-2">
          <UserPlus className="h-4 w-4" />
          Invita Admin
        </Button>
      </div>

      {/* KPI strip */}
      <div className="grid grid-cols-3 gap-3">
        <Card>
          <CardContent className="pt-4 pb-3">
            <p className="text-xs text-muted-foreground">Totale Admin</p>
            <p className="text-2xl font-semibold">{members?.length ?? 0}</p>
          </CardContent>
        </Card>
        <Card>
          <CardContent className="pt-4 pb-3">
            <p className="text-xs text-muted-foreground">Sessioni attive</p>
            <p className="text-2xl font-semibold">
              {(members ?? []).reduce((s, m) => s + m.activeSessions, 0)}
            </p>
          </CardContent>
        </Card>
        <Card>
          <CardContent className="pt-4 pb-3">
            <p className="text-xs text-muted-foreground">Accesso completo</p>
            <p className="text-2xl font-semibold">
              {
                (members ?? []).filter((m) =>
                  !m.permissions ||
                  Object.values(m.permissions).every((v) =>
                    typeof v === "boolean" ? v : true
                  )
                ).length
              }
            </p>
          </CardContent>
        </Card>
      </div>

      {/* Ricerca */}
      <div className="relative">
        <Search className="absolute left-3 top-2.5 h-4 w-4 text-muted-foreground" />
        <Input
          placeholder="Cerca admin per nome o email..."
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          className="pl-9"
        />
      </div>

      {/* Matrix dei permessi */}
      <Card>
        <CardHeader>
          <CardTitle className="text-sm">Matrix permessi</CardTitle>
          <CardDescription>
            Abilita/disabilita permessi direttamente dalla tabella. Passa il mouse
            su una riga per applicare un preset rapido.
          </CardDescription>
        </CardHeader>
        <CardContent className="p-0">
          {isLoading ? (
            <div className="p-4 space-y-3">
              {[1, 2, 3].map((i) => (
                <Skeleton key={i} className="h-14 w-full" />
              ))}
            </div>
          ) : filtered.length === 0 ? (
            <div className="flex flex-col items-center gap-2 py-10 text-muted-foreground">
              <p className="text-sm">Nessun admin trovato</p>
            </div>
          ) : (
            <div className="overflow-x-auto">
              <Table>
                <TableHeader>
                  <TableRow>
                    <TableHead className="min-w-[200px]">Admin</TableHead>
                    {PERMISSION_LABELS.filter(p => p.key !== "allowed_company_ids").map(({ key, label, description }) => (
                      <TableHead key={key} className="text-center px-2 min-w-[80px]">
                        <TooltipProvider>
                          <Tooltip>
                            <TooltipTrigger asChild>
                              <span className="cursor-help text-xs">{label}</span>
                            </TooltipTrigger>
                            <TooltipContent>{description}</TooltipContent>
                          </Tooltip>
                        </TooltipProvider>
                      </TableHead>
                    ))}
                    <TableHead className="px-2">Preset</TableHead>
                    <TableHead className="text-right">Attività</TableHead>
                  </TableRow>
                </TableHeader>
                <TableBody>
                  {filtered.map((member) => (
                    <PermissionRow key={member.id} member={member} />
                  ))}
                </TableBody>
              </Table>
            </div>
          )}
        </CardContent>
      </Card>

      {/* Azioni per admin */}
      {!isLoading && filtered.length > 0 && (
        <Card>
          <CardHeader>
            <CardTitle className="text-sm">Azioni individuali</CardTitle>
          </CardHeader>
          <CardContent className="space-y-2">
            {filtered.map((member) => {
              const isSelf = member.id === user?.id;
              return (
                <div
                  key={member.id}
                  className="flex items-center justify-between py-2 border-b last:border-0"
                >
                  <div className="flex items-center gap-3">
                    <Avatar className="h-7 w-7">
                      <AvatarImage src={member.avatarUrl ?? undefined} />
                      <AvatarFallback className="text-xs">
                        {member.firstName[0]}{member.lastName[0]}
                      </AvatarFallback>
                    </Avatar>
                    <span className="text-sm">
                      {member.firstName} {member.lastName}
                      {isSelf && (
                        <Badge variant="outline" className="ml-2 h-4 px-1.5 text-[10px]">Tu</Badge>
                      )}
                    </span>
                  </div>
                  <div className="flex items-center gap-1">
                    <Button
                      variant="ghost"
                      size="sm"
                      className="gap-1 text-xs h-7"
                      onClick={() =>
                        setResetTarget({ id: member.id, name: `${member.firstName} ${member.lastName}` })
                      }
                    >
                      <Key className="h-3 w-3" />
                      Reset password
                    </Button>
                    {!isSelf && (
                      <Button
                        variant="ghost"
                        size="sm"
                        className="gap-1 text-xs h-7 text-destructive hover:text-destructive hover:bg-destructive/10"
                        onClick={() => setToDelete(member)}
                      >
                        <Trash2 className="h-3 w-3" />
                        Elimina
                      </Button>
                    )}
                  </div>
                </div>
              );
            })}
          </CardContent>
        </Card>
      )}

      {/* Dialogs */}
      <InviteAdminDialog open={showInvite} onClose={() => setShowInvite(false)} />
      <DeleteConfirmDialog member={toDelete} onClose={() => setToDelete(null)} />
      {resetTarget && (
        <ResetPasswordDialog
          open={!!resetTarget}
          onClose={() => setResetTarget(null)}
          targetUserId={resetTarget.id}
          targetName={resetTarget.name}
        />
      )}
    </div>
  );
}
