────────────────────────────────────────────────────────────────────────────────

Apri `src/hooks/useAdminRevenueData.ts`. Cerca le righe (circa 279-280) dove


Sostituisci con un calcolo reale basato sui movimenti di abbonamento.
Aggiungi questa funzione RPC SQL prima, poi usala nel hook:

```sql
-- Aggiungi questa funzione RPC nel SQL Editor di Supabase:
CREATE OR REPLACE FUNCTION public.get_mrr_movements_monthly(
  p_months INT DEFAULT 6
)
RETURNS TABLE (
  month          TEXT,
  new_mrr        NUMERIC,
  expansion_mrr  NUMERIC,
  contraction_mrr NUMERIC,
  churn_mrr      NUMERIC
)
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
  RETURN QUERY
  WITH monthly AS (
    SELECT
      TO_CHAR(DATE_TRUNC('month', sl.started_at), 'Mon YY') AS month,
      DATE_TRUNC('month', sl.started_at) AS month_date,
      SUM(
        CASE
          -- New MRR: prima volta che l'azienda passa a un piano a pagamento
          WHEN sl.previous_plan_id IS NULL AND sp.price > 0 THEN sp.price
          ELSE 0
        END
      ) AS new_mrr,
      SUM(
        CASE
          -- Expansion: upgrade a piano più costoso
          WHEN sl.previous_plan_id IS NOT NULL
               AND sp.price > prev_sp.price THEN sp.price - prev_sp.price
          ELSE 0
        END
      ) AS expansion_mrr,
      SUM(
        CASE
          -- Contraction: downgrade a piano meno costoso
          WHEN sl.previous_plan_id IS NOT NULL
               AND sp.price < prev_sp.price THEN prev_sp.price - sp.price
          ELSE 0
        END
      ) AS contraction_mrr,
      SUM(
        CASE
          -- Churn: cancellazione (passaggio a piano gratuito o null)
          WHEN sl.status = 'cancelled' THEN COALESCE(prev_sp.price, 0)
          ELSE 0
        END
      ) AS churn_mrr
    FROM public.subscription_logs sl
    LEFT JOIN public.subscription_plans sp      ON sp.id = sl.plan_id
    LEFT JOIN public.subscription_plans prev_sp ON prev_sp.id = sl.previous_plan_id
    WHERE sl.started_at >= NOW() - (p_months || ' months')::INTERVAL
    GROUP BY month_date, month
    ORDER BY month_date
  )
  SELECT
    month,
    COALESCE(new_mrr, 0)::NUMERIC,
    COALESCE(expansion_mrr, 0)::NUMERIC,
    COALESCE(contraction_mrr, 0)::NUMERIC,
    COALESCE(churn_mrr, 0)::NUMERIC
  FROM monthly;
END;
$$;

GRANT EXECUTE ON FUNCTION public.get_mrr_movements_monthly TO authenticated;
```

Poi nel file `src/hooks/useAdminRevenueData.ts`, sostituisci il blocco che

```typescript
// Sostituisci la query mrrMovements esistente con:
const { data: mrrMovData } = await supabase
  .rpc("get_mrr_movements_monthly", { p_months: 6 });

const mrrMovements = (mrrMovData ?? []).map((row: any) => ({
  month:          row.month,
  newMrr:         Number(row.new_mrr)         / 100,
  expansionMrr:   Number(row.expansion_mrr)   / 100,
  contractionMrr: Number(row.contraction_mrr) / 100,
  churnMrr:       Number(row.churn_mrr)       / 100,
  netNew: (
    Number(row.new_mrr) + Number(row.expansion_mrr) -
    Number(row.contraction_mrr) - Number(row.churn_mrr)
  ) / 100,
}));
```

```tsx
function usePlatformSettings() {
  return useQuery({
    queryKey: ["platform-settings"],
    staleTime: 300_000,
    queryFn: async () => {
      const { data, error } = await supabase
        .from("platform_settings")
        .select("*")
        .single();
      if (error) throw error;
      return data as PlatformSettings;
    },
  });
}

// ─── HOOK SALVA SETTINGS ──────────────────────────────────────────────────────

function useSavePlatformSettings() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (updates: Partial<PlatformSettings>) => {
      const { error } = await supabase.functions.invoke("update-platform-settings", {
        body: updates,
      });
      if (error) throw error;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["platform-settings"] });
      toast.success("Impostazioni salvate");
    },
    onError: (err: any) =>
      toast.error("Errore nel salvataggio", { description: err.message }),
  });
}

      variant="outline"
      size="sm"
```

```typescript
import Stripe from "https://esm.sh/stripe@14.21.0";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",

serve(async (req) => {

      Deno.env.get("SUPABASE_ANON_KEY") ?? "",
      { global: { headers: { Authorization: authHeader } } }

    const supabaseAdmin = createClient(
      Deno.env.get("SUPABASE_URL") ?? "",
      Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? ""
    );

    const { data: { user } } = await supabase.auth.getUser();
    if (!user) throw new Error("Unauthorized");

    // Verifica super_admin
    const { data: role } = await supabaseAdmin
      .from("user_roles")
      .select("role")
      .eq("user_id", user.id)
      .single();
    if (role?.role !== "super_admin") throw new Error("Forbidden");

    const { integration } = await req.json();

    // Recupera settings piattaforma
    const { data: settings } = await supabaseAdmin
      .from("platform_settings")
      .select("*")
      .single();

    let ok = false;
    let message = "";

    switch (integration) {
      case "stripe": {
        if (!settings?.stripe_secret_key) {
          message = "Stripe secret key non configurata";
          break;
        }
        try {
          const stripe = new Stripe(settings.stripe_secret_key, {
            apiVersion: "2023-10-16",
          });
          // Test semplice: recupera le impostazioni account
          await stripe.accounts.retrieve();
          ok = true;
          message = "Stripe connesso correttamente";
        } catch (e: any) {
          message = e.message;
        }
        break;
      }

      case "google_calendar": {
        if (!settings?.google_calendar_client_id) {
          message = "Client ID non configurato";
          break;
        }
        // Verifica che il Client ID sia un formato valido
        ok = settings.google_calendar_client_id.includes(".apps.googleusercontent.com");
        message = ok ? "Formato Client ID valido" : "Formato Client ID non valido";
        break;
      }

      case "google_maps": {
        if (!settings?.google_maps_api_key) {
          message = "API Key non configurata";
          break;
        }
        try {
          const res = await fetch(
            `https://maps.googleapis.com/maps/api/geocode/json?address=Roma&key=${settings.google_maps_api_key}`
          );
          const json = await res.json();
          ok = json.status === "OK";
          message = ok ? "Google Maps API funzionante" : `Errore: ${json.status}`;
        } catch (e: any) {
          message = e.message;
        }
        break;
      }

      default:
        message = "Integrazione sconosciuta";
    }

    return new Response(
      JSON.stringify({ ok, message }),
      { headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  } catch (err: any) {
    return new Response(
      JSON.stringify({ error: err.message }),
      { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  }
});
```
