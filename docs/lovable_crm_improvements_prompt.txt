```typescript
export interface DealHealthInput {
  daysSinceLastActivity: number | null  // gg dall'ultima nota/chiamata/email
  daysSinceStageChange: number          // gg nell'attuale stage
  hasNextActionSet: boolean             // campo next_action compilato
  hasExpectedCloseDate: boolean         // data chiusura prevista settata
  closeDate: Date | null                // se è passata (deal scaduto?)
  probabilityPercent: number            // 0-100
  numberOfActivities: number            // note + appuntamenti totali
  hasDecisionMaker: boolean            // contatto è decision maker?
  avgStageDays: number                  // media storica giorni in questo stage
}

export interface DealHealth {
  score: number           // 0-100
  status: 'healthy' | 'at_risk' | 'critical' | 'dead'
  badge: string           // testo breve da mostrare in UI
  color: string           // classe CSS tailwind
  reasons: string[]       // lista di motivazioni leggibili
}

export function calculateDealHealth(input: DealHealthInput): DealHealth {
  let score = 100
  const reasons: string[] = []

  // Attività recente (-30 se nessuna attività negli ultimi 7gg)
  if (input.daysSinceLastActivity === null || input.daysSinceLastActivity > 14) {
    score -= 30
    reasons.push('Nessuna attività recente')
  } else if (input.daysSinceLastActivity > 7) {
    score -= 15
    reasons.push('Attività scarsa (ultima > 7gg)')
  }

  // Stage stagnante (confronta con media storica)
  if (input.avgStageDays > 0 && input.daysSinceStageChange > input.avgStageDays * 1.5) {
    score -= 20
    reasons.push(`Bloccato in stage ${Math.round(input.daysSinceStageChange)}gg (media: ${Math.round(input.avgStageDays)}gg)`)
  }

  // Nessuna prossima azione settata
  if (!input.hasNextActionSet) {
    score -= 15
    reasons.push('Nessuna azione pianificata')
  }

  // Nessuna data di chiusura
  if (!input.hasExpectedCloseDate) {
    score -= 10
    reasons.push('Data chiusura non settata')
  }

  // Close date passata
  if (input.closeDate && input.closeDate < new Date()) {
    score -= 20
    reasons.push('Data chiusura scaduta!')
  }

  // Poche attività totali (deal non lavorato)
  if (input.numberOfActivities < 2) {
    score -= 10
    reasons.push('Meno di 2 contatti registrati')
  }

  score = Math.max(0, score)

  const status = score >= 70 ? 'healthy'
    : score >= 40 ? 'at_risk'
    : score >= 20 ? 'critical'
    : 'dead'

  const badge = status === 'healthy' ? 'In buona salute'
    : status === 'at_risk' ? 'A rischio'
    : status === 'critical' ? 'Critico'
    : 'Da recuperare'

  const color = status === 'healthy' ? 'text-emerald-600 bg-emerald-50'
    : status === 'at_risk' ? 'text-amber-600 bg-amber-50'
    : 'text-red-600 bg-red-50'

  return { score, status, badge, color, reasons }
}
```

```typescript
export type LeadTemperature = 'hot' | 'warm' | 'cold' | 'frozen'

export interface LeadTemperatureResult {
  temperature: LeadTemperature
  label: string              // "Caldo 🔥" / "Tiepido" / "Freddo" / "Inattivo ❄️"
  colorClass: string         // classe Tailwind
  decayDays: number          // giorni dall'ultima attività
  reason: string             // motivazione breve
}

export function getLeadTemperature(
  lastActivityDate: Date | null,
  lastContactedDate: Date | null,
  hasOpenOpportunity: boolean,
  recentAppointment: boolean      // appuntamento nelle ultime 48h
): LeadTemperatureResult {
  const now = new Date()

  // Se c'è un appuntamento recente o attività in 48h → CALDO
  if (recentAppointment) {
    return { temperature: 'hot', label: 'Caldo 🔥', colorClass: 'text-red-600 bg-red-50', decayDays: 0, reason: 'Appuntamento recente' }
  }

  const daysSinceActivity = lastActivityDate
    ? differenceInDays(now, lastActivityDate)
    : 999

  if (daysSinceActivity <= 3) {
    return { temperature: 'hot', label: 'Caldo 🔥', colorClass: 'text-red-600 bg-red-50', decayDays: daysSinceActivity, reason: `Attivo ${daysSinceActivity}gg fa` }
  }
  if (daysSinceActivity <= 14) {
    return { temperature: 'warm', label: 'Tiepido', colorClass: 'text-amber-600 bg-amber-50', decayDays: daysSinceActivity, reason: `Attivo ${daysSinceActivity}gg fa` }
  }
  if (daysSinceActivity <= 30) {
    return { temperature: 'cold', label: 'Freddo', colorClass: 'text-blue-600 bg-blue-50', decayDays: daysSinceActivity, reason: `Inattivo da ${daysSinceActivity}gg` }
  }
  return { temperature: 'frozen', label: 'Inattivo ❄️', colorClass: 'text-slate-500 bg-slate-100', decayDays: daysSinceActivity, reason: `Fermo da ${daysSinceActivity}gg` }
}
```

```typescript
export interface NextBestActionSuggestion {
  action: string          // testo dell'azione suggerita
  urgency: 'today' | 'this_week' | 'optional'
  icon: string            // emoji o nome icona
  reason: string          // perché questa azione
}

export function suggestNextAction(
  stageName: string,
  daysSinceLastActivity: number | null,
  hasScheduledAppointment: boolean,
  daysSinceCreated: number,
  lostDealsCount: number,           // quante volte si è perso in questo stage
  previousActions: string[]         // ultime azioni fatte
): NextBestActionSuggestion {
  const days = daysSinceLastActivity ?? 99

  // Logica per stage "Nuovo Lead" o primo contatto
  if (stageName.toLowerCase().includes('nuovo') || daysSinceCreated < 2) {
    if (days > 1) return {
      action: 'Chiama entro oggi — i lead freschi si raffreddano in 24h',
      urgency: 'today',
      icon: '📞',
      reason: 'Il tasso di risposta cala del 70% dopo 24h dal primo contatto'
    }
  }

  // Nessun appuntamento schedulato
  if (!hasScheduledAppointment && days > 3) {
    return {
      action: 'Proponi un appuntamento di 20 minuti',
      urgency: days > 7 ? 'today' : 'this_week',
      icon: '📅',
      reason: 'Senza appuntamento fissato le probabilità di chiusura sono basse'
    }
  }

  // Deal fermo da troppo
  if (days > 14) {
    return {
      action: 'Contatto di rottura — offri qualcosa di nuovo o chiudi il deal',
      urgency: 'today',
      icon: '⚡',
      reason: `Fermo da ${days} giorni: o si sblocca o si qualifica come perso`
    }
  }

  // Default: follow-up
  return {
    action: 'Follow-up: aggiorna sullo stato della trattativa',
    urgency: 'this_week',
    icon: '💬',
    reason: 'Mantieni vivo il contatto'
  }
}
```

```typescript
      if (error) throw error
      return data
    },
    enabled: !!companyId,
    staleTime: 15 * 60 * 1000,
  })
}
```
