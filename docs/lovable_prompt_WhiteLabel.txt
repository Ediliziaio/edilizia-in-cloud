import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { supabase } from "@/integrations/supabase/client";
import type { CompanyBranding } from "@/types/branding";

/** Carica il branding dell'azienda corrente (per company admin) */
export function useCompanyBranding(companyId: string | undefined) {
  return useQuery({
    queryKey: ["company-branding", companyId],
    queryFn: async () => {
      if (!companyId) return null;
      const { data, error } = await supabase
        .from("company_branding")
        .select("*")
        .eq("company_id", companyId)
        .maybeSingle();
      if (error) throw error;
      return data as CompanyBranding | null;
    },
    enabled: !!companyId,
  });
}

/** Carica il branding dal subdomain o custom domain (per login page pubblica) */
export function useBrandingByDomain(domain: string | null) {
  return useQuery({
    queryKey: ["branding-by-domain", domain],
    queryFn: async () => {
      if (!domain) return null;
      // Prima cerca su custom_domain verificato
      const { data: byCustom } = await supabase
        .from("company_branding")
        .select("*")
        .eq("custom_domain", domain)
        .eq("custom_domain_verified", true)
        .eq("is_active", true)
        .maybeSingle();
      if (byCustom) return byCustom as CompanyBranding;

      // Poi cerca su subdomain (es. "rossi" dal hostname "rossi.platform.com")
      const subdomain = domain.split(".")[0];
      const { data: bySub } = await supabase
        .from("company_branding")
        .select("*")
        .eq("subdomain", subdomain)
        .eq("is_active", true)
        .maybeSingle();
      return (bySub as CompanyBranding) || null;
    },
    enabled: !!domain,
  });
}

export function useSaveCompanyBranding(companyId: string | undefined) {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (data: Partial<CompanyBranding>) => {
      if (!companyId) throw new Error("companyId richiesto");
      const { error } = await supabase
        .from("company_branding")
        .upsert(
          { company_id: companyId, ...data, updated_at: new Date().toISOString() },
          { onConflict: "company_id" }
        );
      if (error) throw error;
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ["company-branding", companyId] });
    },
  });
}

export function useRequestDomainVerification(companyId: string | undefined) {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (customDomain: string) => {
      if (!companyId) throw new Error("companyId richiesto");
      // Genera un CNAME target univoco per la verifica
      const cnameTarget = `verify-${companyId.slice(0, 8)}.platform-verify.com`;
      const { error } = await supabase
        .from("company_branding")
        .upsert({
          company_id: companyId,
          custom_domain: customDomain,
          custom_domain_cname: cnameTarget,
          custom_domain_verified: false,
          updated_at: new Date().toISOString(),
        }, { onConflict: "company_id" });
      if (error) throw error;
      return cnameTarget;
    },
    onSuccess: () => qc.invalidateQueries({ queryKey: ["company-branding", companyId] }),
  });
}

export function useVerifyCustomDomain(companyId: string | undefined) {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async () => {
      // Chiama Edge Function che verifica il CNAME DNS
      const { data, error } = await supabase.functions.invoke("verify-custom-domain", {
        body: { company_id: companyId },
      });
      if (error) throw error;
      return data as { verified: boolean; error?: string };
    },
    onSuccess: () => qc.invalidateQueries({ queryKey: ["company-branding", companyId] }),
  });
}

```typescript
import React, { createContext, useContext, useEffect } from "react";
import type { BrandingTheme } from "@/types/branding";
import { DEFAULT_BRANDING } from "@/types/branding";

interface BrandingContextType {
  theme: BrandingTheme;
  isWhiteLabel: boolean;
}

const BrandingContext = createContext<BrandingContextType>({
  theme: DEFAULT_BRANDING,
  isWhiteLabel: false,
});

export function BrandingProvider({
  children,
  branding,
}: {
  children: React.ReactNode;
  branding?: BrandingTheme | null;
}) {
  const theme = branding || DEFAULT_BRANDING;
  const isWhiteLabel = !!branding && branding.platformName !== DEFAULT_BRANDING.platformName;

  // Applica CSS custom properties per il colore primario
  useEffect(() => {
    if (!branding?.primaryColor) return;
    const hex = branding.primaryColor.replace("#", "");
    const r = parseInt(hex.substring(0, 2), 16);
    const g = parseInt(hex.substring(2, 4), 16);
    const b = parseInt(hex.substring(4, 6), 16);

    // Converti in HSL per Shadcn/Tailwind
    const rNorm = r / 255, gNorm = g / 255, bNorm = b / 255;
    const max = Math.max(rNorm, gNorm, bNorm);
    const min = Math.min(rNorm, gNorm, bNorm);
    let h = 0, s = 0;
    const l = (max + min) / 2;
    if (max !== min) {
      const d = max - min;
      s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
      switch (max) {
        case rNorm: h = ((gNorm - bNorm) / d + (gNorm < bNorm ? 6 : 0)) / 6; break;
        case gNorm: h = ((bNorm - rNorm) / d + 2) / 6; break;
        case bNorm: h = ((rNorm - gNorm) / d + 4) / 6; break;
      }
    }
    const hslStr = `${Math.round(h * 360)} ${Math.round(s * 100)}% ${Math.round(l * 100)}%`;

    // Applica al :root per sovrascrivere il tema Shadcn
    document.documentElement.style.setProperty("--primary", hslStr);
    document.documentElement.style.setProperty("--ring", hslStr);

    return () => {
      // Ripristina i colori originali all'unmount
      document.documentElement.style.removeProperty("--primary");
      document.documentElement.style.removeProperty("--ring");
    };
  }, [branding?.primaryColor]);

  // Imposta favicon dinamico
  useEffect(() => {
    if (!branding?.faviconUrl) return;
    const link = document.querySelector("link[rel='icon']") as HTMLLinkElement
      || document.createElement("link");
    link.rel = "icon";
    link.href = branding.faviconUrl;
    document.head.appendChild(link);
    return () => { link.href = "/favicon.ico"; };
  }, [branding?.faviconUrl]);

  // Imposta document title
  useEffect(() => {
    if (branding?.platformName) {
      document.title = branding.platformName;
    }
    return () => { document.title = "Edilizia in Cloud"; };
  }, [branding?.platformName]);

  return (
    <BrandingContext.Provider value={{ theme, isWhiteLabel }}>
      {children}
    </BrandingContext.Provider>
  );
}

export const useBranding = () => useContext(BrandingContext);
```

STEP 10 — EDGE FUNCTION: verifica CNAME DNS
================================================================

Crea: supabase/functions/verify-custom-domain/index.ts

```typescript
import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from "jsr:@supabase/supabase-js@2";

const supabase = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
);

Deno.serve(async (req) => {
  const { company_id } = await req.json();

  // Carica il branding dell'azienda
  const { data: branding, error } = await supabase
    .from("company_branding")
    .select("custom_domain, custom_domain_cname")
    .eq("company_id", company_id)
    .single();

  if (error || !branding?.custom_domain || !branding?.custom_domain_cname) {
    return Response.json({ verified: false, error: "Nessun dominio configurato" });
  }

  try {
    // Verifica il CNAME via DNS over HTTPS (Cloudflare)
    const dnsUrl = `https://cloudflare-dns.com/dns-query?name=${encodeURIComponent(branding.custom_domain)}&type=CNAME`;
    const dnsResp = await fetch(dnsUrl, {
      headers: { Accept: "application/dns-json" },
    });
    const dnsData = await dnsResp.json();

    const cnameAnswer = dnsData.Answer?.find((a: any) => a.type === 5); // 5 = CNAME
    const pointsToExpected = cnameAnswer?.data?.replace(/\.$/, "") === branding.custom_domain_cname;

    if (pointsToExpected) {
      // Segna come verificato
      await supabase.from("company_branding")
        .update({
          custom_domain_verified: true,
          custom_domain_verified_at: new Date().toISOString(),
        })
        .eq("company_id", company_id);

      return Response.json({ verified: true });
    } else {
      return Response.json({
        verified: false,
        error: `CNAME trovato: ${cnameAnswer?.data || "nessuno"} — atteso: ${branding.custom_domain_cname}`,
      });
    }
  } catch (err) {
    return Response.json({ verified: false, error: "Errore DNS: " + String(err) });
  }
});
```

NOTA SUL CUSTOM DOMAIN — ROUTING INFRASTRUTTURALE
================================================================

La verifica CNAME funziona lato codice. Per far SÌ che il dominio custom
punta effettivamente alla tua app Vite (ospitata su Vercel/Netlify/altro),
devi configurare il tuo hosting per accettare wildcard custom domains:

- VERCEL: Settings → Domains → aggiungi il dominio. Ogni dominio custom
  verificato va aggiunto manualmente nel pannello Vercel.
- NETLIFY: Site settings → Domain management → custom domains.
- Per automazione: usa l'API Vercel/Netlify per aggiungere il dominio
  automaticamente quando viene verificato (aggiungerlo all'Edge Function
  verify-custom-domain dopo la verifica CNAME).

Il frontend già gestisce la detection del dominio: useBrandingByDomain()
rileva l'hostname e carica il branding corretto — tutto funziona senza
modifiche al routing Vite.
