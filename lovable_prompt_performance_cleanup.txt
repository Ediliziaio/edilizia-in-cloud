### FIX 20: Utility function exportToCSV

La logica di export CSV è copiata in almeno 3 file:
- src/pages/azienda/OrdersList.tsx
- src/pages/azienda/CashFlowForecast.tsx
- src/pages/azienda/Warehouse.tsx

Crea il file `src/lib/exportCsv.ts`:

```typescript
// src/lib/exportCsv.ts
export function exportToCSV(
  rows: string[][],
  filename: string
): void {
  const csvContent = rows
    .map((row) =>
      row.map((cell) => {
        const str = String(cell ?? "");
        // Escape celle con virgola, newline o virgolette
        if (str.includes(",") || str.includes("\n") || str.includes('"')) {
          return `"${str.replace(/"/g, '""')}"`;
        }
        return str;
      }).join(",")
    )
    .join("\n");

  const blob = new Blob(["\uFEFF" + csvContent], {
    type: "text/csv;charset=utf-8;",
  });
  const url = URL.createObjectURL(blob);
  const link = document.createElement("a");
  link.href = url;
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  URL.revokeObjectURL(url);
}
```

Poi nei 3 file sopra, rimuovi la logica CSV duplicata e importa:
```typescript
import { exportToCSV } from "@/lib/exportCsv";
```

### FIX 23: src/pages/azienda/Calendar.tsx — localStorage senza debounce

Il salvataggio di visibleEmployeeIds e visibleTeamIds in localStorage viene
eseguito ad ogni render quando le liste cambiano. Aggiungi debounce:

```typescript
import { useRef } from "react";

// Nel componente, dopo gli state:
const saveTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);

// Negli useEffect che salvano in localStorage, wrappa con debounce:
useEffect(() => {
  if (saveTimerRef.current) clearTimeout(saveTimerRef.current);
  saveTimerRef.current = setTimeout(() => {
    localStorage.setItem("calendar_visible_employees", JSON.stringify(visibleEmployeeIds));
  }, 500); // 500ms debounce
  return () => {
    if (saveTimerRef.current) clearTimeout(saveTimerRef.current);
  };
}, [visibleEmployeeIds]);

// Stessa cosa per visibleTeamIds e altre preferenze calendario salvate in localStorage
```
