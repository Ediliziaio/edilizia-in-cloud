-- Tabella notifiche in-app
CREATE TABLE IF NOT EXISTS public.notifications (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id    uuid NOT NULL REFERENCES public.companies(id) ON DELETE CASCADE,
  user_id       uuid NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  type          text NOT NULL,
  -- Tipi validi: 'order_status_changed' | 'installment_due' | 'ticket_new'
  --              | 'ticket_reply' | 'task_assigned' | 'task_due' | 'generic'
  title         text NOT NULL,
  body          text,
  entity_type   text,  -- 'order' | 'ticket' | 'task' | 'installment' | 'contact'
  entity_id     uuid,
  action_url    text,  -- percorso relativo, es. /azienda/ordini/uuid
  is_read       boolean NOT NULL DEFAULT false,
  is_dismissed  boolean NOT NULL DEFAULT false,
  created_at    timestamptz NOT NULL DEFAULT now()
);

-- Indici per performance
CREATE INDEX idx_notifications_user_id    ON public.notifications(user_id);
CREATE INDEX idx_notifications_company_id ON public.notifications(company_id);
CREATE INDEX idx_notifications_is_read    ON public.notifications(is_read) WHERE is_read = false;
CREATE INDEX idx_notifications_created_at ON public.notifications(created_at DESC);

-- RLS
ALTER TABLE public.notifications ENABLE ROW LEVEL SECURITY;

-- Ogni utente vede solo le proprie notifiche
CREATE POLICY "users_read_own_notifications"
  ON public.notifications FOR SELECT
  USING (auth.uid() = user_id);

-- Ogni utente può aggiornare solo is_read e is_dismissed delle proprie
CREATE POLICY "users_update_own_notifications"
  ON public.notifications FOR UPDATE
  USING (auth.uid() = user_id)
  WITH CHECK (auth.uid() = user_id);

-- Solo service_role inserisce (tramite trigger PG o funzione RPC)
-- Gli utenti NON possono inserire notifiche proprie
```

-- Funzione helper chiamabile dai trigger (SECURITY DEFINER)
CREATE OR REPLACE FUNCTION public.create_notification(
  p_company_id  uuid,
  p_user_id     uuid,
  p_type        text,
  p_title       text,
  p_body        text        DEFAULT NULL,
  p_entity_type text        DEFAULT NULL,
  p_entity_id   uuid        DEFAULT NULL,
  p_action_url  text        DEFAULT NULL
)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_id uuid;
BEGIN
  -- Non creare notifiche se l'utente non esiste più
  IF NOT EXISTS (SELECT 1 FROM auth.users WHERE id = p_user_id) THEN
    RETURN NULL;
  END IF;

  INSERT INTO public.notifications (
    company_id, user_id, type, title, body,
    entity_type, entity_id, action_url
  ) VALUES (
    p_company_id, p_user_id, p_type, p_title, p_body,
    p_entity_type, p_entity_id, p_action_url
  ) RETURNING id INTO v_id;

  RETURN v_id;
END;
$$;
```

```typescript
import { useEffect } from "react";
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { supabase } from "@/integrations/supabase/client";
import { useAuth } from "@/hooks/useAuth";

export interface Notification {
  id: string;
  company_id: string;
  user_id: string;
  type: string;
  title: string;
  body: string | null;
  entity_type: string | null;
  entity_id: string | null;
  action_url: string | null;
  is_read: boolean;
  is_dismissed: boolean;
  created_at: string;
}

export function useNotifications() {
  const { profile, company } = useAuth();
  const queryClient = useQueryClient();
  const companyId = company?.id;
  const userId = profile?.id;

  // Query principale: notifiche non dismissed
  const { data: notifications = [], isLoading } = useQuery<Notification[]>({
    queryKey: ["notifications", companyId, userId],
    queryFn: async () => {
      const { data, error } = await supabase
        .from("notifications")
        .select("*")
        .eq("company_id", companyId!)
        .eq("user_id", userId!)
        .eq("is_dismissed", false)
        .order("created_at", { ascending: false })
        .limit(50);
      if (error) throw error;
      return data as Notification[];
    },
    enabled: !!companyId && !!userId,
    staleTime: 30 * 1000,
    gcTime: 5 * 60 * 1000,
  });

  // Subscription realtime: aggiorna la cache quando arriva una nuova notifica
  useEffect(() => {
    if (!companyId || !userId) return;

    const channel = supabase
      .channel(`notifications:${userId}`)
      .on(
        "postgres_changes",
        {
          event: "INSERT",
          schema: "public",
          table: "notifications",
          filter: `user_id=eq.${userId}`,
        },
        (payload) => {
          // Aggiungi la nuova notifica in cima alla cache senza refetch
          queryClient.setQueryData<Notification[]>(
            ["notifications", companyId, userId],
            (old = []) => [payload.new as Notification, ...old]
          );
        }
      )
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [companyId, userId, queryClient]);

  // Numero non lette
  const unreadCount = notifications.filter((n) => !n.is_read).length;

  // Segna una notifica come letta
  const markAsRead = useMutation({
    mutationFn: async (id: string) => {
      const { error } = await supabase
        .from("notifications")
        .update({ is_read: true })
        .eq("id", id)
        .eq("user_id", userId!);
      if (error) throw error;
    },
    onMutate: async (id) => {
      // Optimistic update
      queryClient.setQueryData<Notification[]>(
        ["notifications", companyId, userId],
        (old = []) => old.map((n) => (n.id === id ? { ...n, is_read: true } : n))
      );
    },
  });

  // Segna tutte come lette
  const markAllAsRead = useMutation({
    mutationFn: async () => {
      const { error } = await supabase.rpc("mark_all_notifications_read", {
        p_company_id: companyId!,
      });
      if (error) throw error;
    },
    onMutate: async () => {
      queryClient.setQueryData<Notification[]>(
        ["notifications", companyId, userId],
        (old = []) => old.map((n) => ({ ...n, is_read: true }))
      );
    },
  });

  // Dismissta (nasconde dal pannello)
  const dismiss = useMutation({
    mutationFn: async (id: string) => {
      const { error } = await supabase
        .from("notifications")
        .update({ is_dismissed: true })
        .eq("id", id)
        .eq("user_id", userId!);
      if (error) throw error;
    },
    onMutate: async (id) => {
      queryClient.setQueryData<Notification[]>(
        ["notifications", companyId, userId],
        (old = []) => old.filter((n) => n.id !== id)
      );
    },
  });

  return {
    notifications,
    unreadCount,
    isLoading,
    markAsRead: markAsRead.mutate,
    markAllAsRead: markAllAsRead.mutate,
    dismiss: dismiss.mutate,
  };
}
```

TRIGGER 1 — Nuovo ticket → notifica agli admin
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

```sql
CREATE OR REPLACE FUNCTION public.notify_new_ticket()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_admin RECORD;
  v_customer_name text;
BEGIN
  -- Recupera nome cliente
  SELECT first_name || ' ' || last_name INTO v_customer_name
  FROM public.profiles WHERE id = NEW.customer_id LIMIT 1;

  -- Notifica all'assegnatario se c'è, altrimenti a tutti gli admin
  IF NEW.assigned_to IS NOT NULL THEN
    PERFORM public.create_notification(
      NEW.company_id,
      NEW.assigned_to,
      'ticket_new',
      'Nuovo ticket: ' || NEW.subject,
      'Da: ' || COALESCE(v_customer_name, 'Cliente'),
      'ticket',
      NEW.id,
      '/azienda/ticket/' || NEW.id
    );
  ELSE
    -- Notifica tutti i company_admin e company_staff
    FOR v_admin IN
      SELECT p.id FROM public.profiles p
      JOIN public.user_roles ur ON ur.user_id = p.id
      WHERE p.company_id = NEW.company_id
        AND ur.role IN ('company_admin', 'company_staff')
    LOOP
      PERFORM public.create_notification(
        NEW.company_id,
        v_admin.id,
        'ticket_new',
        'Nuovo ticket: ' || NEW.subject,
        'Da: ' || COALESCE(v_customer_name, 'Cliente'),
        'ticket',
        NEW.id,
        '/azienda/ticket/' || NEW.id
      );
    END LOOP;
  END IF;

  RETURN NEW;
END;
$$;

CREATE TRIGGER trg_notify_new_ticket
  AFTER INSERT ON public.tickets
  FOR EACH ROW EXECUTE FUNCTION public.notify_new_ticket();
```

-- Il nome esatto potrebbe essere 'ticket_messages' o 'support_messages'.

CREATE OR REPLACE FUNCTION public.notify_ticket_reply()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_ticket  RECORD;
  v_sender_name text;
BEGIN
  -- Prendi dati del ticket
  SELECT t.*, p.first_name || ' ' || p.last_name AS sender_name
  INTO v_ticket
  FROM public.tickets t
  LEFT JOIN public.profiles p ON p.id = NEW.sender_id
  WHERE t.id = NEW.ticket_id;

  -- Notifica all'assigned_to se è diverso dal mittente
  IF v_ticket.assigned_to IS NOT NULL AND v_ticket.assigned_to <> NEW.sender_id THEN
    PERFORM public.create_notification(
      v_ticket.company_id,
      v_ticket.assigned_to,
      'ticket_reply',
      'Risposta su: ' || v_ticket.subject,
      COALESCE(v_ticket.sender_name, 'Utente') || ' ha risposto al ticket',
      'ticket',
      NEW.ticket_id,
      '/azienda/ticket/' || NEW.ticket_id
    );
  END IF;

  RETURN NEW;
END;
$$;

TRIGGER 3 — Task assegnato → notifica all'assegnatario
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

-- Adatta 'marketing_tasks' al nome reale della tabella task
CREATE OR REPLACE FUNCTION public.notify_task_assigned()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_due_text text;
BEGIN
  -- Solo se c'è un assegnatario E è diverso da chi ha creato il task
  IF NEW.assigned_to IS NULL THEN RETURN NEW; END IF;
  IF NEW.assigned_to = NEW.created_by THEN RETURN NEW; END IF;

  v_due_text := CASE
    WHEN NEW.due_date IS NOT NULL THEN ' (scadenza: ' || NEW.due_date::date::text || ')'
    ELSE ''
  END;

  PERFORM public.create_notification(
    NEW.company_id,
    NEW.assigned_to,
    'task_assigned',
    'Nuovo task assegnato: ' || COALESCE(NEW.title, 'Senza titolo'),
    'Ti è stato assegnato un task' || v_due_text,
    'task',
    NEW.id,
    '/azienda/marketing/task'
  );

DECLARE
  v_new_status_name text;
  v_creator         RECORD;
BEGIN
  -- Solo se lo stato è effettivamente cambiato
  IF OLD.current_status_id = NEW.current_status_id THEN RETURN NEW; END IF;

  SELECT name INTO v_new_status_name

      'Ordine ' || NEW.order_code || ': ' || COALESCE(v_new_status_name, 'nuovo stato'),
      COALESCE(NEW.description, 'Stato aggiornato'),
      'order',
      NEW.id,
      '/azienda/ordini/' || NEW.id
    );
  END IF;

  RETURN NEW;
END;

```typescript
import { useSearchParams, useNavigate } from "react-router-dom";
import { useCallback } from "react";

/**
 * Hook per gestire filtri persistenti in URL params.
 * Aggiorna l'URL senza aggiungere entry nella history (replace:true)
 * per non "spammare" la cronologia con ogni cambiamento di filtro.
 */
export function useFilterParams<T extends Record<string, string>>(
  defaults: T
) {
  const [searchParams, setSearchParams] = useSearchParams();

  // Leggi un parametro (con fallback al default)
  const getParam = useCallback(
    (key: keyof T): string => {
      return searchParams.get(key as string) ?? defaults[key];
    },
    [searchParams, defaults]
  );

  // Aggiorna uno o più parametri
  const setParam = useCallback(
    (key: keyof T, value: string) => {
      setSearchParams(
        (prev) => {
          const next = new URLSearchParams(prev);
          if (value === defaults[key] || value === "") {
            next.delete(key as string); // rimuovi se è il default (URL pulito)
          } else {
            next.set(key as string, value);
          }
          return next;
        },
        { replace: true } // non aggiunge voce in history
      );
    },
    [setSearchParams, defaults]
  );

  // Aggiorna più parametri in una volta
  const setParams = useCallback(
    (updates: Partial<T>) => {
      setSearchParams(
        (prev) => {
          const next = new URLSearchParams(prev);
          for (const [key, value] of Object.entries(updates)) {
            if (value === defaults[key] || value === "") {
              next.delete(key);
            } else {
              next.set(key, value as string);
            }
          }
          return next;
        },
        { replace: true }
      );
    },
    [setSearchParams, defaults]
  );

  // Reset tutti i filtri
  const resetParams = useCallback(() => {
    setSearchParams({}, { replace: true });
  }, [setSearchParams]);

  return { getParam, setParam, setParams, resetParams };
}
```

"pending" a "completed" o viceversa). Aggiungi il pattern completo:

```typescript
const toggleTaskMutation = useMutation({
  mutationFn: async ({ taskId, newStatus }: { taskId: string; newStatus: string }) => {
    const { error } = await supabase
      .from("marketing_tasks")  // adatta al nome tabella reale
      .update({ status: newStatus })
      .eq("id", taskId);
    if (error) throw error;
  },

  // OTTIMISTICO: aggiorna la cache PRIMA della risposta server
  onMutate: async ({ taskId, newStatus }) => {
    // Cancella eventuali refetch in corso per questa query
    await queryClient.cancelQueries({ queryKey: ["marketing-tasks", companyId] });

    // Salva snapshot per rollback
    const previousTasks = queryClient.getQueryData(["marketing-tasks", companyId]);

    // Aggiorna la cache immediatamente
    queryClient.setQueryData(
      ["marketing-tasks", companyId],
      (old: any[] = []) =>
        old.map((t) => (t.id === taskId ? { ...t, status: newStatus } : t))
    );

    return { previousTasks }; // passato a onError
  },

  // ROLLBACK se il server risponde con errore
  onError: (_err, _vars, context) => {
    if (context?.previousTasks) {
      queryClient.setQueryData(["marketing-tasks", companyId], context.previousTasks);
    }
    toast.error("Errore nell'aggiornamento del task");
  },

  // Sincronizza con il server in ogni caso (successo o errore)
  onSettled: () => {
    queryClient.invalidateQueries({ queryKey: ["marketing-tasks", companyId] });
  },
});
```

const toggleActiveMutation = useMutation({
  mutationFn: async ({ id, is_active }: { id: string; is_active: boolean }) => {
    const { error } = await supabase
      .from("article_templates")
      .update({ is_active })
      .eq("id", id);
    if (error) throw error;
  },
  onMutate: async ({ id, is_active }) => {
    await queryClient.cancelQueries({ queryKey: ["article-templates", companyId] });
    const prev = queryClient.getQueryData(["article-templates", companyId]);
    queryClient.setQueryData(
      ["article-templates", companyId],
      (old: any[] = []) => old.map(a => a.id === id ? {...a, is_active} : a)
    );
    return { prev };
  },
  onError: (_e, _v, ctx) => {
    queryClient.setQueryData(["article-templates", companyId], ctx?.prev);
  },
  onSettled: () => {
    queryClient.invalidateQueries({ queryKey: ["article-templates", companyId] });
  },
});
