STEP 1 — Crea record permessi automaticamente per ogni nuovo super_admin
────────────────────────────────────────────────────────────────────────────

Esegui in Supabase SQL Editor:

```sql
-- Migration: superadmin_auto_permissions_record

-- 1. Funzione trigger: crea record permessi con NO_ACCESS quando viene
--    creato un nuovo profilo con role = 'super_admin'
CREATE OR REPLACE FUNCTION create_superadmin_permissions_on_role_assign()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  -- Solo se il ruolo cambia a super_admin (o è già super_admin su INSERT)
  IF NEW.role = 'super_admin' AND (TG_OP = 'INSERT' OR OLD.role IS DISTINCT FROM 'super_admin') THEN
    INSERT INTO super_admin_permissions (
      user_id,
      can_manage_companies,
      can_manage_plans,
      can_manage_tickets,
      can_manage_referrals,
      can_manage_admins,
      can_view_platform_stats,
      allowed_company_ids
    ) VALUES (
      NEW.id,
      false,   -- NO_ACCESS by default
      false,
      false,
      false,
      false,
      false,
      NULL     -- NULL = nessuna restrizione (se abilitato sarà via UI)
    )
    ON CONFLICT (user_id) DO NOTHING;  -- non sovrascrivere se esiste già
  END IF;
  RETURN NEW;
END;
$$;

-- Trigger su profiles
DROP TRIGGER IF EXISTS trg_create_superadmin_permissions ON profiles;
CREATE TRIGGER trg_create_superadmin_permissions
  AFTER INSERT OR UPDATE OF role ON profiles
  FOR EACH ROW
  EXECUTE FUNCTION create_superadmin_permissions_on_role_assign();

-- 2. Retroattivo: crea record per tutti i super_admin esistenti senza record
INSERT INTO super_admin_permissions (
  user_id,
  can_manage_companies,
  can_manage_plans,
  can_manage_tickets,
  can_manage_referrals,
  can_manage_admins,
  can_view_platform_stats,
  allowed_company_ids
)
SELECT
  p.id,
  true,  -- per gli admin già esistenti, mantieni accesso completo (migrazione non distruttiva)
  true,
  true,
  true,
  true,
  true,
  NULL
FROM profiles p
WHERE p.role = 'super_admin'
  AND NOT EXISTS (
    SELECT 1 FROM super_admin_permissions sap WHERE sap.user_id = p.id
  );
```

────────────────────────────────────────────────────────────────────────────

```sql
-- Migration: superadmin_rls_company_restrictions

-- Funzione helper: restituisce true se l'utente può vedere questa company
CREATE OR REPLACE FUNCTION superadmin_can_access_company(p_company_id uuid)
RETURNS boolean
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_role text;
  v_allowed_ids uuid[];
BEGIN
  -- Recupera ruolo utente
  SELECT role INTO v_role FROM profiles WHERE id = auth.uid();

  -- Non è un superadmin → usa le policy normali
  IF v_role != 'super_admin' THEN
    RETURN false;
  END IF;

  -- Recupera le restrizioni company del superadmin
  SELECT allowed_company_ids INTO v_allowed_ids
  FROM super_admin_permissions
  WHERE user_id = auth.uid();

  -- NULL o array vuoto = nessuna restrizione → accesso a tutto
  IF v_allowed_ids IS NULL OR array_length(v_allowed_ids, 1) IS NULL THEN
    RETURN true;
  END IF;

  -- Verifica che la company sia nella lista
  RETURN p_company_id = ANY(v_allowed_ids);
END;
$$;

-- RLS policy su companies per i superadmin con restrizioni
-- (Aggiungi questa policy ACCANTO alle policy esistenti, non sostituirla)

-- Prima controlla se c'è già una policy superadmin su companies:
-- DROP POLICY IF EXISTS "superadmin_restricted_access" ON companies;

CREATE POLICY "superadmin_restricted_company_access" ON companies
  FOR SELECT
  USING (
    -- I membri normali della company vedono la propria (policy esistente gestisce questo)
    -- I superadmin senza restrizioni vedono tutto
    -- I superadmin con restrizioni vedono solo le proprie
    (
      SELECT role FROM profiles WHERE id = auth.uid()
    ) = 'super_admin'
    AND superadmin_can_access_company(id)
  );

-- Nota: questa policy si AGGIUNGE a quelle esistenti per i member normali.
-- Supabase usa OR tra policies dello stesso tipo.
-- Verifica che le policy esistenti usino ancora OR corretto.
```

LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_result jsonb;
BEGIN
  -- Verifica che sia un superadmin
  IF NOT EXISTS (
    SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'super_admin'
  ) THEN
    RAISE EXCEPTION 'Access denied';
  END IF;

  SELECT jsonb_build_object(
    'companies', (
      SELECT jsonb_agg(
        jsonb_build_object(
          'id',            c.id,
          'name',          c.name,
          'slug',          c.slug,
          'is_active',     c.is_active,
          'created_at',    c.created_at,
          'plan',          c.plan,
          'ordini_attivi', (
            SELECT COUNT(*) FROM orders o
            WHERE o.company_id = c.id AND o.stato NOT IN ('completato', 'annullato')
          ),
          'user_count', (
            SELECT COUNT(*) FROM company_members cm WHERE cm.company_id = c.id
          ),
          'last_access', (
            SELECT MAX(al.created_at) FROM audit_logs al
            WHERE al.details->>'company_id' = c.id::text
          )
        )
        ORDER BY c.name
      )
      FROM companies c
      WHERE (p_search IS NULL OR c.name ILIKE '%' || p_search || '%')
      LIMIT p_limit OFFSET p_offset
    ),
    'total', (
      SELECT COUNT(*) FROM companies c
      WHERE (p_search IS NULL OR c.name ILIKE '%' || p_search || '%')
    )
  ) INTO v_result;

  RETURN v_result;
END;
$$;

REVOKE ALL ON FUNCTION get_admin_companies_list(text, int, int) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION get_admin_companies_list(text, int, int) TO authenticated;

const ROUTE_PERMISSIONS: Record<string, keyof SuperAdminPermissions> = {
  "/admin/aziende":    "can_manage_companies",

    location.pathname.startsWith(r)
