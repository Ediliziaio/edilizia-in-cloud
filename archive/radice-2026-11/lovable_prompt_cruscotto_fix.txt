### BUG 1 — Cash Runway formula sbagliata (FinanzaCashFlow.tsx)

PROBLEMA: La formula attuale divide l'income totale per il burn rate giornaliero,
non il saldo disponibile. Risultato: runway gonfiato e spesso errato.

Cerca in FinanzaCashFlow.tsx il blocco che calcola cashRunwayDays e sostituiscilo:

```typescript
// ❌ PRIMA (ERRATO):
const netAvailable = finance.thisMonthIncome - (finance.thisMonthOutflow * (dayOfMonth / daysInMonth));
const runway = dailyBurn > 0 ? Math.round((finance.thisMonthIncome) / dailyBurn) : 999;

// ✅ DOPO (CORRETTO):
// Il runway si basa sul saldo netto del mese (entrate - uscite già sostenute)
// diviso per il burn rate giornaliero stimato
const outflowSoFar = finance.thisMonthOutflow; // uscite reali di questo mese
const netBalance   = finance.thisMonthIncome - outflowSoFar; // saldo netto a oggi
const dailyBurn    = daysInMonth > 0 ? outflowSoFar / dayOfMonth : 0;
                     // burn rate basato sui giorni passati del mese (non proiettato)
const cashRunwayDays = dailyBurn > 0
  ? Math.max(0, Math.round(netBalance / dailyBurn))
  : netBalance > 0 ? 999 : 0;
// Se dailyBurn è 0 e c'è saldo positivo → nessun problema (∞)
// Se dailyBurn è 0 e saldo è 0 o negativo → 0 giorni

// Visualizza:
// 0-7 giorni → rosso, warning critico
// 8-30 giorni → arancio, warning
// 31-90 giorni → giallo, attenzione
// >90 giorni → verde
// 999 → mostra "Stabile" invece di "∞"
const runwayLabel = cashRunwayDays >= 999
  ? 'Stabile'
  : cashRunwayDays > 365
  ? `${Math.round(cashRunwayDays / 30)} mesi`
  : `${cashRunwayDays}gg`;
```

Aggiorna anche il display del runway aggiungendo un tooltip:
```tsx
<TooltipProvider>
  <Tooltip>
    <TooltipTrigger asChild>
      <span className="cursor-help underline decoration-dotted">{runwayLabel}</span>
    </TooltipTrigger>
    <TooltipContent>
      <p>Giorni stimati prima di esaurire il saldo netto</p>
      <p>Basato su: saldo {formatCurrency(netBalance)} / burn {formatCurrency(dailyBurn)}/gg</p>
    </TooltipContent>
  </Tooltip>
</TooltipProvider>
```

### BUG 2 — Calcolo margine non pesato (useCruscottoData.ts)

PROBLEMA: Il margine medio viene calcolato come media semplice per ordine,
non come media ponderata per fatturato. Un ordine da €100 con margine 80%
e uno da €10.000 con margine 5% danno "42.5%" invece del corretto "5.7%".

Trova la funzione `calc` in useCruscottoData.ts e sostituiscila:

```typescript
// ❌ PRIMA (media semplice per ordine):
const calc = (orders: any[]) => {
  let revenue = 0, totalMarginPct = 0, marginCount = 0;
  orders.forEach((o: any) => {
    // ... media per ordine
    if (total > 0 && totalCost > 0) {
      totalMarginPct += ((total - totalCost) / total) * 100;
      marginCount++;
    }
  });
  return { revenue, margin: marginCount > 0 ? totalMarginPct / marginCount : 0 };
};

// ✅ DOPO (media ponderata per fatturato):
const calc = (orders: any[]) => {
  let totalRevenue = 0;
  let totalCostAll = 0;

  orders.forEach((o: any) => {
    const revenue = Number(o.total_amount) || 0;
    totalRevenue += revenue;

    const articleCost = (o.order_items || []).reduce(
      (s: number, i: any) => s + (Number(i.purchase_price || 0) * Number(i.quantity || 1)), 0
    );
    const laborCost =
      (o.order_employees || []).reduce((s: number, e: any) => s + Number(e.total_cost || 0), 0) +
      (o.order_external_teams || []).reduce((s: number, e: any) => s + Number(e.total_cost || 0), 0);

    totalCostAll += articleCost + laborCost;
  });

  // Margine ponderato corretto: (totale fatturato - totale costi) / totale fatturato
  const weightedMargin = totalRevenue > 0
    ? ((totalRevenue - totalCostAll) / totalRevenue) * 100
    : 0;

  // Clamp tra -100% e +100% per evitare valori impossibili
  const margin = Math.min(100, Math.max(-100, weightedMargin));

  return { revenue: totalRevenue, margin };
};
```

```typescript
const { data: paymentsData } = useQuery({
  queryKey: ['cruscotto-installments', companyId, filters.dateFrom, filters.dateTo],
  queryFn: async () => {

    const { data, error } = await (supabase as any)
```

PROMPT 2 — PERFORMANCE: CONSOLIDA LE QUERY IN UNA RPC
================================================================================

Attualmente il cruscotto fa 4-5 query separate. Consolida in una RPC server-side.

### STEP 1: Crea la migrazione SQL

```sql
-- RPC principale del cruscotto
CREATE OR REPLACE FUNCTION get_cruscotto_stats(
  p_company_id UUID,
  p_date_from DATE,
  p_date_to DATE
)
RETURNS JSONB AS $$
DECLARE
  v_result JSONB;
  v_prev_from DATE := p_date_from - (p_date_to - p_date_from + 1);
  v_prev_to DATE   := p_date_from - 1;
BEGIN
  SELECT jsonb_build_object(

    -- ── FINANZA ──────────────────────────────────────────
    'finance', jsonb_build_object(
      'revenue_current', (
        SELECT COALESCE(SUM(total_amount), 0)
        FROM orders
        WHERE company_id = p_company_id
          AND created_at::date BETWEEN p_date_from AND p_date_to
          AND status NOT IN ('cancelled', 'annullato')
      ),
      'revenue_previous', (
        SELECT COALESCE(SUM(total_amount), 0)
        FROM orders
        WHERE company_id = p_company_id
          AND created_at::date BETWEEN v_prev_from AND v_prev_to
          AND status NOT IN ('cancelled', 'annullato')
      ),
      'costs_current', (
        SELECT COALESCE(SUM(amount), 0)
        FROM company_costs
        WHERE company_id = p_company_id
          AND cost_date BETWEEN p_date_from AND p_date_to
      ),
      'costs_previous', (
        SELECT COALESCE(SUM(amount), 0)
        FROM company_costs
        WHERE company_id = p_company_id
          AND cost_date BETWEEN v_prev_from AND v_prev_to
      ),
      -- Margine ponderato corretto lato server
      'margin_pct', (
        SELECT CASE
          WHEN SUM(o.total_amount) > 0 THEN
            ROUND(((SUM(o.total_amount) - COALESCE(SUM(
              (SELECT SUM(oi.purchase_price * oi.quantity) FROM order_items oi WHERE oi.order_id = o.id)
            ), 0)) / SUM(o.total_amount)) * 100, 2)
          ELSE 0
        END
        FROM orders o
        WHERE o.company_id = p_company_id
          AND o.created_at::date BETWEEN p_date_from AND p_date_to
          AND o.status NOT IN ('cancelled', 'annullato')
      )
    ),

    -- ── ORDINI ───────────────────────────────────────────
    'orders', jsonb_build_object(
      'total_current', (
        SELECT COUNT(*) FROM orders
        WHERE company_id = p_company_id
          AND created_at::date BETWEEN p_date_from AND p_date_to
      ),
      'total_previous', (
        SELECT COUNT(*) FROM orders
        WHERE company_id = p_company_id
          AND created_at::date BETWEEN v_prev_from AND v_prev_to
      ),
      'in_progress', (
        SELECT COUNT(*) FROM orders
        WHERE company_id = p_company_id
          AND status NOT IN ('completato', 'consegnato', 'cancelled', 'annullato')
      ),
      'late', (
        SELECT COUNT(*) FROM orders
        WHERE company_id = p_company_id
          AND expected_date < CURRENT_DATE
          AND work_end_date IS NULL
          AND status NOT IN ('completato', 'consegnato', 'cancelled', 'annullato')
      ),
      'avg_value', (
        SELECT ROUND(AVG(total_amount), 2) FROM orders
        WHERE company_id = p_company_id
          AND created_at::date BETWEEN p_date_from AND p_date_to
          AND total_amount > 0
      )
    ),

    -- ── CLIENTI ──────────────────────────────────────────
    'customers', jsonb_build_object(
      'total', (SELECT COUNT(*) FROM customers WHERE company_id = p_company_id),
      'new_current', (
        SELECT COUNT(*) FROM customers
        WHERE company_id = p_company_id
          AND created_at::date BETWEEN p_date_from AND p_date_to
      ),
      'new_previous', (
        SELECT COUNT(*) FROM customers
        WHERE company_id = p_company_id
          AND created_at::date BETWEEN v_prev_from AND v_prev_to
      )
    ),

    -- ── TICKET ───────────────────────────────────────────
    'tickets', jsonb_build_object(
      'open', (
        SELECT COUNT(*) FROM tickets
        WHERE company_id = p_company_id AND status = 'open'
      ),
      'resolved_current', (
        SELECT COUNT(*) FROM tickets
        WHERE company_id = p_company_id
          AND status = 'resolved'
          AND updated_at::date BETWEEN p_date_from AND p_date_to
      )
    ),

    -- ── SCADENZE PROSSIMI 30 GIORNI ──────────────────────
    'upcoming_payments', (
      SELECT jsonb_agg(jsonb_build_object(
        'amount', oi.amount,
        'expected_date', oi.expected_date,
        'order_number', o.order_number
      ) ORDER BY oi.expected_date)
      FROM order_installments oi
      JOIN orders o ON oi.order_id = o.id
      WHERE o.company_id = p_company_id
        AND oi.is_paid = false
        AND oi.expected_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 30
    ),

    -- ── GRAFICO RICAVI 6 MESI ────────────────────────────
    'monthly_revenue', (
      SELECT jsonb_agg(jsonb_build_object(
        'month', to_char(m.month_start, 'Mon YY'),
        'revenue', COALESCE(o.rev, 0),
        'costs', COALESCE(c.costs, 0)
      ) ORDER BY m.month_start)
      FROM generate_series(
        date_trunc('month', CURRENT_DATE) - interval '5 months',
        date_trunc('month', CURRENT_DATE),
        interval '1 month'
      ) AS m(month_start)
      LEFT JOIN (
        SELECT date_trunc('month', created_at) AS m, SUM(total_amount) AS rev
        FROM orders
        WHERE company_id = p_company_id AND status NOT IN ('cancelled', 'annullato')
        GROUP BY 1
      ) o ON o.m = m.month_start
      LEFT JOIN (
        SELECT date_trunc('month', cost_date::timestamptz) AS m, SUM(amount) AS costs
        FROM company_costs WHERE company_id = p_company_id
        GROUP BY 1
      ) c ON c.m = m.month_start
    )

  ) INTO v_result;

  RETURN v_result;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;

GRANT EXECUTE ON FUNCTION get_cruscotto_stats TO authenticated;
```

```typescript
export function useCruscottoData(companyId: string | undefined, filters: DashboardFilters) {
  const { dateFrom, dateTo } = getDateRange(filters);

  const { data, isLoading, error } = useQuery({
    queryKey: ['cruscotto-stats', companyId, dateFrom, dateTo],

      return data as CruscottoStats;

    refetchOnWindowFocus: true,
  });

}
```

### C — Error boundary per sezione

Crea: src/components/cruscotto/SectionErrorBoundary.tsx

```tsx
import { Component, ReactNode } from 'react';
import { AlertTriangle } from 'lucide-react';

interface Props { children: ReactNode; sectionName: string; }
interface State { hasError: boolean; error?: Error; }

export class SectionErrorBoundary extends Component<Props, State> {
  state: State = { hasError: false };

  static getDerivedStateFromError(error: Error) {
    return { hasError: true, error };
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="rounded-xl border border-destructive/30 bg-destructive/5 p-4">
          <div className="flex items-center gap-2 text-destructive">
            <AlertTriangle className="h-4 w-4" />
            <span className="text-sm font-medium">
              Errore nel caricamento: {this.props.sectionName}
            </span>
          </div>
          <p className="text-xs text-muted-foreground mt-1">
            {this.state.error?.message}
          </p>
        </div>
      );
    }
    return this.props.children;
  }
}
```
