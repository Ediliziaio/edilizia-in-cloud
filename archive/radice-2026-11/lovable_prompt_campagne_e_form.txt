CREATE TABLE IF NOT EXISTS campaigns (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,

  -- Identità campagna
  name TEXT NOT NULL,
  description TEXT,
  status TEXT NOT NULL DEFAULT 'draft',
  -- 'draft' | 'active' | 'paused' | 'completed' | 'archived'

  -- Tipo e canale
  type TEXT NOT NULL DEFAULT 'mixed',
  -- 'email' | 'social' | 'google_ads' | 'seo' | 'event' | 'referral' | 'mixed'
  channel TEXT,
  -- es. 'facebook', 'instagram', 'google', 'linkedin', 'newsletter', ecc.

  -- Date
  start_date DATE,
  end_date DATE,

  -- Budget
  budget_total    NUMERIC(12,2) DEFAULT 0,   -- budget pianificato
  budget_spent    NUMERIC(12,2) DEFAULT 0,   -- speso effettivo (aggiornato manualmente o via costi)
  budget_currency TEXT DEFAULT 'EUR',

  -- UTM parameters (generati automaticamente, modificabili)
  utm_source   TEXT,    -- es. 'facebook', 'newsletter', 'google'
  utm_medium   TEXT,    -- es. 'cpc', 'email', 'social'
  utm_campaign TEXT,    -- es. 'promo_estate_2025'
  utm_content  TEXT,    -- es. 'banner_hero', 'link_footer'
  utm_term     TEXT,    -- parole chiave (Google Ads)

  -- URL di destinazione
  target_url TEXT,      -- landing page della campagna
  short_url  TEXT,      -- URL corto generato (con UTM appesi)

  -- Obiettivi
  goal_type TEXT,
  -- 'leads' | 'sales' | 'brand_awareness' | 'traffic' | 'conversions'
  goal_value NUMERIC(12,2),    -- es. "genera 50 lead" o "€10.000 di vendite"

  -- Metriche aggregate (aggiornate da trigger o job)
  total_clicks       INTEGER DEFAULT 0,
  total_impressions  INTEGER DEFAULT 0,
  total_leads        INTEGER DEFAULT 0,    -- contatti generati
  total_opportunities INTEGER DEFAULT 0,  -- opportunità attribuite
  total_revenue      NUMERIC(12,2) DEFAULT 0,  -- ricavi attribuiti

  -- Note
  notes TEXT,

  -- Metadata
  created_by UUID REFERENCES profiles(id),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- FUNZIONE RPC: Calcola ROI e metriche campagna
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION get_campaign_stats(p_campaign_id UUID)
RETURNS JSONB AS $$
DECLARE
  v_result JSONB;
  v_spent  NUMERIC;
  v_revenue NUMERIC;
  v_leads  INTEGER;
  v_opps   INTEGER;
  v_clicks INTEGER;
  v_conversions INTEGER;
BEGIN
  -- Speso
  SELECT COALESCE(SUM(amount), 0) INTO v_spent
  FROM campaign_expenses WHERE campaign_id = p_campaign_id;

  -- Ricavi attribuiti
  SELECT COALESCE(SUM(attributed_revenue), 0) INTO v_revenue
  FROM campaign_attributions WHERE campaign_id = p_campaign_id;

  -- Lead (contatti attribuiti)
  SELECT COUNT(*) INTO v_leads
  FROM campaign_attributions
  WHERE campaign_id = p_campaign_id AND entity_type = 'contact';

  -- Opportunità
  SELECT COUNT(*) INTO v_opps
  FROM campaign_attributions
  WHERE campaign_id = p_campaign_id AND entity_type = 'opportunity';

  -- Click
  SELECT COUNT(*) INTO v_clicks
  FROM campaign_clicks WHERE campaign_id = p_campaign_id;

  -- Conversioni (click che hanno generato un contatto)
  SELECT COUNT(*) INTO v_conversions
  FROM campaign_clicks WHERE campaign_id = p_campaign_id AND converted = true;

  -- Aggiorna contatori campagna
  UPDATE campaigns SET
    budget_spent       = v_spent,
    total_clicks       = v_clicks,
    total_leads        = v_leads,
    total_opportunities = v_opps,
    total_revenue      = v_revenue,
    updated_at         = NOW()
  WHERE id = p_campaign_id;

  SELECT jsonb_build_object(
    'budget_spent',    v_spent,
    'revenue',         v_revenue,
    'roi_percent',     CASE WHEN v_spent > 0
                         THEN ROUND(((v_revenue - v_spent) / v_spent * 100)::NUMERIC, 1)
                         ELSE NULL END,
    'roi_absolute',    v_revenue - v_spent,
    'leads',           v_leads,
    'opportunities',   v_opps,
    'clicks',          v_clicks,
    'conversions',     v_conversions,
    'conversion_rate', CASE WHEN v_clicks > 0
                         THEN ROUND((v_conversions::NUMERIC / v_clicks * 100), 1)
                         ELSE 0 END,
    'cost_per_lead',   CASE WHEN v_leads > 0
                         THEN ROUND((v_spent / v_leads)::NUMERIC, 2)
                         ELSE NULL END,
    'cost_per_click',  CASE WHEN v_clicks > 0
                         THEN ROUND((v_spent / v_clicks)::NUMERIC, 4)
                         ELSE NULL END
  ) INTO v_result;

  RETURN v_result;
END;
$$ LANGUAGE plpgsql;

-- ════════════════════════════════════════════════════════════════
-- FUNZIONE: Genera URL con UTM parameters
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION generate_utm_url(p_campaign_id UUID)
RETURNS TEXT AS $$
DECLARE
  v_campaign campaigns%ROWTYPE;
  v_url TEXT;
  v_params TEXT := '';
BEGIN
  SELECT * INTO v_campaign FROM campaigns WHERE id = p_campaign_id;
  IF v_campaign.target_url IS NULL THEN RETURN NULL; END IF;

  v_url := v_campaign.target_url;
  IF v_campaign.utm_source   IS NOT NULL THEN v_params := v_params || '&utm_source='   || v_campaign.utm_source; END IF;
  IF v_campaign.utm_medium   IS NOT NULL THEN v_params := v_params || '&utm_medium='   || v_campaign.utm_medium; END IF;
  IF v_campaign.utm_campaign IS NOT NULL THEN v_params := v_params || '&utm_campaign=' || v_campaign.utm_campaign; END IF;
  IF v_campaign.utm_content  IS NOT NULL THEN v_params := v_params || '&utm_content='  || v_campaign.utm_content; END IF;
  IF v_campaign.utm_term     IS NOT NULL THEN v_params := v_params || '&utm_term='     || v_campaign.utm_term; END IF;

  IF LENGTH(v_params) > 0 THEN
    v_params := SUBSTRING(v_params FROM 2); -- rimuovi & iniziale
    v_url := v_url || CASE WHEN v_url LIKE '%?%' THEN '&' ELSE '?' END || v_params;
  END IF;

  RETURN v_url;
END;
$$ LANGUAGE plpgsql;

-- Trigger: aggiorna short_url quando cambiano i parametri UTM
CREATE OR REPLACE FUNCTION trg_update_campaign_utm()
RETURNS TRIGGER AS $$
BEGIN
  NEW.short_url := generate_utm_url(NEW.id);
  NEW.updated_at := NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_campaigns_utm ON campaigns;
CREATE TRIGGER trg_campaigns_utm
  BEFORE UPDATE ON campaigns
  FOR EACH ROW
  WHEN (
    OLD.utm_source IS DISTINCT FROM NEW.utm_source OR
    OLD.utm_medium IS DISTINCT FROM NEW.utm_medium OR
    OLD.utm_campaign IS DISTINCT FROM NEW.utm_campaign OR
    OLD.target_url IS DISTINCT FROM NEW.target_url
  )
  EXECUTE FUNCTION trg_update_campaign_utm();

-- ════════════════════════════════════════════════════════════════
-- Trigger: aggiungi spesa alla campagna quando si aggiunge expense
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION trg_update_campaign_spent()
RETURNS TRIGGER AS $$
DECLARE v_cid UUID;
BEGIN
  v_cid := COALESCE(NEW.campaign_id, OLD.campaign_id);
  UPDATE campaigns SET
    budget_spent = COALESCE((
      SELECT SUM(amount) FROM campaign_expenses WHERE campaign_id = v_cid
    ), 0),
    updated_at = NOW()
  WHERE id = v_cid;
  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_campaign_expenses ON campaign_expenses;
CREATE TRIGGER trg_campaign_expenses
  AFTER INSERT OR UPDATE OR DELETE ON campaign_expenses
  FOR EACH ROW EXECUTE FUNCTION trg_update_campaign_spent();

────────────────────────────────────────────────────────────────────
FUNZIONE: supabase/functions/campaign-tracker/index.ts
────────────────────────────────────────────────────────────────────
Questa funzione riceve i parametri UTM dal sito del cliente
(tramite pixel JS o redirect) e li registra nella tabella campaign_clicks.
È pubblica (no JWT) perché viene chiamata dal browser del visitatore.

```typescript
// supabase/functions/campaign-tracker/index.ts
// verify_jwt: false — endpoint pubblico

import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const supabase = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
);

const CORS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "POST, GET, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type",
};

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response(null, { headers: CORS });

  const url = new URL(req.url);

  // Parametri UTM (via GET querystring o POST body)
  let params: Record<string, string> = {};
  if (req.method === "GET") {
    params = Object.fromEntries(url.searchParams.entries());
  } else {
    try { params = await req.json(); } catch { params = {}; }
  }

  const {
    utm_source, utm_medium, utm_campaign, utm_content, utm_term,
    company_id, landing_page, referrer,
  } = params;

  if (!company_id || !utm_campaign) {
    return new Response(JSON.stringify({ error: "company_id e utm_campaign richiesti" }), {
      status: 400, headers: { ...CORS, "Content-Type": "application/json" },
    });
  }

  // Trova campagna corrispondente
  const { data: campaign } = await supabase
    .from("campaigns")
    .select("id")
    .eq("company_id", company_id)
    .eq("utm_campaign", utm_campaign)
    .eq("status", "active")
    .maybeSingle();

  if (!campaign) {
    return new Response(JSON.stringify({ tracked: false, reason: "Campaign not found or inactive" }), {
      headers: { ...CORS, "Content-Type": "application/json" },
    });
  }

  // Device detection
  const ua = req.headers.get("user-agent") || "";
  const device_type = /mobile|android|iphone/i.test(ua) ? "mobile" :
                      /tablet|ipad/i.test(ua) ? "tablet" : "desktop";

  // Hash IP per privacy (non salviamo mai l'IP raw)
  const ip = req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ||
             req.headers.get("cf-connecting-ip") || "unknown";
  const ipHash = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(ip))
    .then(buf => Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2,"0")).join("").substring(0, 16));

  // Inserisci click
  const { error } = await supabase.from("campaign_clicks").insert({
    campaign_id: campaign.id,
    company_id,
    utm_source: utm_source || null,
    utm_medium: utm_medium || null,
    utm_campaign,
    utm_content: utm_content || null,
    utm_term: utm_term || null,
    ip_hash: ipHash,
    user_agent: ua.substring(0, 200),
    referrer: referrer || req.headers.get("referer") || null,
    landing_page: landing_page || null,
    device_type,
    clicked_at: new Date().toISOString(),
  });

  if (error) {
    return new Response(JSON.stringify({ tracked: false, error: error.message }), {
      status: 500, headers: { ...CORS, "Content-Type": "application/json" },
    });
  }

  // Risponde con pixel 1x1 trasparente se richiesto via GET (pixel tracking)
  if (req.method === "GET" && url.searchParams.get("pixel") === "1") {
    const pixel = new Uint8Array([
      0x47,0x49,0x46,0x38,0x39,0x61,0x01,0x00,0x01,0x00,0x80,0x00,
      0x00,0xff,0xff,0xff,0x00,0x00,0x00,0x21,0xf9,0x04,0x00,0x00,
      0x00,0x00,0x00,0x2c,0x00,0x00,0x00,0x00,0x01,0x00,0x01,0x00,
      0x00,0x02,0x02,0x44,0x01,0x00,0x3b,
    ]);
    return new Response(pixel, {
      headers: { ...CORS, "Content-Type": "image/gif", "Cache-Control": "no-cache" },
    });
  }

  return new Response(JSON.stringify({ tracked: true, campaign_id: campaign.id }), {
    headers: { ...CORS, "Content-Type": "application/json" },
  });
});
```

-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION trg_update_form_stats()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE lead_forms SET
    total_submissions = (
      SELECT COUNT(*) FROM form_submissions
      WHERE form_id = NEW.form_id AND status != 'spam'
    ),
    updated_at = NOW()
  WHERE id = NEW.form_id;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_form_submission_stats ON form_submissions;
CREATE TRIGGER trg_form_submission_stats
  AFTER INSERT ON form_submissions
  FOR EACH ROW EXECUTE FUNCTION trg_update_form_stats();

-- Aggiorna total_views
CREATE OR REPLACE FUNCTION trg_update_form_views()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE lead_forms SET
    total_views = (SELECT COUNT(*) FROM form_views WHERE form_id = NEW.form_id),
    updated_at = NOW()
  WHERE id = NEW.form_id;
  RETURN NEW;
END;

FUNZIONE A: supabase/functions/form-submit/index.ts
────────────────────────────────────────────────────────────────────
Gestisce la submission pubblica del form: crea/aggiorna il contatto,
lancia le automazioni marketing, attribuisce la campagna.
È PUBBLICA (no JWT) perché viene chiamata dal browser del visitatore.

```typescript
// supabase/functions/form-submit/index.ts
// verify_jwt: false

import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const supabase = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
);

const CORS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type",
};

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response(null, { headers: CORS });
  if (req.method !== "POST") return new Response("Method not allowed", { status: 405 });

  const json = (data: unknown, status = 200) =>
    new Response(JSON.stringify(data), {
      status, headers: { ...CORS, "Content-Type": "application/json" },
    });

  let body: Record<string, unknown>;
  try { body = await req.json(); } catch { return json({ error: "Invalid JSON" }, 400); }

  const { form_id, form_slug, data: formData, utm, meta } = body as {
    form_id?: string;
    form_slug?: string;
    data: Record<string, string>;
    utm?: Record<string, string>;
    meta?: Record<string, string>;
  };

  // Recupera il form
  let formQuery = supabase.from("lead_forms").select("*");
  if (form_id) formQuery = formQuery.eq("id", form_id);
  else if (form_slug) formQuery = formQuery.eq("slug", form_slug);
  else return json({ error: "form_id o form_slug richiesto" }, 400);

  const { data: form } = await formQuery.eq("status", "published").single();
  if (!form) return json({ error: "Form not found or not published" }, 404);

  const fields = form.fields as Array<Record<string, unknown>>;
  const settings = form.settings as Record<string, unknown>;

  // ── Validazione campi required ────────────────────────────
  const errors: Record<string, string> = {};
  for (const field of fields) {
    if (field.required && field.type !== "divider" && field.type !== "heading") {
      const val = formData[field.id as string];
      if (!val || String(val).trim() === "") {
        errors[field.id as string] = `${field.label} è obbligatorio`;
      }
    }
  }
  if (Object.keys(errors).length > 0) return json({ success: false, errors }, 400);

  // ── Mappa i campi del form ai campi del contatto ───────────
  const contactData: Record<string, unknown> = {
    company_id: form.company_id,
    acquisition_source: "form",
    utm_source: utm?.utm_source,
    utm_medium: utm?.utm_medium,
    utm_campaign: utm?.utm_campaign,
    campaign_id: form.campaign_id || null,
    status: "lead",
  };

  for (const field of fields) {
    const val = formData[field.id as string];
    if (!val || !field.mapping) continue;

    const mapping = field.mapping as string;
    if (mapping === "first_name") contactData.first_name = val;
    else if (mapping === "last_name") contactData.last_name = val;
    else if (mapping === "email") contactData.email = val;
    else if (mapping === "phone") contactData.phone = val;
    else if (mapping === "company") contactData.company = val;
    else if (mapping === "notes") contactData.notes = val;
    // Campi custom → salvati in extra_data JSONB se la colonna esiste
  }

  // Nome obbligatorio: prova a costruirlo
  if (!contactData.first_name && !contactData.last_name) {
    const nameField = fields.find(f => f.mapping === "first_name" || f.mapping === "last_name");
    if (!nameField) contactData.first_name = "Lead";
  }

  // ── Crea o aggiorna contatto (upsert per email) ────────────
  let contactId: string | null = null;
  let isNew = true;

  if (contactData.email) {
    const { data: existing } = await supabase
      .from("contacts")
      .select("id")
      .eq("company_id", form.company_id)
      .eq("email", contactData.email)
      .maybeSingle();

    if (existing) {
      contactId = existing.id;
      isNew = false;
      // Aggiorna solo campi non vuoti
      const updateData = Object.fromEntries(
        Object.entries(contactData).filter(([, v]) => v !== null && v !== undefined)
      );
      await supabase.from("contacts").update(updateData).eq("id", contactId);
    }
  }

  if (!contactId) {
    // Assegna pipeline stage se configurato
    if (settings.pipeline_stage) contactData.pipeline_stage = settings.pipeline_stage;
    if (settings.auto_assign_to) contactData.assigned_to = settings.auto_assign_to;

    const { data: newContact, error: createError } = await supabase
      .from("contacts")
      .insert(contactData)
      .select("id")
      .single();

    if (createError || !newContact) {
      return json({ error: "Errore creazione contatto" }, 500);
    }
    contactId = newContact.id;
  }

  // ── Salva submission ──────────────────────────────────────
  const ua = req.headers.get("user-agent") || "";
  const ip = req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "unknown";
  const ipHash = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(ip))
    .then(buf => Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2,"0")).join("").substring(0, 16));

  const device_type = /mobile|android|iphone/i.test(ua) ? "mobile" :
                      /tablet|ipad/i.test(ua) ? "tablet" : "desktop";

  await supabase.from("form_submissions").insert({
    form_id: form.id,
    company_id: form.company_id,
    data: formData,
    contact_id: contactId,
    is_new_contact: isNew,
    utm_source: utm?.utm_source,
    utm_medium: utm?.utm_medium,
    utm_campaign: utm?.utm_campaign,
    campaign_id: form.campaign_id || null,
    ip_hash: ipHash,
    user_agent: ua.substring(0, 200),
    referrer: meta?.referrer || null,
    landing_page: meta?.landing_page || null,
    device_type,
    status: "processed",
    processed_at: new Date().toISOString(),
  });

  // ── Attribuzione campagna ─────────────────────────────────
  if (form.campaign_id) {
    await supabase.from("campaign_attributions").insert({
      campaign_id: form.campaign_id,
      company_id: form.company_id,
      entity_type: "contact",
      entity_id: contactId,
      attribution_model: "first_touch",
      source: "form",
    });

    // Aggiorna conversion sul click corrispondente (se esiste)
    if (utm?.utm_campaign && ipHash) {
      await supabase.from("campaign_clicks")
        .update({ converted: true, contact_id: contactId })
        .eq("campaign_id", form.campaign_id)
        .eq("ip_hash", ipHash)
        .eq("converted", false)
        .order("clicked_at", { ascending: false })
        .limit(1);
    }
  }

  // ── Innesca automazioni marketing (trigger: contact_created) ─
  // Usa la stessa logica delle Marketing Automations esistenti
  // Il contatto è già stato creato/aggiornato → i trigger DB esistenti
  // per automation_flows si attiveranno automaticamente se configurati.

  // ── Notifica email se configurata ─────────────────────────
  const resendKey = Deno.env.get("RESEND_API_KEY");
  if (settings.notifyOnSubmit && settings.notifyEmail && resendKey) {
    const contactName = `${contactData.first_name || ""} ${contactData.last_name || ""}`.trim();
    const fieldsSummary = fields
      .filter(f => f.type !== "divider" && f.type !== "heading" && formData[f.id as string])
      .map(f => `<tr><td style="padding:4px 8px;font-weight:600">${f.label}</td><td style="padding:4px 8px">${formData[f.id as string]}</td></tr>`)
      .join("");

    await fetch("https://api.resend.com/emails", {
      method: "POST",
      headers: { Authorization: `Bearer ${resendKey}`, "Content-Type": "application/json" },
      body: JSON.stringify({
        from: "Gestionale <noreply@gestionale.app>",
        to: [settings.notifyEmail],
        subject: `🔔 Nuovo lead da ${form.name}: ${contactName}`,
        html: `
          <h2>Nuovo lead acquisito</h2>
          <p>Il form <strong>${form.name}</strong> ha ricevuto una nuova compilazione.</p>
          <table style="border-collapse:collapse;width:100%">${fieldsSummary}</table>
          <p style="margin-top:16px">
            <a href="${Deno.env.get("APP_URL")}/azienda/contatti/${contactId}"
               style="background:#1E40AF;color:white;padding:8px 16px;text-decoration:none;border-radius:4px">
              Visualizza contatto
            </a>
          </p>
        `,
      }),
    });
  }

  // ── Risposta con redirect URL se configurato ──────────────
  return json({
    success: true,
    contact_id: contactId,
    is_new_contact: isNew,
    redirect_url: settings.redirectUrl || null,
    thank_you_title: settings.thankYouTitle || "Grazie!",
    thank_you_message: settings.thankYouMessage || "Abbiamo ricevuto la tua richiesta. Ti contatteremo presto.",
  });
});
```

// supabase/functions/form-render/index.ts
// verify_jwt: false

import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const supabase = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
);

Deno.serve(async (req) => {
  const url = new URL(req.url);
  const slug = url.pathname.split("/").pop();

  if (!slug) return new Response("Not found", { status: 404 });

  const { data: form } = await supabase
    .from("lead_forms")
    .select("*")
    .eq("slug", slug)
    .eq("status", "published")
    .single();

  if (!form) {
    return new Response("<h1>Form non trovato</h1>", {
      status: 404, headers: { "Content-Type": "text/html" },
    });
  }

```sql
-- Funzione chiamata da form-submit edge function dopo aver creato il contatto
-- Inserisce nella automation_queue i flow con trigger 'form_submitted' attivi
CREATE OR REPLACE FUNCTION trigger_form_automations(
  p_company_id UUID,
  p_form_id UUID,
  p_contact_id UUID,
  p_submission_data JSONB
)
RETURNS VOID AS $$
DECLARE v_flow RECORD; v_trigger_node UUID; v_first_action UUID; v_enrollment_id UUID;
BEGIN
  FOR v_flow IN
    SELECT af.id FROM automation_flows af
    WHERE af.company_id = p_company_id
      AND af.is_active = true
      AND af.trigger_type = 'form_submitted'
      AND (
        af.trigger_config->>'form_id' = p_form_id::TEXT
        OR af.trigger_config->>'form_id' IS NULL  -- qualsiasi form
      )
  LOOP
    SELECT id INTO v_trigger_node FROM automation_nodes
    WHERE flow_id = v_flow.id AND node_type = 'trigger' LIMIT 1;

    SELECT target_node_id INTO v_first_action FROM automation_connections
    WHERE flow_id = v_flow.id AND source_node_id = v_trigger_node LIMIT 1;

    IF v_first_action IS NULL THEN CONTINUE; END IF;

    INSERT INTO automation_enrollments (
      flow_id, company_id, contact_id, current_node_id, status, context_data
    ) VALUES (
      v_flow.id, p_company_id, p_contact_id, v_first_action, 'active',
      jsonb_build_object('form_id', p_form_id, 'submission_data', p_submission_data)
    ) RETURNING id INTO v_enrollment_id;

    INSERT INTO automation_queue (
      enrollment_id, flow_id, node_id, company_id, contact_id, scheduled_at, status
    ) VALUES (
      v_enrollment_id, v_flow.id, v_first_action, p_company_id, p_contact_id, NOW(), 'pending'
    );

    UPDATE automation_flows SET total_runs = total_runs + 1 WHERE id = v_flow.id;
  END LOOP;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```
