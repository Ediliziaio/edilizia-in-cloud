MODULO A: UTM Attribution tracking stile GHL
MODULO B: Form Builder embeddabile (in Impostazioni)

FILOSOFIA:
  NON si creano campagne nel sistema.
  Il sistema LEGGE gli UTM dalle URL delle pagine dove arrivano i lead
  (via Meta Ads, Google Ads, TikTok, ecc.) e attribuisce automaticamente
  ogni contatto/lead alla sorgente corretta.

  Proprio come GoHighLevel:
    → fbclid, gclid, ttclid, msclkid vengono catturati e salvati
    → UTM standard: utm_source, utm_medium, utm_campaign, utm_content, utm_term
    → Ad-level: utm_campaign (= nome campagna), utm_content (= nome adset/ad)
    → Prima visita = first touch; ultima visita prima della conversione = last touch
    → Ogni Contatto ha una tab "Attribuzione" che mostra da dove è arrivato
    → La Reportistica mostra breakdown per source / medium / campaign / ad

POSIZIONAMENTO SIDEBAR:
  Modulo A → nessuna voce di menu propria (i dati si vedono nel Contatto e in Reportistica)
  Modulo B → Impostazioni → sezione "Form & Acquisizione"

```sql
-- ════════════════════════════════════════════════════════════════
-- TABELLA 1: attribution_sessions
-- Ogni sessione di visita con i suoi parametri UTM
-- Una sessione = una visita univoca identificata da session_id (uuid generato lato JS)
-- ════════════════════════════════════════════════════════════════
CREATE TABLE attribution_sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,

  -- Identificatori sessione
  session_id TEXT NOT NULL,                    -- UUID generato dal tracking script
  visitor_id TEXT,                              -- ID visitatore persistente (cookie 1st party)

  -- UTM standard
  utm_source TEXT,        -- es. "facebook", "google", "instagram"
  utm_medium TEXT,        -- es. "cpc", "social", "email"
  utm_campaign TEXT,      -- nome campagna: es. "promo_estate_2025"
  utm_content TEXT,       -- nome adset o variante: es. "adset_remarketing"
  utm_term TEXT,          -- parola chiave (Google): es. "software gestionale"

  -- Click ID piattaforme pubblicitarie
  fbclid TEXT,            -- Facebook / Instagram click ID
  gclid TEXT,             -- Google Ads click ID
  ttclid TEXT,            -- TikTok click ID
  msclkid TEXT,           -- Microsoft Ads click ID
  li_fat_id TEXT,         -- LinkedIn click ID

  -- URL completa di atterraggio
  landing_url TEXT,
  referrer TEXT,

  -- Device / browser
  user_agent TEXT,
  device_type TEXT CHECK (device_type IN ('desktop', 'mobile', 'tablet', 'unknown')) DEFAULT 'unknown',

  -- IP hashed per privacy (SHA-256, mai raw)
  ip_hash TEXT,

  -- Stato
  converted_at TIMESTAMPTZ,         -- quando ha convertito (compilato form o creato contatto)
  contact_id UUID REFERENCES contacts(id) ON DELETE SET NULL,

  created_at TIMESTAMPTZ DEFAULT now(),

  UNIQUE(company_id, session_id)
);

-- ════════════════════════════════════════════════════════════════
-- TABELLA 2: contact_attributions
-- Attribuzione finale associata a un Contatto
-- Mantiene first touch e last touch separati
-- ════════════════════════════════════════════════════════════════
CREATE TABLE contact_attributions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  contact_id UUID NOT NULL REFERENCES contacts(id) ON DELETE CASCADE UNIQUE,

  -- First touch (prima visita in assoluto)
  ft_source TEXT,
  ft_medium TEXT,
  ft_campaign TEXT,
  ft_content TEXT,
  ft_term TEXT,
  ft_fbclid TEXT,
  ft_gclid TEXT,
  ft_ttclid TEXT,
  ft_landing_url TEXT,
  ft_session_id TEXT,
  ft_at TIMESTAMPTZ,

  -- Last touch (ultima visita prima della conversione)
  lt_source TEXT,
  lt_medium TEXT,
  lt_campaign TEXT,
  lt_content TEXT,
  lt_term TEXT,
  lt_fbclid TEXT,
  lt_gclid TEXT,
  lt_ttclid TEXT,
  lt_landing_url TEXT,
  lt_session_id TEXT,
  lt_at TIMESTAMPTZ,

  -- Totale touchpoint registrati
  touchpoint_count INT DEFAULT 1,

  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- ════════════════════════════════════════════════════════════════
-- ALTER TABLE contacts: aggiungi colonne quick-access attribution
-- (ridondanti con contact_attributions ma utili per query rapide e filtri)
-- ════════════════════════════════════════════════════════════════
ALTER TABLE contacts
  ADD COLUMN IF NOT EXISTS attr_source TEXT,
  ADD COLUMN IF NOT EXISTS attr_medium TEXT,
  ADD COLUMN IF NOT EXISTS attr_campaign TEXT,
  ADD COLUMN IF NOT EXISTS attr_content TEXT,
  ADD COLUMN IF NOT EXISTS attr_model TEXT CHECK (attr_model IN ('first_touch', 'last_touch')) DEFAULT 'last_touch';
-- attr_model: quale modello viene usato per popolare attr_* (di default last touch)

-- ════════════════════════════════════════════════════════════════
-- INDICI
-- ════════════════════════════════════════════════════════════════
CREATE INDEX idx_attribution_sessions_company ON attribution_sessions(company_id);
CREATE INDEX idx_attribution_sessions_session ON attribution_sessions(session_id);
CREATE INDEX idx_attribution_sessions_contact ON attribution_sessions(contact_id);
CREATE INDEX idx_attribution_sessions_fbclid ON attribution_sessions(fbclid) WHERE fbclid IS NOT NULL;

-- RLS
-- ════════════════════════════════════════════════════════════════
ALTER TABLE attribution_sessions ENABLE ROW LEVEL SECURITY;
ALTER TABLE contact_attributions ENABLE ROW LEVEL SECURITY;

CREATE POLICY "attribution_sessions_company_isolation" ON attribution_sessions
  USING (company_id = (SELECT company_id FROM users WHERE id = auth.uid()));

CREATE POLICY "contact_attributions_company_isolation" ON contact_attributions
  USING (company_id = (SELECT company_id FROM users WHERE id = auth.uid()));

-- ════════════════════════════════════════════════════════════════
-- FUNZIONE: get_attribution_report(company_id, date_from, date_to, group_by)
-- Ritorna breakdown lead per source/medium/campaign/content
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION get_attribution_report(
  p_company_id UUID,
  p_date_from TIMESTAMPTZ DEFAULT NOW() - INTERVAL '30 days',
  p_date_to TIMESTAMPTZ DEFAULT NOW(),
  p_group_by TEXT DEFAULT 'source'  -- 'source' | 'medium' | 'campaign' | 'content'
)
RETURNS TABLE (
  dimension TEXT,
  total_sessions BIGINT,
  total_leads BIGINT,
  conversion_rate NUMERIC,
  sources TEXT[]
) LANGUAGE sql STABLE AS $$
  SELECT
    CASE p_group_by
      WHEN 'source'   THEN COALESCE(utm_source, 'direct')
      WHEN 'medium'   THEN COALESCE(utm_medium, 'none')
      WHEN 'campaign' THEN COALESCE(utm_campaign, 'none')
      WHEN 'content'  THEN COALESCE(utm_content, 'none')
    END AS dimension,
    COUNT(*) AS total_sessions,
    COUNT(contact_id) AS total_leads,
    ROUND(
      COUNT(contact_id)::NUMERIC / NULLIF(COUNT(*), 0) * 100, 2
    ) AS conversion_rate,
    ARRAY_AGG(DISTINCT COALESCE(utm_source, 'direct')) AS sources
  FROM attribution_sessions
  WHERE
    company_id = p_company_id
    AND created_at BETWEEN p_date_from AND p_date_to
  GROUP BY 1
  ORDER BY total_leads DESC;
$$;

-- ════════════════════════════════════════════════════════════════
-- FUNZIONE: attach_attribution_to_contact(session_id, contact_id, company_id)
-- Chiamata quando un lead viene identificato (form submit o creazione manuale)
-- Popola contact_attributions con first/last touch
-- ════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION attach_attribution_to_contact(
  p_session_id TEXT,
  p_contact_id UUID,
  p_company_id UUID
)
RETURNS void LANGUAGE plpgsql AS $$
DECLARE
  v_session attribution_sessions%ROWTYPE;
  v_existing contact_attributions%ROWTYPE;
BEGIN
  -- Prendi la sessione
  SELECT * INTO v_session
  FROM attribution_sessions
  WHERE session_id = p_session_id AND company_id = p_company_id;

  IF NOT FOUND THEN RETURN; END IF;

  -- Aggiorna sessione come convertita
  UPDATE attribution_sessions
  SET contact_id = p_contact_id, converted_at = now()
  WHERE session_id = p_session_id AND company_id = p_company_id;

  -- Cerca attribuzione esistente per il contatto
  SELECT * INTO v_existing FROM contact_attributions
  WHERE contact_id = p_contact_id AND company_id = p_company_id;

  IF NOT FOUND THEN
    -- Prima conversione: first touch = last touch = questa sessione
    INSERT INTO contact_attributions (
      company_id, contact_id,
      ft_source, ft_medium, ft_campaign, ft_content, ft_term,
      ft_fbclid, ft_gclid, ft_ttclid, ft_landing_url, ft_session_id, ft_at,
      lt_source, lt_medium, lt_campaign, lt_content, lt_term,
      lt_fbclid, lt_gclid, lt_ttclid, lt_landing_url, lt_session_id, lt_at,
      touchpoint_count
    ) VALUES (
      p_company_id, p_contact_id,
      v_session.utm_source, v_session.utm_medium, v_session.utm_campaign,
      v_session.utm_content, v_session.utm_term,
      v_session.fbclid, v_session.gclid, v_session.ttclid,
      v_session.landing_url, p_session_id, v_session.created_at,
      -- last touch uguale al first touch (primo contatto)
      v_session.utm_source, v_session.utm_medium, v_session.utm_campaign,
      v_session.utm_content, v_session.utm_term,
      v_session.fbclid, v_session.gclid, v_session.ttclid,
      v_session.landing_url, p_session_id, v_session.created_at,
      1
    );
  ELSE
    -- Contatto già esistente: aggiorna solo last touch e incrementa touchpoint
    UPDATE contact_attributions SET
      lt_source = v_session.utm_source,
      lt_medium = v_session.utm_medium,
      lt_campaign = v_session.utm_campaign,
      lt_content = v_session.utm_content,
      lt_term = v_session.utm_term,
      lt_fbclid = v_session.fbclid,
      lt_gclid = v_session.gclid,
      lt_ttclid = v_session.ttclid,
      lt_landing_url = v_session.landing_url,
      lt_session_id = p_session_id,
      lt_at = v_session.created_at,
      touchpoint_count = touchpoint_count + 1,
      updated_at = now()
    WHERE contact_id = p_contact_id AND company_id = p_company_id;
  END IF;

  -- Aggiorna colonne quick-access su contacts (usa last touch di default)
  UPDATE contacts SET
    attr_source   = v_session.utm_source,
    attr_medium   = v_session.utm_medium,
    attr_campaign = v_session.utm_campaign,
    attr_content  = v_session.utm_content
  WHERE id = p_contact_id;
END;
$$;
```

### Edge Function: supabase/functions/attribution-capture/index.ts

```typescript
import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from "jsr:@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "content-type, x-company-id",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    return new Response(null, { headers: corsHeaders });
  }

  const supabase = createClient(
    Deno.env.get("SUPABASE_URL")!,
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
  );

  try {
    const body = await req.json();
    const {
      company_id,
      session_id,
      visitor_id,
      // UTM standard
      utm_source,
      utm_medium,
      utm_campaign,
      utm_content,
      utm_term,
      // Click IDs piattaforme
      fbclid,
      gclid,
      ttclid,
      msclkid,
      li_fat_id,
      // Page info
      landing_url,
      referrer,
      user_agent,
    } = body;

    if (!company_id || !session_id) {
      return new Response(
        JSON.stringify({ error: "company_id e session_id obbligatori" }),
        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    // Verifica che company_id esista
    const { data: company } = await supabase
      .from("companies")
      .select("id")
      .eq("id", company_id)
      .single();

    if (!company) {
      return new Response(
        JSON.stringify({ error: "company non trovata" }),
        { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    // Detect device type
    const ua = user_agent || req.headers.get("user-agent") || "";
    let device_type: string = "desktop";
    if (/mobile|android|iphone|ipod/i.test(ua)) device_type = "mobile";
    else if (/ipad|tablet/i.test(ua)) device_type = "tablet";

    // Hash IP per privacy (SHA-256)
    const rawIp =
      req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ||
      req.headers.get("x-real-ip") ||
      "unknown";
    const ipBuffer = await crypto.subtle.digest(
      "SHA-256",
      new TextEncoder().encode(rawIp)
    );
    const ipHash = Array.from(new Uint8Array(ipBuffer))
      .map((b) => b.toString(16).padStart(2, "0"))
      .join("");

    // Upsert sessione (se la stessa session_id arriva di nuovo, aggiorna)
    const { error } = await supabase
      .from("attribution_sessions")
      .upsert({
        company_id,
        session_id,
        visitor_id: visitor_id || null,
        utm_source: utm_source || null,
        utm_medium: utm_medium || null,
        utm_campaign: utm_campaign || null,
        utm_content: utm_content || null,
        utm_term: utm_term || null,
        fbclid: fbclid || null,
        gclid: gclid || null,
        ttclid: ttclid || null,
        msclkid: msclkid || null,
        li_fat_id: li_fat_id || null,
        landing_url: landing_url || null,
        referrer: referrer || null,
        user_agent: ua || null,
        device_type,
        ip_hash: ipHash,
      }, {
        onConflict: "company_id,session_id",
        ignoreDuplicates: false,
      });

    if (error) throw error;

    return new Response(
      JSON.stringify({ ok: true, session_id }),
      { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  } catch (err) {
    console.error("attribution-capture error:", err);
    return new Response(
      JSON.stringify({ error: "Errore interno" }),
      { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  }
});
```

```typescript
// src/constants/trackingSnippet.ts

export function getTrackingSnippet(companyId: string, supabaseUrl: string): string {
  return `<!-- UTM Attribution Tracker -->
<script>
(function() {
  var COMPANY_ID = '${companyId}';
  var ENDPOINT   = '${supabaseUrl}/functions/v1/attribution-capture';

  // ── Genera / recupera IDs persistenti ──────────────────────────
  function uuid() {
    return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
      var r = Math.random() * 16 | 0;
      return (c === 'x' ? r : (r & 0x3 | 0x8)).toString(16);
    });
  }

  var visitorId = localStorage.getItem('_attr_vid');
  if (!visitorId) { visitorId = uuid(); localStorage.setItem('_attr_vid', visitorId); }

  var sessionId = sessionStorage.getItem('_attr_sid');
  if (!sessionId) { sessionId = uuid(); sessionStorage.setItem('_attr_sid', sessionId); }

  // ── Leggi parametri UTM e click ID dall'URL ────────────────────
  var params = new URLSearchParams(window.location.search);
  var utmKeys = ['utm_source','utm_medium','utm_campaign','utm_content','utm_term',
                 'fbclid','gclid','ttclid','msclkid','li_fat_id'];
  var data = {};
  var hasUtm = false;

  utmKeys.forEach(function(k) {
    var v = params.get(k);
    if (v) { data[k] = v; hasUtm = true; }
  });

  // ── Persisti UTM in localStorage (per multi-pagina) ──────────
  if (hasUtm) {
    localStorage.setItem('_attr_utm', JSON.stringify(data));
  } else {
    var stored = localStorage.getItem('_attr_utm');
    if (stored) {
      try { data = JSON.parse(stored); hasUtm = true; } catch(e) {}
    }
  }

  // ── Invia al backend solo se ci sono UTM o click ID ───────────
  if (!hasUtm) return;

  var payload = Object.assign({}, data, {
    company_id:   COMPANY_ID,
    session_id:   sessionId,
    visitor_id:   visitorId,
    landing_url:  window.location.href,
    referrer:     document.referrer || null,
    user_agent:   navigator.userAgent
  });

  fetch(ENDPOINT, {
    method:  'POST',
    headers: { 'Content-Type': 'application/json' },
    body:    JSON.stringify(payload),
    keepalive: true
  }).catch(function() {});  // silenzioso in caso di errore

  // ── Esponi session_id globalmente (usato dai form per l'attribuzione) ──
  window._attrSessionId = sessionId;
})();
</script>`;
}
```

### Hook: src/hooks/useContactAttribution.ts

```typescript
import { useQuery } from "@tanstack/react-query";
import { supabase } from "@/lib/supabase";

export interface ContactAttributionData {
  id: string;
  contact_id: string;
  // First touch
  ft_source: string | null;
  ft_medium: string | null;
  ft_campaign: string | null;
  ft_content: string | null;
  ft_term: string | null;
  ft_fbclid: string | null;
  ft_gclid: string | null;
  ft_ttclid: string | null;
  ft_landing_url: string | null;
  ft_at: string | null;
  // Last touch
  lt_source: string | null;
  lt_medium: string | null;
  lt_campaign: string | null;
  lt_content: string | null;
  lt_term: string | null;
  lt_fbclid: string | null;
  lt_gclid: string | null;
  lt_ttclid: string | null;
  lt_landing_url: string | null;
  lt_at: string | null;
  // Meta
  touchpoint_count: number;
  // Tutte le sessioni del contatto
  sessions: AttributionSession[];
}

export interface AttributionSession {
  session_id: string;
  utm_source: string | null;
  utm_medium: string | null;
  utm_campaign: string | null;
  utm_content: string | null;
  fbclid: string | null;
  gclid: string | null;
  ttclid: string | null;
  landing_url: string | null;
  device_type: string;
  created_at: string;
  converted_at: string | null;
}

export function useContactAttribution(contactId: string) {
  return useQuery({
    queryKey: ["contact-attribution", contactId],
    queryFn: async (): Promise<ContactAttributionData | null> => {
      // Attribuzione principale
      const { data: attr } = await supabase
        .from("contact_attributions")
        .select("*")
        .eq("contact_id", contactId)
        .single();

      if (!attr) return null;

      // Tutte le sessioni associate
      const { data: sessions } = await supabase
        .from("attribution_sessions")
        .select("session_id, utm_source, utm_medium, utm_campaign, utm_content, fbclid, gclid, ttclid, landing_url, device_type, created_at, converted_at")
        .eq("contact_id", contactId)
        .order("created_at", { ascending: false });

      return { ...attr, sessions: sessions || [] };
    },
    enabled: !!contactId,
  });
}
```

```sql
DECLARE

  v_flow RECORD;
BEGIN
  SELECT * INTO v_sub FROM form_submissions WHERE id = p_submission_id;
  IF NOT FOUND OR v_sub.contact_id IS NULL THEN RETURN; END IF;

  FOR v_flow IN
    SELECT id FROM automation_flows
    WHERE company_id = v_sub.company_id
      AND is_active = true
      AND trigger_type = 'form_submitted'
      AND (trigger_config->>'form_id' = v_sub.form_id::text
           OR trigger_config->>'form_id' IS NULL)
  LOOP
    INSERT INTO automation_enrollments (flow_id, contact_id, company_id, status, trigger_data)

      v_flow.id,
      v_sub.contact_id,
      v_sub.company_id,
      'active',

        'submission_id', p_submission_id,
        'form_id', v_sub.form_id,
        'utm_source', v_sub.utm_source,
        'utm_campaign', v_sub.utm_campaign
      )
    )
    ON CONFLICT DO NOTHING;

END;
```

### supabase/functions/form-submit/index.ts

```typescript
import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from "jsr:@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") return new Response(null, { headers: corsHeaders });

  const supabase = createClient(
    Deno.env.get("SUPABASE_URL")!,
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
  );

  try {
    const body = await req.json();
    const {
      form_id,
      company_id,
      data: formData,  // { field_id: valore }
      session_id,      // dal tracking script window._attrSessionId
      visitor_id,
    } = body;

    if (!form_id || !company_id || !formData) {
      return new Response(
        JSON.stringify({ error: "Dati mancanti" }),
        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    // Recupera il form e la sua configurazione
    const { data: form, error: formErr } = await supabase
      .from("lead_forms")
      .select("*")
      .eq("id", form_id)
      .eq("company_id", company_id)
      .eq("is_active", true)
      .eq("is_published", true)
      .single();

    if (formErr || !form) {
      return new Response(
        JSON.stringify({ error: "Form non trovato o non attivo" }),
        { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    // Recupera UTM dalla sessione (se presente)
    let sessionData: Record<string, string | null> = {};
    if (session_id) {
      const { data: session } = await supabase
        .from("attribution_sessions")
        .select("utm_source, utm_medium, utm_campaign, utm_content, utm_term, fbclid, gclid, ttclid")
        .eq("session_id", session_id)
        .eq("company_id", company_id)
        .single();
      if (session) sessionData = session;
    }

    // Estrai mappature campi → contatto
    const fields: Array<{
      id: string; type: string; label: string; mapping?: string; required?: boolean;
    }> = form.fields || [];

    // Validazione campi required
    const missingFields: string[] = [];
    for (const field of fields) {
      if (field.required && field.type !== "heading" && field.type !== "paragraph" && field.type !== "divider") {
        const val = formData[field.id];
        if (!val || (typeof val === "string" && val.trim() === "")) {
          missingFields.push(field.label);
        }
      }
    }
    if (missingFields.length > 0) {
      return new Response(
        JSON.stringify({ error: `Campi obbligatori mancanti: ${missingFields.join(", ")}` }),
        { status: 422, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    // Costruisci oggetto contatto dalle mappature
    const contactData: Record<string, string> = {};
    for (const field of fields) {
      if (field.mapping && formData[field.id] !== undefined) {
        // mapping = "first_name" | "last_name" | "email" | "phone" | "company" | "notes" | ...
        contactData[field.mapping] = formData[field.id];
      }
    }

    // IP hashing
    const rawIp =
      req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ||
      req.headers.get("x-real-ip") || "unknown";
    const ipBuffer = await crypto.subtle.digest(
      "SHA-256", new TextEncoder().encode(rawIp)
    );
    const ipHash = Array.from(new Uint8Array(ipBuffer))
      .map((b) => b.toString(16).padStart(2, "0")).join("");

    // Device type
    const ua = req.headers.get("user-agent") || "";
    let device_type = "desktop";
    if (/mobile|android|iphone|ipod/i.test(ua)) device_type = "mobile";
    else if (/ipad|tablet/i.test(ua)) device_type = "tablet";

    // Upsert contatto per email (deduplication)
    let contactId: string | null = null;
    const settings = form.settings || {};

    if (settings.createContact !== false && contactData.email) {
      const { data: existingContact } = await supabase
        .from("contacts")
        .select("id")
        .eq("company_id", company_id)
        .eq("email", contactData.email)
        .single();

      if (existingContact) {
        // Aggiorna campi del contatto esistente
        const updatePayload: Record<string, unknown> = {
          ...contactData,
          attr_source: sessionData.utm_source || undefined,
          attr_medium: sessionData.utm_medium || undefined,
          attr_campaign: sessionData.utm_campaign || undefined,
          attr_content: sessionData.utm_content || undefined,
        };
        // Rimuovi campi undefined
        Object.keys(updatePayload).forEach(k => updatePayload[k] === undefined && delete updatePayload[k]);

        await supabase
          .from("contacts")
          .update(updatePayload)
          .eq("id", existingContact.id);
        contactId = existingContact.id;
      } else {
        // Crea nuovo contatto
        const newContact: Record<string, unknown> = {
          company_id,
          ...contactData,
          source: "form",
          attr_source: sessionData.utm_source || null,
          attr_medium: sessionData.utm_medium || null,
          attr_campaign: sessionData.utm_campaign || null,
          attr_content: sessionData.utm_content || null,
        };

        // Assegna a utente se configurato nel form
        if (settings.assignedUserId) newContact.assigned_user_id = settings.assignedUserId;

        // Tags di default
        if (settings.defaultTags?.length > 0) newContact.tags = settings.defaultTags;

        const { data: created } = await supabase
          .from("contacts")
          .insert(newContact)
          .select("id")
          .single();

        contactId = created?.id || null;

        // Se c'è un pipeline configurato, crea anche un deal/opportunità
        if (contactId && settings.defaultPipelineId && settings.defaultStageId) {
          await supabase.from("opportunities").insert({
            company_id,
            contact_id: contactId,
            pipeline_id: settings.defaultPipelineId,
            stage_id: settings.defaultStageId,
            title: `Lead da form: ${form.name}`,
          });
        }
      }
    }

    // Crea la submission
    const { data: submission } = await supabase
      .from("form_submissions")
      .insert({
        form_id,
        company_id,
        data: formData,
        contact_id: contactId,
        session_id: session_id || null,
        utm_source: sessionData.utm_source || null,
        utm_medium: sessionData.utm_medium || null,
        utm_campaign: sessionData.utm_campaign || null,
        utm_content: sessionData.utm_content || null,
        utm_term: sessionData.utm_term || null,
        fbclid: sessionData.fbclid || null,
        gclid: sessionData.gclid || null,
        ip_hash: ipHash,
        user_agent: ua,
        device_type,
      })
      .select("id")
      .single();

    // Attribuzione UTM sul contatto
    if (contactId && session_id) {
      await supabase.rpc("attach_attribution_to_contact", {
        p_session_id: session_id,
        p_contact_id: contactId,
        p_company_id: company_id,
      });
    }

    // Trigger automazioni Marketing
    if (submission?.id) {
      await supabase.rpc("trigger_form_automations", {
        p_submission_id: submission.id,
      });
    }

    // Notifica email interna
    if (settings.notifyOnSubmit && settings.notifyEmail) {
      // Usa Supabase Edge Function separata o SMTP diretto
      // (integra con il sistema email esistente del progetto)
      const notifyPayload = {
        to: settings.notifyEmail,
        subject: `Nuovo lead da form: ${form.name}`,
        body: Object.entries(formData)
          .map(([k, v]) => {
            const field = fields.find(f => f.id === k);
            return `${field?.label || k}: ${v}`;
          })
          .join("\n"),
      };
      // Invoke sistema email esistente — adatta al tuo hook
      await supabase.functions.invoke("send-notification-email", {
        body: notifyPayload
      }).catch(() => {}); // Non bloccare la risposta se fallisce
    }

    return new Response(
      JSON.stringify({
        ok: true,
        contact_id: contactId,
        redirect_url: settings.redirectUrl || null,
        success_title: form.theme?.successTitle || "Grazie!",
        success_message: form.theme?.successMessage || "Richiesta ricevuta.",
      }),
      { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  } catch (err) {
    console.error("form-submit error:", err);
    return new Response(
      JSON.stringify({ error: "Errore interno" }),
      { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } }
    );
  }
});
```

### Hook: src/hooks/useFormBuilder.ts

```typescript
import { useState, useCallback } from "react";
import { nanoid } from "nanoid";

export interface FormField {
  id: string;
  type: "text" | "email" | "phone" | "textarea" | "select" | "checkbox" |
        "radio" | "date" | "number" | "file" | "heading" | "paragraph" |
        "divider" | "hidden";
  label: string;
  placeholder?: string;
  required?: boolean;
  options?: string[];             // per select e radio
  mapping?: string;               // "email" | "first_name" | "last_name" | ecc.
  value?: string;                 // per hidden
}

export function useFormBuilder(initialFields: FormField[] = []) {
  const [fields, setFields] = useState<FormField[]>(initialFields);
  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [isDirty, setIsDirty] = useState(false);

  const addField = useCallback((type: FormField["type"]) => {
    const defaults: Partial<Record<FormField["type"], Partial<FormField>>> = {
      email: { label: "Email", mapping: "email", required: true },
      phone: { label: "Telefono", mapping: "phone" },
      text: { label: "Nome" },
      textarea: { label: "Messaggio", placeholder: "Scrivi qui…" },
      select: { label: "Selezione", options: ["Opzione 1", "Opzione 2"] },
      radio: { label: "Scelta", options: ["Sì", "No"] },
      checkbox: { label: "Accetto la privacy policy", required: true },
      heading: { label: "Titolo sezione" },
      paragraph: { label: "Testo descrittivo della sezione." },
    };
    const newField: FormField = {
      id: nanoid(),
      type,
      label: type.charAt(0).toUpperCase() + type.slice(1),
      ...defaults[type],
    };
    setFields(prev => [...prev, newField]);
    setSelectedId(newField.id);
    setIsDirty(true);
  }, []);

  const updateField = useCallback((id: string, updates: Partial<FormField>) => {
    setFields(prev => prev.map(f => f.id === id ? { ...f, ...updates } : f));
    setIsDirty(true);
  }, []);

  const removeField = useCallback((id: string) => {
    setFields(prev => prev.filter(f => f.id !== id));
    setSelectedId(prev => prev === id ? null : prev);
    setIsDirty(true);
  }, []);

  const reorderFields = useCallback((activeId: string, overId: string) => {
    setFields(prev => {
      const oldIndex = prev.findIndex(f => f.id === activeId);
      const newIndex = prev.findIndex(f => f.id === overId);
      const result = [...prev];
      const [removed] = result.splice(oldIndex, 1);
      result.splice(newIndex, 0, removed);
      return result;
    });
    setIsDirty(true);
  }, []);

  const selectedField = fields.find(f => f.id === selectedId) || null;

  return {
    fields,
    selectedId,
    selectedField,
    isDirty,
    setIsDirty,
    addField,
    updateField,
    removeField,
    reorderFields,
    setSelectedId,
  };
}
```
