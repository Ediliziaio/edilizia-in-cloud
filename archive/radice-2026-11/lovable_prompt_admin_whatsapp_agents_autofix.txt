────────────────────────────────────────────────────────────────────────────
BUG 1 — AutomationEnrollmentsTab.tsx: filtro company_id MANCANTE
────────────────────────────────────────────────────────────────────────────

FILE: src/components/marketing/automations/AutomationEnrollmentsTab.tsx

Il problema: la query su `automation_enrollments` usa solo `.eq("flow_id", flowId)`
ma NON filtra per `company_id`. Questo significa che se due aziende diverse
hanno un flow con lo stesso ID (improbabile ma possibile con UUID), i dati
si mescolano. Soprattutto, non c'è protezione server-side aggiuntiva.

FIX — Aggiungi il filtro company_id alla query:

```tsx
// Cerca nell'hook o nella queryFn la query su automation_enrollments
// e aggiungi .eq("company_id", companyId) PRIMA di .eq("flow_id", flowId)

// PRIMA (bug):
const { data, count } = await supabase
  .from("automation_enrollments")
  .select("*", { count: "exact" })
  .eq("flow_id", flowId)
  // ... altri filtri

// DOPO (corretto):
const { data, count } = await supabase
  .from("automation_enrollments")
  .select("*", { count: "exact" })
  .eq("company_id", companyId)   // <-- AGGIUNTO PRIMO
  .eq("flow_id", flowId)
  // ... altri filtri invariati
```

Il `companyId` deve essere passato come prop al componente o letto
da `useCompany()` / `useAdminMarketing()` (se siamo nell'area superadmin).

Aggiorna anche la firma del componente se necessario:
```tsx
// Se companyId non è già tra le props, aggiungilo:
interface AutomationEnrollmentsTabProps {
  flowId: string;
  companyId: string;   // <-- AGGIUNTO
}
```

E aggiorna tutti i punti in cui AutomationEnrollmentsTab è istanziato
(es. MarketingAutomationBuilder.tsx) per passare il companyId:
```tsx
<AutomationEnrollmentsTab flowId={flowId} companyId={companyId} />
```

BUG 4 — MarketingAutomations.tsx: logica statusFilter confusa + enabled guard
────────────────────────────────────────────────────────────────────────────

FILE: src/pages/azienda/marketing/MarketingAutomations.tsx

Il problema: la logica del filtro status usa una tripla ternaria confusa:
```tsx
// PRIMA (confuso):
const statusFilter = urlFilter === "published" ? "published"
  : listFilter === "deleted" ? "archived"
  : listFilter === "needs_review" ? "draft"
  : "all";
```

Questa logica NON gestisce il caso in cui `urlFilter` sia "draft" o "archived"
direttamente dall'URL. Se l'utente naviga a `?filter=archived`, la condizione
non viene mai raggiunta perché il primo ternario controlla solo "published".

FIX — Riscrivi con una logica chiara:
```tsx
// DOPO (leggibile e corretto):
const getStatusFilter = (): "all" | "published" | "draft" | "archived" => {
  // Prima controlla il filtro URL (passato via query param)
  if (urlFilter === "published") return "published";
  if (urlFilter === "draft")     return "draft";
  if (urlFilter === "archived")  return "archived";

  // Poi il filtro tab della UI
  if (listFilter === "needs_review") return "draft";
  if (listFilter === "deleted")      return "archived";

  return "all";
};

const statusFilter = getStatusFilter();
```

Aggiorna anche il componente `AutomationFlowsList` che riceve `statusFilter`
come prop, assicurandoti che accetti i valori `"all" | "published" | "draft" | "archived"`.

Gli Agenti AI sono bot conversazionali configurabili che rispondono
automaticamente su WhatsApp, Email o Chat alle domande dei lead/prospect.

```sql
-- Migration: ai_agents_for_admin_marketing

-- 1. Tabella principale agenti AI
CREATE TABLE IF NOT EXISTS ai_agents (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id      uuid NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  name            text NOT NULL,
  description     text,
  channel         text NOT NULL CHECK (channel IN ('whatsapp', 'email', 'chat', 'all')),
  status          text NOT NULL DEFAULT 'draft' CHECK (status IN ('draft', 'active', 'paused')),
  system_prompt   text NOT NULL DEFAULT '',
  temperature     numeric(3,2) NOT NULL DEFAULT 0.7 CHECK (temperature BETWEEN 0 AND 1),
  max_tokens      int NOT NULL DEFAULT 1000,
  model           text NOT NULL DEFAULT 'gpt-4o-mini',
  -- Comportamento
  fallback_message       text,           -- messaggio se l'AI non sa rispondere
  escalation_trigger     text,           -- parole chiave che escalano a umano
  collect_contact_info   boolean NOT NULL DEFAULT false,
  -- Statistiche (denormalized per performance)
  total_conversations    int NOT NULL DEFAULT 0,
  total_messages         int NOT NULL DEFAULT 0,
  avg_satisfaction       numeric(3,2),
  -- Meta
  created_at      timestamptz NOT NULL DEFAULT now(),
  updated_at      timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX idx_ai_agents_company ON ai_agents(company_id, status);

-- 2. Knowledge base dell'agente (documenti, FAQ, testi)
CREATE TABLE IF NOT EXISTS ai_agent_knowledge (
  id          uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  agent_id    uuid NOT NULL REFERENCES ai_agents(id) ON DELETE CASCADE,
  company_id  uuid NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  type        text NOT NULL CHECK (type IN ('text', 'faq', 'document', 'url')),
  title       text NOT NULL,
  content     text,                       -- per type='text' e 'faq'
  url         text,                       -- per type='url'
  file_path   text,                       -- per type='document' (Supabase Storage)
  file_name   text,
  is_active   boolean NOT NULL DEFAULT true,
  created_at  timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX idx_ai_knowledge_agent ON ai_agent_knowledge(agent_id, is_active);

-- 3. Log conversazioni dell'agente
CREATE TABLE IF NOT EXISTS ai_agent_conversations (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  agent_id      uuid NOT NULL REFERENCES ai_agents(id) ON DELETE CASCADE,
  company_id    uuid NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  contact_id    uuid REFERENCES marketing_contacts(id) ON DELETE SET NULL,
  channel       text NOT NULL,
  status        text NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'completed', 'escalated')),
  messages_count  int NOT NULL DEFAULT 0,
  satisfaction    int CHECK (satisfaction BETWEEN 1 AND 5),
  escalated_to    uuid REFERENCES profiles(id) ON DELETE SET NULL,
  started_at    timestamptz NOT NULL DEFAULT now(),
  ended_at      timestamptz
);

CREATE INDEX idx_ai_conv_agent ON ai_agent_conversations(agent_id, company_id, started_at DESC);

-- 4. Messaggi delle conversazioni
CREATE TABLE IF NOT EXISTS ai_agent_messages (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  conversation_id uuid NOT NULL REFERENCES ai_agent_conversations(id) ON DELETE CASCADE,
  role            text NOT NULL CHECK (role IN ('user', 'assistant', 'system')),
  content         text NOT NULL,
  tokens_used     int,
  created_at      timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX idx_ai_messages_conv ON ai_agent_messages(conversation_id, created_at);

-- 5. RLS per tutte le tabelle
ALTER TABLE ai_agents              ENABLE ROW LEVEL SECURITY;
ALTER TABLE ai_agent_knowledge     ENABLE ROW LEVEL SECURITY;
ALTER TABLE ai_agent_conversations ENABLE ROW LEVEL SECURITY;
ALTER TABLE ai_agent_messages      ENABLE ROW LEVEL SECURITY;

-- Policy standard: company members + superadmin
CREATE POLICY "ai_agents_company_access" ON ai_agents FOR ALL USING (
  EXISTS (SELECT 1 FROM company_members WHERE company_id = ai_agents.company_id AND user_id = auth.uid())
  OR EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'super_admin')
);

CREATE POLICY "ai_knowledge_company_access" ON ai_agent_knowledge FOR ALL USING (

  OR EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'super_admin')
);

CREATE POLICY "ai_conversations_company_access" ON ai_agent_conversations FOR ALL USING (
  EXISTS (SELECT 1 FROM company_members WHERE company_id = ai_agent_conversations.company_id AND user_id = auth.uid())

CREATE POLICY "ai_messages_conv_access" ON ai_agent_messages FOR ALL USING (

    WHERE c.id = ai_agent_messages.conversation_id

CREATE OR REPLACE FUNCTION update_ai_agent_timestamp()
RETURNS TRIGGER LANGUAGE plpgsql AS $$

CREATE TRIGGER trg_ai_agent_updated
```

```ts
export function useAiAgents(companyId: string) {
  return useQuery({
    queryKey: ["ai-agents", companyId],
    queryFn: async () => {
      const { data, error } = await supabase
        .from("ai_agents")
        .select("*")
        .eq("company_id", companyId)
        .order("created_at", { ascending: false });
      if (error) throw error;
      return (data ?? []) as AiAgent[];
    },
    enabled: !!companyId,
    staleTime: 2 * 60 * 1000,
  });
}

export function useAiAgentConversations(agentId: string, page = 0, pageSize = 25) {
  return useQuery({
    queryKey: ["ai-agent-conversations", agentId, page],
    queryFn: async () => {
      const { data, count, error } = await supabase
        .from("ai_agent_conversations")
        .select("*, marketing_contacts(first_name, last_name, email)", { count: "exact" })
        .eq("agent_id", agentId)
        .order("started_at", { ascending: false })
        .range(page * pageSize, (page + 1) * pageSize - 1);
      if (error) throw error;
      return { conversations: (data ?? []) as AiAgentConversation[], total: count ?? 0 };
    },
    enabled: !!agentId,
    staleTime: 60 * 1000,
    placeholderData: (prev) => prev,
  });
}
```
