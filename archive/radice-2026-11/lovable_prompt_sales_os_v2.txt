-- 7. TRIGGER — logga ogni cambio valore opportunità
-- ============================================================
CREATE OR REPLACE FUNCTION trigger_log_deal_value_change()
RETURNS TRIGGER AS $$
DECLARE
  v_company_id UUID;
BEGIN
  -- Solo se il valore è cambiato e non è la prima insert
  IF TG_OP = 'UPDATE' AND OLD.value IS NOT NULL AND NEW.value IS NOT NULL
     AND OLD.value <> NEW.value THEN

    -- Recupera company_id
    v_company_id := NEW.company_id;

    INSERT INTO deal_value_history (
      company_id, opportunity_id, old_value, new_value, changed_by
    ) VALUES (
      v_company_id, NEW.id, OLD.value, NEW.value, auth.uid()
    );

    -- Se initial_value non era impostato, impostalo al valore originale
    IF NEW.initial_value IS NULL THEN
      NEW.initial_value := OLD.value;
    END IF;
  END IF;

  -- Al primo insert, imposta initial_value = value
  IF TG_OP = 'INSERT' AND NEW.value IS NOT NULL AND NEW.initial_value IS NULL THEN
    NEW.initial_value := NEW.value;
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS log_deal_value_change ON marketing_opportunities;
CREATE TRIGGER log_deal_value_change
  BEFORE INSERT OR UPDATE OF value ON marketing_opportunities
  FOR EACH ROW
  EXECUTE FUNCTION trigger_log_deal_value_change();

-- ============================================================
-- 8. TRIGGER — aggiorna stage_entered_at a ogni cambio stage
-- ============================================================
CREATE OR REPLACE FUNCTION trigger_track_stage_entry()
RETURNS TRIGGER AS $$
BEGIN
  IF TG_OP = 'UPDATE' AND OLD.stage_id IS DISTINCT FROM NEW.stage_id THEN
    NEW.stage_entered_at := NOW();
    -- Se entry_stage_id è NULL (prima volta che tracciamo), impostalo
    IF NEW.entry_stage_id IS NULL THEN
      NEW.entry_stage_id := OLD.stage_id;
    END IF;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS track_stage_entry ON marketing_opportunities;
CREATE TRIGGER track_stage_entry
  BEFORE UPDATE OF stage_id ON marketing_opportunities
  FOR EACH ROW
  EXECUTE FUNCTION trigger_track_stage_entry();

-- 9. FUNZIONE — calcola deal health score
-- ============================================================
CREATE OR REPLACE FUNCTION calculate_deal_health_score(
  p_bant_budget TEXT,
  p_bant_authority TEXT,
  p_bant_need TEXT,
  p_bant_timeline TEXT,
  p_has_next_action BOOLEAN,
  p_has_expected_close_date BOOLEAN,
  p_days_in_stage INTEGER,
  p_stalled_threshold INTEGER
) RETURNS INTEGER AS $$
DECLARE
  score INTEGER := 0;
BEGIN
  -- BANT: Budget (25 punti)
  score := score + CASE p_bant_budget
    WHEN 'confermato'  THEN 25
    WHEN 'stimato'     THEN 15
    WHEN 'sconosciuto' THEN 5
    WHEN 'no_budget'   THEN 0
    ELSE 0
  END;

  -- BANT: Authority (20 punti)
  score := score + CASE p_bant_authority
    WHEN 'decisore'    THEN 20
    WHEN 'influencer'  THEN 10
    WHEN 'da_scoprire' THEN 3
    ELSE 0
  END;

  -- BANT: Need (25 punti)
  score := score + CASE p_bant_need
    WHEN 'urgente'      THEN 25
    WHEN 'pianificato'  THEN 15
    WHEN 'esplorativo'  THEN 5
    WHEN 'no_need'      THEN 0
    ELSE 0
  END;

  -- BANT: Timeline (15 punti)
  score := score + CASE p_bant_timeline
    WHEN 'entro_30gg'  THEN 15
    WHEN 'entro_90gg'  THEN 10
    WHEN 'oltre_90gg'  THEN 5
    WHEN 'indefinito'  THEN 0
    ELSE 0
  END;

  -- Bonus operativi (15 punti)
  IF p_has_next_action        THEN score := score + 8;  END IF;
  IF p_has_expected_close_date THEN score := score + 7;  END IF;

  -- Penalità stallo (fino a -20 punti)
  IF p_days_in_stage > p_stalled_threshold * 2 THEN
    score := score - 20;
  ELSIF p_days_in_stage > p_stalled_threshold THEN
    score := score - 10;
  END IF;

  RETURN GREATEST(0, LEAST(100, score));
END;
$$ LANGUAGE plpgsql IMMUTABLE;

-- 10. RPC — Win/Loss deep analytics
-- ============================================================
CREATE OR REPLACE FUNCTION get_win_loss_analysis(
  p_company_id UUID,
  p_date_from  DATE DEFAULT NULL,
  p_date_to    DATE DEFAULT NULL
)
RETURNS JSONB AS $$
DECLARE
  v_result JSONB;
BEGIN
  WITH closed_opps AS (
    SELECT
      o.id,
      o.status,
      o.value,
      o.initial_value,
      o.lost_reason_category,
      o.competitor_won,
      o.source,
      mc.city,
      mc.province,
      o.assigned_to,
      o.stage_id,
      ps.name AS stage_name,
      o.created_at,
      o.updated_at,
      EXTRACT(DAY FROM o.updated_at - o.created_at)::INTEGER AS cycle_days,
      CASE
        WHEN o.initial_value > 0 AND o.value < o.initial_value
        THEN ROUND(((o.initial_value - o.value) / o.initial_value) * 100, 1)
        ELSE 0
      END AS discount_pct
    FROM marketing_opportunities o
    LEFT JOIN marketing_contacts mc ON mc.id = o.contact_id
    LEFT JOIN marketing_pipeline_stages ps ON ps.id = o.stage_id
    WHERE o.company_id = p_company_id
      AND o.status IN ('won', 'lost', 'abandoned')
      AND (p_date_from IS NULL OR o.updated_at >= p_date_from)
      AND (p_date_to   IS NULL OR o.updated_at <= p_date_to + INTERVAL '1 day')
  ),
  summary AS (
    SELECT
      COUNT(*) FILTER (WHERE status = 'won')  AS won_count,
      COUNT(*) FILTER (WHERE status IN ('lost','abandoned')) AS lost_count,
      COUNT(*) AS total_closed,
      ROUND(AVG(cycle_days) FILTER (WHERE status = 'won'), 1) AS avg_win_cycle_days,
      ROUND(AVG(cycle_days) FILTER (WHERE status IN ('lost','abandoned')), 1) AS avg_loss_cycle_days,
      COALESCE(SUM(value) FILTER (WHERE status = 'won'), 0) AS total_won_value,
      ROUND(AVG(discount_pct) FILTER (WHERE status = 'won' AND discount_pct > 0), 1) AS avg_discount_pct,
      ROUND(AVG(value) FILTER (WHERE status = 'won'), 0) AS avg_deal_size_won,
      ROUND(AVG(value) FILTER (WHERE status IN ('lost','abandoned')), 0) AS avg_deal_size_lost
    FROM closed_opps
  ),
  by_lost_reason AS (
    SELECT
      COALESCE(lost_reason_category, 'non_specificato') AS reason,
      COUNT(*) AS count,
      ROUND(COUNT(*) * 100.0 / NULLIF((SELECT COUNT(*) FROM closed_opps WHERE status IN ('lost','abandoned')), 0), 1) AS pct
    FROM closed_opps
    WHERE status IN ('lost', 'abandoned')
    GROUP BY COALESCE(lost_reason_category, 'non_specificato')
    ORDER BY count DESC
  ),
  by_competitor AS (
    SELECT
      COALESCE(competitor_won, 'nessuno / sconosciuto') AS competitor,
      COUNT(*) AS count,
      ROUND(AVG(value), 0) AS avg_deal_size
    FROM closed_opps
    WHERE status IN ('lost', 'abandoned')
    GROUP BY COALESCE(competitor_won, 'nessuno / sconosciuto')
    ORDER BY count DESC
    LIMIT 10
  ),
  by_source AS (
    SELECT
      COALESCE(source, 'sconosciuto') AS source,
      COUNT(*) FILTER (WHERE status = 'won')                 AS won,
      COUNT(*) FILTER (WHERE status IN ('lost','abandoned'))  AS lost,
      ROUND(
        COUNT(*) FILTER (WHERE status = 'won') * 100.0 /
        NULLIF(COUNT(*), 0), 1
      ) AS win_rate,
      ROUND(AVG(value) FILTER (WHERE status = 'won'), 0) AS avg_won_value
    FROM closed_opps
    GROUP BY COALESCE(source, 'sconosciuto')
    ORDER BY won DESC
  ),
  by_deal_size AS (
    SELECT
      CASE
        WHEN value < 5000       THEN '< €5k'
        WHEN value < 15000      THEN '€5k–15k'
        WHEN value < 30000      THEN '€15k–30k'
        WHEN value < 60000      THEN '€30k–60k'
        ELSE '> €60k'
      END AS size_bucket,
      COUNT(*) FILTER (WHERE status = 'won')                AS won,
      COUNT(*) FILTER (WHERE status IN ('lost','abandoned')) AS lost,
      ROUND(
        COUNT(*) FILTER (WHERE status = 'won') * 100.0 /
        NULLIF(COUNT(*), 0), 1
      ) AS win_rate
    FROM closed_opps
    GROUP BY size_bucket
    ORDER BY MIN(value)
  ),
  by_stage_lost AS (
    SELECT
      stage_name,
      COUNT(*) AS lost_count,
      ROUND(COUNT(*) * 100.0 / NULLIF((SELECT COUNT(*) FROM closed_opps WHERE status IN ('lost','abandoned')), 0), 1) AS pct
    FROM closed_opps
    WHERE status IN ('lost', 'abandoned') AND stage_name IS NOT NULL
    GROUP BY stage_name
    ORDER BY lost_count DESC
  ),
  by_salesperson AS (
    SELECT
      o.assigned_to,
      COALESCE(p.full_name, 'Non assegnato') AS name,
      COUNT(*) FILTER (WHERE o.status = 'won')                AS won,
      COUNT(*) FILTER (WHERE o.status IN ('lost','abandoned')) AS lost,
      ROUND(
        COUNT(*) FILTER (WHERE o.status = 'won') * 100.0 /
        NULLIF(COUNT(*), 0), 1
      ) AS win_rate,
      ROUND(AVG(o.discount_pct) FILTER (WHERE o.status = 'won' AND o.discount_pct > 0), 1) AS avg_discount_pct
    FROM closed_opps o
    LEFT JOIN profiles p ON p.id = o.assigned_to
    GROUP BY o.assigned_to, p.full_name
    ORDER BY won DESC
  )
  SELECT jsonb_build_object(
    'summary',         (SELECT row_to_json(summary) FROM summary),
    'by_lost_reason',  (SELECT jsonb_agg(row_to_json(r)) FROM by_lost_reason r),
    'by_competitor',   (SELECT jsonb_agg(row_to_json(r)) FROM by_competitor r),
    'by_source',       (SELECT jsonb_agg(row_to_json(r)) FROM by_source r),
    'by_deal_size',    (SELECT jsonb_agg(row_to_json(r)) FROM by_deal_size r),
    'by_stage_lost',   (SELECT jsonb_agg(row_to_json(r)) FROM by_stage_lost r),
    'by_salesperson',  (SELECT jsonb_agg(row_to_json(r)) FROM by_salesperson r)
  ) INTO v_result;

  RETURN v_result;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 11. RPC — activity timeline per opportunità
-- ============================================================
CREATE OR REPLACE FUNCTION get_opportunity_activity_timeline(
  p_company_id    UUID,
  p_opportunity_id UUID
)
RETURNS TABLE (
  event_id    UUID,
  event_type  TEXT,  -- 'activity', 'note', 'stage_change', 'value_change',
                     --  'objection', 'quote_sent', 'status_change'
  event_title TEXT,
  event_body  TEXT,
  actor_name  TEXT,
  created_at  TIMESTAMPTZ
) AS $$
DECLARE
  v_contact_id UUID;
BEGIN
  SELECT contact_id INTO v_contact_id
  FROM marketing_opportunities
  WHERE id = p_opportunity_id AND company_id = p_company_id;

  RETURN QUERY

  -- Note sull'opportunità
  SELECT
    mon.id,
    'note'::TEXT,
    'Nota aggiunta'::TEXT,
    LEFT(mon.content, 200),
    COALESCE(p.full_name, 'Sistema'),
    mon.created_at
  FROM marketing_opportunity_notes mon
  LEFT JOIN profiles p ON p.id = mon.created_by
  WHERE mon.opportunity_id = p_opportunity_id

  UNION ALL

  -- Attività sul contatto
  SELECT
    mca.id,
    'activity'::TEXT,
    COALESCE(mca.activity_type, 'Attività'),
    LEFT(COALESCE(mca.description, mca.notes, ''), 200),
    COALESCE(p.full_name, 'Sistema'),
    mca.created_at
  FROM marketing_contact_activities mca
  LEFT JOIN profiles p ON p.id = mca.created_by
  WHERE mca.contact_id = v_contact_id
    AND mca.company_id = p_company_id
    AND (mca.opportunity_id = p_opportunity_id OR mca.opportunity_id IS NULL)

  UNION ALL

  -- Storico variazioni valore
  SELECT
    dvh.id,
    'value_change'::TEXT,
    CASE
      WHEN dvh.change_amount < 0
      THEN 'Valore ridotto: ' || TO_CHAR(ABS(dvh.change_amount), 'FM€999,999') || ' (' || dvh.change_pct::TEXT || '%)'
      ELSE 'Valore aumentato: +' || TO_CHAR(dvh.change_amount, 'FM€999,999')
    END,
    'Da ' || TO_CHAR(dvh.old_value, 'FM€999,999') || ' → ' || TO_CHAR(dvh.new_value, 'FM€999,999'),
    COALESCE(p.full_name, 'Sistema'),
    dvh.created_at
  FROM deal_value_history dvh
  LEFT JOIN profiles p ON p.id = dvh.changed_by
  WHERE dvh.opportunity_id = p_opportunity_id

  UNION ALL

  -- Obiezioni
  SELECT
    oo.id,
    'objection'::TEXT,
    'Obiezione: ' || oo.category,
    LEFT(oo.objection_text, 200)
      || CASE WHEN oo.response_given IS NOT NULL
              THEN ' → Risposta: ' || LEFT(oo.response_given, 100)
              ELSE '' END,
    COALESCE(p.full_name, 'Team'),
    oo.raised_at
  FROM opportunity_objections oo
  LEFT JOIN profiles p ON p.id = oo.created_by
  WHERE oo.opportunity_id = p_opportunity_id

  ORDER BY created_at DESC
  LIMIT 100;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================================
-- 12. RPC — aggiorna deal health score bulk
-- ============================================================
CREATE OR REPLACE FUNCTION refresh_deal_health_scores(p_company_id UUID)
RETURNS INTEGER AS $$
DECLARE
  updated_count INTEGER := 0;
BEGIN
  UPDATE marketing_opportunities o
  SET deal_health_score = calculate_deal_health_score(
    o.bant_budget,
    o.bant_authority,
    o.bant_need,
    o.bant_timeline,
    o.next_action IS NOT NULL,
    o.expected_close_date IS NOT NULL,
    EXTRACT(DAY FROM NOW() - o.stage_entered_at)::INTEGER,
    COALESCE(ps.stalled_threshold_days, 14)
  )
  FROM marketing_pipeline_stages ps
  WHERE o.stage_id = ps.id
    AND o.company_id = p_company_id
    AND o.status = 'open';

  GET DIAGNOSTICS updated_count = ROW_COUNT;
  RETURN updated_count;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

export function useUpdateBANTMutation() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async ({
      opportunityId,
      bant,
    }: {
      opportunityId: string;
      bant: {
        bant_budget?: BantBudget;
        bant_authority?: BantAuthority;
        bant_need?: BantNeed;
        bant_timeline?: BantTimeline;
      };
    }) => {
      // Calcola il nuovo health score inline
      const { data: opp } = await supabase
        .from('marketing_opportunities')
        .select('bant_budget, bant_authority, bant_need, bant_timeline, next_action, expected_close_date, stage_entered_at, stage_id')
        .eq('id', opportunityId)
        .single();

      const merged = { ...opp, ...bant };

      const { data: scoreData } = await supabase.rpc('calculate_deal_health_score', {
        p_bant_budget: merged.bant_budget ?? null,
        p_bant_authority: merged.bant_authority ?? null,
        p_bant_need: merged.bant_need ?? null,
        p_bant_timeline: merged.bant_timeline ?? null,
        p_has_next_action: !!merged.next_action,
        p_has_expected_close_date: !!merged.expected_close_date,
        p_days_in_stage: merged.stage_entered_at
          ? Math.floor((Date.now() - new Date(merged.stage_entered_at).getTime()) / 86400000)
          : 0,
        p_stalled_threshold: 14,
      });

      const { error } = await supabase
        .from('marketing_opportunities')
        .update({
          ...bant,
          deal_health_score: scoreData ?? null,
          updated_at: new Date().toISOString(),
        })
        .eq('id', opportunityId);
      if (error) throw error;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['sales-os-v2'] });
      queryClient.invalidateQueries({ queryKey: ['marketing', 'opportunities'] });
    },
  });
}

                      <td className={`py-1.5 text-right ${
                        (s.avg_discount_pct ?? 0) > 20 ? 'text-red-600 font-bold' :
                        (s.avg_discount_pct ?? 0) > 10 ? 'text-yellow-600' : 'text-muted-foreground'
                      }`}>
                        {s.avg_discount_pct ? pct(s.avg_discount_pct) : '—'}
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            ) : (
              <p className="text-xs text-muted-foreground">Nessun dato.</p>
            )}
          </CardContent>
        </Card>
      </div>

    </div>
  );
}

function WinLossSummaryCompact({ companyId }: { companyId: string }) {
  const { data: analysis, isLoading } = useWinLossAnalysis(companyId);

  if (isLoading) return (
    <div className="text-sm text-muted-foreground animate-pulse p-4">Analisi...</div>
  );
  if (!analysis) return (
    <div className="text-sm text-muted-foreground p-4">
      Chiudi alcune opportunità per vedere l'analisi.
    </div>
  );

  const winRate = analysis.summary.total_closed > 0
    ? (analysis.summary.won_count / analysis.summary.total_closed) * 100
    : 0;

  const topLostReason = analysis.by_lost_reason?.[0];
  const topCompetitor = analysis.by_competitor?.find(c => c.competitor !== 'nessuno / sconosciuto');

  return (
    <div className="space-y-3">
      {/* Mini summary */}
      <div className="grid grid-cols-3 gap-3">
        <Card className="p-3">
          <div className="text-xs text-muted-foreground">Win Rate</div>
          <div className={`text-2xl font-bold ${winRate >= 30 ? 'text-green-600' : winRate >= 15 ? 'text-yellow-600' : 'text-red-600'}`}>
            {winRate.toFixed(1)}%
          </div>
          <div className="text-xs text-muted-foreground mt-0.5">
            {analysis.summary.won_count}V / {analysis.summary.lost_count}P
          </div>
        </Card>
        <Card className="p-3">
          <div className="text-xs text-muted-foreground">Motivo perdita top</div>
          <div className="text-sm font-semibold mt-1 capitalize">
            {topLostReason ? topLostReason.reason.replace(/_/g, ' ') : '—'}
          </div>
          <div className="text-xs text-muted-foreground">
            {topLostReason ? `${topLostReason.pct}% dei persi` : ''}
          </div>
        </Card>
        <Card className="p-3">
          <div className="text-xs text-muted-foreground">Sconto medio</div>
          <div className={`text-2xl font-bold ${(analysis.summary.avg_discount_pct ?? 0) > 15 ? 'text-red-600' : 'text-muted-foreground'}`}>
            {analysis.summary.avg_discount_pct ? `${analysis.summary.avg_discount_pct}%` : '—'}
          </div>
          <div className="text-xs text-muted-foreground">su deal vinti</div>
        </Card>
      </div>
      {/* Top 3 motivi perdita */}

                  style={{ width: `${r.pct}%` }}
                />
              </div>
            </div>

    </div>
  );
