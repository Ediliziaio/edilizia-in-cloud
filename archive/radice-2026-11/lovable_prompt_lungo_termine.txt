```tsx
// InlineField — campo testo/select/email/tel editabile inline
// Props:
//   label: string
//   value: string | null | undefined
//   field: string                         // nome campo per onSave
//   type?: "text" | "email" | "tel" | "select" | "textarea"
//   options?: { value: string; label: string }[]  // solo se type="select"
//   onSave: (field: string, value: string) => Promise<void>
//   icon?: React.ReactNode

// Stato interno: isEditing (boolean), localValue (string)
// Quando isEditing=false: mostra valore + icona matita on-hover
// Quando isEditing=true: mostra Input/Select/Textarea + tasto Salva + tasto X
// onSave: chiama prop, poi isEditing=false; su errore rimane aperto + toast error
// Se value è null/undefined mostra placeholder "—" in grigio

import { useState } from "react";
import { Input } from "@/components/ui/input";
import { Textarea } from "@/components/ui/textarea";
import { Button } from "@/components/ui/button";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Pencil, Check, X } from "lucide-react";
import { toast } from "sonner";

interface InlineFieldProps {
  label: string;
  value?: string | null;
  field: string;
  type?: "text" | "email" | "tel" | "select" | "textarea";
  options?: { value: string; label: string }[];
  onSave: (field: string, value: string) => Promise<void>;
  icon?: React.ReactNode;
}

export function InlineField({ label, value, field, type = "text", options, onSave, icon }: InlineFieldProps) {
  const [isEditing, setIsEditing] = useState(false);
  const [localValue, setLocalValue] = useState(value ?? "");
  const [saving, setSaving] = useState(false);

  const handleSave = async () => {
    setSaving(true);
    try {
      await onSave(field, localValue);
      setIsEditing(false);
    } catch {
      toast.error("Errore nel salvataggio");
    } finally {
      setSaving(false);
    }
  };

  const handleCancel = () => {
    setLocalValue(value ?? "");
    setIsEditing(false);
  };

  if (!isEditing) {
    return (
      <div className="group flex items-start gap-2 py-1">
        {icon && <span className="mt-0.5 text-muted-foreground">{icon}</span>}
        <div className="flex-1 min-w-0">
          <p className="text-xs text-muted-foreground">{label}</p>
          <p className={`text-sm truncate ${!value ? "text-muted-foreground italic" : ""}`}>
            {value || "—"}
          </p>
        </div>
        <Button
          variant="ghost"
          size="icon"
          className="h-6 w-6 opacity-0 group-hover:opacity-100 transition-opacity"
          onClick={() => setIsEditing(true)}
        >
          <Pencil className="h-3 w-3" />
        </Button>
      </div>
    );
  }

  return (
    <div className="flex flex-col gap-1 py-1">
      <p className="text-xs text-muted-foreground">{label}</p>
      {type === "select" && options ? (
        <Select value={localValue} onValueChange={setLocalValue}>
          <SelectTrigger className="h-8 text-sm">
            <SelectValue />
          </SelectTrigger>
          <SelectContent>
            {options.map(o => (
              <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>
            ))}
          </SelectContent>
        </Select>
      ) : type === "textarea" ? (
        <Textarea
          value={localValue}
          onChange={e => setLocalValue(e.target.value)}
          className="text-sm min-h-[60px]"
          autoFocus
        />
      ) : (
        <Input
          type={type}
          value={localValue}
          onChange={e => setLocalValue(e.target.value)}
          className="h-8 text-sm"
          autoFocus
          onKeyDown={e => {
            if (e.key === "Enter") handleSave();
            if (e.key === "Escape") handleCancel();
          }}
        />
      )}
      <div className="flex gap-1">
        <Button size="sm" className="h-7 text-xs" onClick={handleSave} disabled={saving}>
          <Check className="h-3 w-3 mr-1" /> Salva
        </Button>
        <Button size="sm" variant="ghost" className="h-7 text-xs" onClick={handleCancel}>
          <X className="h-3 w-3" />
        </Button>
      </div>
    </div>
  );
}
```

PROMPT [RPC1] — RPC AGGREGATE PER DASHBOARD
                Da 16+ query parallele a 1-2 RPC PostgreSQL
════════════════════════════════════════════════════════════════════════════

Il file src/hooks/useCompanyDashboardData.ts esegue ~16 query Supabase
in parallelo via Promise.all. Ogni query è un round-trip HTTP separato.
Questo causa:
  - 16 richieste HTTP invece di 1
  - Dati potenzialmente inconsistenti (ogni query legge in momenti diversi)
  - Latenza percepita alta anche su connessioni veloci

La soluzione è una singola RPC PostgreSQL che ritorna tutti i dati in
un solo JSON strutturato.

────────────────────────────────────────────────────────────────────────────
STEP 1 — Crea la RPC get_dashboard_data() in Supabase
────────────────────────────────────────────────────────────────────────────

Nel SQL editor di Supabase, esegui questa migration:

```sql
-- Migration: create_get_dashboard_data_rpc
-- Aggregates all dashboard data into a single RPC call

CREATE OR REPLACE FUNCTION get_dashboard_data(p_company_id uuid)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_result jsonb;
  v_ordini_attivi int;
  v_ordini_valore numeric;
  v_fatturato_mese numeric;
  v_costi_mese numeric;
  v_ticket_aperti int;
  v_task_in_scadenza int;
  v_dipendenti_attivi int;
  v_rate_in_scadenza int;
  v_contatti_marketing int;
  v_opportunita_aperte int;
  v_valore_pipeline numeric;
BEGIN
  -- Verifica che l'utente abbia accesso a questa company
  IF NOT EXISTS (
    SELECT 1 FROM company_members
    WHERE company_id = p_company_id
    AND user_id = auth.uid()
  ) THEN
    RAISE EXCEPTION 'Access denied';
  END IF;

  -- Ordini attivi (non completati, non annullati)
  SELECT COUNT(*), COALESCE(SUM(importo_totale), 0)
  INTO v_ordini_attivi, v_ordini_valore
  FROM orders
  WHERE company_id = p_company_id
    AND stato NOT IN ('completato', 'annullato');

  -- Fatturato mese corrente (ordini completati questo mese)
  SELECT COALESCE(SUM(importo_totale), 0)
  INTO v_fatturato_mese
  FROM orders
  WHERE company_id = p_company_id
    AND stato = 'completato'
    AND date_trunc('month', updated_at) = date_trunc('month', now());

  -- Costi mese corrente
  SELECT COALESCE(SUM(importo), 0)
  INTO v_costi_mese
  FROM costs
  WHERE company_id = p_company_id
    AND date_trunc('month', data_costo) = date_trunc('month', now());

  -- Ticket aperti
  SELECT COUNT(*)
  INTO v_ticket_aperti
  FROM tickets
  WHERE company_id = p_company_id
    AND stato NOT IN ('chiuso', 'risolto');

  -- Task in scadenza (prossimi 7 giorni)
  SELECT COUNT(*)
  INTO v_task_in_scadenza
  FROM tasks
  WHERE company_id = p_company_id
    AND completed = false
    AND due_date IS NOT NULL
    AND due_date <= now() + interval '7 days'
    AND due_date >= now();

  -- Dipendenti attivi
  SELECT COUNT(*)
  INTO v_dipendenti_attivi
  FROM employees
  WHERE company_id = p_company_id
    AND attivo = true;

  -- Rate in scadenza (prossimi 30 giorni, non pagate)
  SELECT COUNT(*)
  INTO v_rate_in_scadenza
  FROM order_installments oi
  JOIN orders o ON o.id = oi.order_id
  WHERE o.company_id = p_company_id
    AND oi.pagata = false
    AND oi.data_scadenza IS NOT NULL
    AND oi.data_scadenza <= now() + interval '30 days'
    AND oi.data_scadenza >= now();

  -- Contatti marketing
  SELECT COUNT(*)
  INTO v_contatti_marketing
  FROM marketing_contacts
  WHERE company_id = p_company_id;

  -- Opportunità aperte + valore pipeline
  SELECT COUNT(*), COALESCE(SUM(valore_stimato), 0)
  INTO v_opportunita_aperte, v_valore_pipeline
  FROM opportunities
  WHERE company_id = p_company_id
    AND stato NOT IN ('vinta', 'persa', 'annullata');

  -- Costruisci JSON risultato
  v_result := jsonb_build_object(
```

```sql
CREATE OR REPLACE FUNCTION get_cashflow_summary(
  p_company_id uuid,
  p_months_ahead int DEFAULT 3
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_result jsonb;
BEGIN
  -- Verifica accesso
  IF NOT EXISTS (
    SELECT 1 FROM company_members
    WHERE company_id = p_company_id AND user_id = auth.uid()
  ) THEN
    RAISE EXCEPTION 'Access denied';
  END IF;

  SELECT jsonb_build_object(

    -- Saldo attuale (somma conti correnti)
    'saldo_attuale', (
      SELECT COALESCE(SUM(saldo), 0)
      FROM conti_bancari
      WHERE company_id = p_company_id AND attivo = true
    ),

    -- Entrate previste nei prossimi N mesi (rate non pagate su ordini attivi)
    'entrate_previste', (
      SELECT COALESCE(SUM(oi.importo), 0)
      FROM order_installments oi
      JOIN orders o ON o.id = oi.order_id
      WHERE o.company_id = p_company_id
        AND oi.pagata = false
        AND oi.data_scadenza >= now()
        AND oi.data_scadenza <= now() + (p_months_ahead || ' months')::interval
    ),

    -- Uscite previste nei prossimi N mesi
    'uscite_previste', (
      SELECT COALESCE(SUM(importo), 0)
      FROM costs
      WHERE company_id = p_company_id
        AND pagato = false
        AND data_scadenza >= now()
        AND data_scadenza <= now() + (p_months_ahead || ' months')::interval
    ),

    -- Incassato mese corrente
    'incassato_mese', (
      SELECT COALESCE(SUM(oi.importo), 0)
      FROM order_installments oi
      JOIN orders o ON o.id = oi.order_id
      WHERE o.company_id = p_company_id
        AND oi.pagata = true
        AND date_trunc('month', oi.data_pagamento) = date_trunc('month', now())
    ),

    -- Speso mese corrente
    'speso_mese', (
      SELECT COALESCE(SUM(importo), 0)
      FROM costs
      WHERE company_id = p_company_id
        AND pagato = true
        AND date_trunc('month', data_pagamento) = date_trunc('month', now())
    ),

    -- Forecast mensile (array di oggetti {mese, entrate, uscite})
    'forecast_mensile', (
      SELECT jsonb_agg(m ORDER BY m->>'mese')
      FROM (
        SELECT jsonb_build_object(
          'mese', to_char(gs.mese, 'YYYY-MM'),
          'entrate', COALESCE((
            SELECT SUM(oi.importo)
            FROM order_installments oi
            JOIN orders o ON o.id = oi.order_id
            WHERE o.company_id = p_company_id
              AND oi.pagata = false
              AND date_trunc('month', oi.data_scadenza) = gs.mese
          ), 0),
          'uscite', COALESCE((
            SELECT SUM(importo)
            FROM costs
            WHERE company_id = p_company_id
              AND pagato = false
              AND date_trunc('month', data_scadenza) = gs.mese
          ), 0)
        ) AS m
        FROM generate_series(
          date_trunc('month', now()),
          date_trunc('month', now()) + ((p_months_ahead - 1) || ' months')::interval,
          '1 month'
        ) AS gs(mese)
      ) subq
    ),

    'generated_at', now()

  ) INTO v_result;

  RETURN v_result;
END;
$$;

REVOKE ALL ON FUNCTION get_cashflow_summary(uuid, int) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION get_cashflow_summary(uuid, int) TO authenticated;
```

```tsx
// Generazione .ics:
function generateICS(appointment: { titolo: string; data: string; ora_inizio: string; ora_fine: string; luogo?: string; note?: string }): string {
  const startDt = appointment.data.replace(/-/g, "") + "T" + (appointment.ora_inizio ?? "090000").replace(/:/g, "");
  const endDt   = appointment.data.replace(/-/g, "") + "T" + (appointment.ora_fine   ?? "100000").replace(/:/g, "");
  return [
    "BEGIN:VCALENDAR",
    "VERSION:2.0",
    "PRODID:-//Edilizia in Cloud//IT",
    "BEGIN:VEVENT",
    `DTSTART:${startDt}`,
    `DTEND:${endDt}`,
    `SUMMARY:${appointment.titolo}`,
    appointment.luogo ? `LOCATION:${appointment.luogo}` : "",
    appointment.note  ? `DESCRIPTION:${appointment.note}` : "",
    `UID:${Date.now()}@ediliziacloud`,
    "END:VEVENT",
    "END:VCALENDAR",
  ].filter(Boolean).join("\r\n");
}

const handleAddToCalendar = (appointment: AppointmentRow) => {
  const ics = generateICS(appointment);
  const blob = new Blob([ics], { type: "text/calendar" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = `appuntamento-${appointment.data}.ics`;
  a.click();
  URL.revokeObjectURL(url);
};

// staleTime: 10 minuti (appuntamenti non cambiano spesso)
```
