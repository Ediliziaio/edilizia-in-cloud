```sql
-- ============================================================
-- INTERNAL AUTOMATION SYSTEM - Mirror of Marketing Automations
-- ============================================================

-- Main flow table (mirrors automation_flows but for Gestione Interna)
CREATE TABLE IF NOT EXISTS internal_automation_flows (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  description TEXT,
  is_active BOOLEAN DEFAULT true,
  trigger_type TEXT NOT NULL,
  trigger_config JSONB DEFAULT '{}',
  created_by UUID REFERENCES profiles(id),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  last_triggered_at TIMESTAMPTZ,
  total_runs INTEGER DEFAULT 0,
  successful_runs INTEGER DEFAULT 0,
  failed_runs INTEGER DEFAULT 0
);

-- Nodes in the visual builder (mirrors automation_nodes)
CREATE TABLE IF NOT EXISTS internal_automation_nodes (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  flow_id UUID NOT NULL REFERENCES internal_automation_flows(id) ON DELETE CASCADE,
  node_type TEXT NOT NULL, -- 'trigger' | 'action' | 'condition' | 'delay' | 'branch'
  node_subtype TEXT NOT NULL, -- es. 'order_status_change', 'send_email', 'if_condition'
  config JSONB DEFAULT '{}',
  position_x FLOAT DEFAULT 0,
  position_y FLOAT DEFAULT 0,
  label TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Connections between nodes (mirrors automation_connections)
CREATE TABLE IF NOT EXISTS internal_automation_connections (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  flow_id UUID NOT NULL REFERENCES internal_automation_flows(id) ON DELETE CASCADE,
  source_node_id UUID NOT NULL REFERENCES internal_automation_nodes(id) ON DELETE CASCADE,
  target_node_id UUID NOT NULL REFERENCES internal_automation_nodes(id) ON DELETE CASCADE,
  condition_branch TEXT, -- 'yes' | 'no' | null
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Enrollments: tracks which entity is running through which flow
CREATE TABLE IF NOT EXISTS internal_automation_enrollments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  flow_id UUID NOT NULL REFERENCES internal_automation_flows(id) ON DELETE CASCADE,
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  entity_type TEXT NOT NULL, -- 'order' | 'employee' | 'ticket' | 'task' | 'cost' | 'warehouse_item'
  entity_id UUID NOT NULL,
  current_node_id UUID REFERENCES internal_automation_nodes(id),
  status TEXT DEFAULT 'active', -- 'active' | 'completed' | 'failed' | 'paused'
  started_at TIMESTAMPTZ DEFAULT NOW(),
  completed_at TIMESTAMPTZ,
  error_message TEXT,
  context_data JSONB DEFAULT '{}' -- snapshot dei dati al momento del trigger
);

-- Async execution queue (mirrors automation_queue)
CREATE TABLE IF NOT EXISTS internal_automation_queue (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  enrollment_id UUID NOT NULL REFERENCES internal_automation_enrollments(id) ON DELETE CASCADE,
  flow_id UUID NOT NULL REFERENCES internal_automation_flows(id) ON DELETE CASCADE,
  node_id UUID NOT NULL REFERENCES internal_automation_nodes(id),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  scheduled_at TIMESTAMPTZ DEFAULT NOW(),
  processed_at TIMESTAMPTZ,
  status TEXT DEFAULT 'pending', -- 'pending' | 'processing' | 'done' | 'failed'
  retry_count INTEGER DEFAULT 0,
  error_message TEXT,
  payload JSONB DEFAULT '{}'
);

-- Execution log (mirrors automation_execution_log)
CREATE TABLE IF NOT EXISTS internal_automation_execution_log (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  enrollment_id UUID REFERENCES internal_automation_enrollments(id),
  flow_id UUID NOT NULL REFERENCES internal_automation_flows(id) ON DELETE CASCADE,
  node_id UUID REFERENCES internal_automation_nodes(id),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  entity_type TEXT,
  entity_id UUID,
  action TEXT NOT NULL,
  status TEXT DEFAULT 'success', -- 'success' | 'error' | 'skipped'
  details JSONB DEFAULT '{}',
  executed_at TIMESTAMPTZ DEFAULT NOW(),
  duration_ms INTEGER
);

-- RLS Policies
ALTER TABLE internal_automation_flows ENABLE ROW LEVEL SECURITY;
ALTER TABLE internal_automation_nodes ENABLE ROW LEVEL SECURITY;
ALTER TABLE internal_automation_connections ENABLE ROW LEVEL SECURITY;
ALTER TABLE internal_automation_enrollments ENABLE ROW LEVEL SECURITY;
ALTER TABLE internal_automation_queue ENABLE ROW LEVEL SECURITY;
ALTER TABLE internal_automation_execution_log ENABLE ROW LEVEL SECURITY;

-- Flows: company members can CRUD their own
CREATE POLICY "company_internal_flows" ON internal_automation_flows
  FOR ALL USING (company_id = get_user_company_id(auth.uid()));

-- Nodes: accessible via flow ownership
CREATE POLICY "company_internal_nodes" ON internal_automation_nodes
  FOR ALL USING (
    flow_id IN (
      SELECT id FROM internal_automation_flows
      WHERE company_id = get_user_company_id(auth.uid())
    )
  );

-- Connections: accessible via flow ownership
CREATE POLICY "company_internal_connections" ON internal_automation_connections
  FOR ALL USING (
    flow_id IN (
      SELECT id FROM internal_automation_flows
      WHERE company_id = get_user_company_id(auth.uid())
    )
  );

-- Enrollments: company members can read, service_role writes
CREATE POLICY "company_internal_enrollments_read" ON internal_automation_enrollments
  FOR SELECT USING (company_id = get_user_company_id(auth.uid()));

-- Queue: service_role only (no user policy needed, edge function uses service_role)
CREATE POLICY "company_internal_queue_read" ON internal_automation_queue
  FOR SELECT USING (company_id = get_user_company_id(auth.uid()));

-- Execution log: company members can read
CREATE POLICY "company_internal_execution_log_read" ON internal_automation_execution_log
  FOR SELECT USING (company_id = get_user_company_id(auth.uid()));

-- Indexes for performance
CREATE INDEX idx_internal_flows_company ON internal_automation_flows(company_id);
CREATE INDEX idx_internal_flows_active ON internal_automation_flows(company_id, is_active);
CREATE INDEX idx_internal_nodes_flow ON internal_automation_nodes(flow_id);
CREATE INDEX idx_internal_connections_flow ON internal_automation_connections(flow_id);
CREATE INDEX idx_internal_enrollments_entity ON internal_automation_enrollments(entity_type, entity_id);
CREATE INDEX idx_internal_enrollments_flow ON internal_automation_enrollments(flow_id, status);
CREATE INDEX idx_internal_queue_pending ON internal_automation_queue(status, scheduled_at)
  WHERE status = 'pending';
CREATE INDEX idx_internal_log_flow ON internal_automation_execution_log(flow_id, executed_at DESC);

-- Function to update updated_at
CREATE OR REPLACE FUNCTION update_internal_automation_flows_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_internal_flows_updated_at
  BEFORE UPDATE ON internal_automation_flows
  FOR EACH ROW EXECUTE FUNCTION update_internal_automation_flows_updated_at();
```

Successivamente crea i database trigger PostgreSQL che popolano la queue quando accadono eventi
nei moduli di Gestione Interna:

```sql
-- ============================================================
-- DATABASE TRIGGERS: populate internal_automation_queue
-- ============================================================

-- Helper function: trova tutti i flow attivi per un dato trigger_type e company_id
-- e inserisce i job in coda
CREATE OR REPLACE FUNCTION trigger_internal_automations(
  p_company_id UUID,
  p_trigger_type TEXT,
  p_entity_type TEXT,
  p_entity_id UUID,
  p_context JSONB DEFAULT '{}'
)
RETURNS VOID AS $$
DECLARE
  v_flow RECORD;
  v_trigger_node RECORD;
  v_enrollment_id UUID;
  v_first_action_node UUID;
BEGIN
  -- Trova tutti i flow attivi per questo trigger e company
  FOR v_flow IN
    SELECT id FROM internal_automation_flows
    WHERE company_id = p_company_id
      AND is_active = true
      AND trigger_type = p_trigger_type
  LOOP
    -- Trova il nodo trigger del flow
    SELECT id INTO v_trigger_node
    FROM internal_automation_nodes
    WHERE flow_id = v_flow.id AND node_type = 'trigger'
    LIMIT 1;

    -- Trova il primo nodo azione connesso al trigger
    SELECT target_node_id INTO v_first_action_node
    FROM internal_automation_connections
    WHERE flow_id = v_flow.id AND source_node_id = v_trigger_node.id
    LIMIT 1;

    IF v_first_action_node IS NULL THEN
      CONTINUE; -- nessuna azione configurata, skip
    END IF;

    -- Crea enrollment
    INSERT INTO internal_automation_enrollments (
      flow_id, company_id, entity_type, entity_id,
      current_node_id, status, context_data
    ) VALUES (
      v_flow.id, p_company_id, p_entity_type, p_entity_id,
      v_first_action_node, 'active', p_context
    ) RETURNING id INTO v_enrollment_id;

    -- Inserisci nella queue per esecuzione asincrona
    INSERT INTO internal_automation_queue (
      enrollment_id, flow_id, node_id, company_id,
      scheduled_at, status, payload
    ) VALUES (
      v_enrollment_id, v_flow.id, v_first_action_node, p_company_id,
      NOW(), 'pending',
      jsonb_build_object(
        'entity_type', p_entity_type,
        'entity_id', p_entity_id,
        'context', p_context
      )
    );

    -- Incrementa counter
    UPDATE internal_automation_flows
    SET total_runs = total_runs + 1, last_triggered_at = NOW()
    WHERE id = v_flow.id;

  END LOOP;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;


-- TRIGGER 1: Ordine creato
CREATE OR REPLACE FUNCTION trg_internal_order_created()
RETURNS TRIGGER AS $$
BEGIN
  PERFORM trigger_internal_automations(
    NEW.company_id, 'order_created', 'order', NEW.id,
    jsonb_build_object(
      'order_number', NEW.order_number,
      'client_name', NEW.client_name,
      'total_amount', NEW.total_amount,
      'status', NEW.status,
      'created_at', NEW.created_at
    )
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS internal_auto_order_created ON orders;
CREATE TRIGGER internal_auto_order_created
  AFTER INSERT ON orders
  FOR EACH ROW EXECUTE FUNCTION trg_internal_order_created();


-- TRIGGER 2: Stato ordine cambiato
CREATE OR REPLACE FUNCTION trg_internal_order_status_change()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.status IS DISTINCT FROM OLD.status THEN
    PERFORM trigger_internal_automations(
      NEW.company_id, 'order_status_change', 'order', NEW.id,
      jsonb_build_object(
        'old_status', OLD.status,
        'new_status', NEW.status,
        'order_number', NEW.order_number,
        'client_name', NEW.client_name,
        'total_amount', NEW.total_amount
      )
    );
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS internal_auto_order_status ON orders;
CREATE TRIGGER internal_auto_order_status
  AFTER UPDATE ON orders
  FOR EACH ROW EXECUTE FUNCTION trg_internal_order_status_change();


-- TRIGGER 3: Pagamento ricevuto su ordine
CREATE OR REPLACE FUNCTION trg_internal_payment_received()
RETURNS TRIGGER AS $$
DECLARE v_order RECORD;
BEGIN
  IF NEW.payment_status = 'paid' AND (OLD.payment_status IS DISTINCT FROM 'paid') THEN
    SELECT * INTO v_order FROM orders WHERE id = NEW.id;
    PERFORM trigger_internal_automations(
      NEW.company_id, 'order_payment_received', 'order', NEW.id,
      jsonb_build_object(
        'order_number', v_order.order_number,
        'client_name', v_order.client_name,
        'total_amount', NEW.total_amount,
        'payment_date', NOW()
      )
    );
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

```typescript
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const supabaseUrl = Deno.env.get("SUPABASE_URL")!;
const supabaseServiceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
const resendApiKey = Deno.env.get("RESEND_API_KEY");

const supabase = createClient(supabaseUrl, supabaseServiceKey);

Deno.serve(async (_req) => {
  const startTime = Date.now();
  let processedCount = 0;

  try {
    // Fetch pending jobs (max 20 per run)
    const { data: jobs, error } = await supabase
      .from("internal_automation_queue")
      .select(`
        *,
        internal_automation_nodes!node_id(node_type, node_subtype, config),
        internal_automation_enrollments!enrollment_id(entity_type, entity_id, context_data)
      `)
      .eq("status", "pending")
      .lte("scheduled_at", new Date().toISOString())
      .order("scheduled_at", { ascending: true })
      .limit(20);

    if (error) throw error;
    if (!jobs || jobs.length === 0) {
      return new Response(JSON.stringify({ processed: 0, message: "No pending jobs" }), {
        headers: { "Content-Type": "application/json" },
      });
    }

    for (const job of jobs) {
      // Mark as processing
      await supabase
        .from("internal_automation_queue")
        .update({ status: "processing" })
        .eq("id", job.id);

      const node = job.internal_automation_nodes;
      const enrollment = job.internal_automation_enrollments;

      if (!node || !enrollment) {
        await markJobFailed(job.id, "Node or enrollment not found");
        continue;
      }

      const context = {
        ...(enrollment.context_data || {}),
        ...(job.payload?.context || {}),
        entity_type: enrollment.entity_type,
        entity_id: enrollment.entity_id,
      };

      try {
        const actionStartTime = Date.now();

        // Route to correct action executor
        let result: { success: boolean; details?: Record<string, unknown>; error?: string };

        switch (node.node_subtype) {
          case "create_task":
            result = await executeCreateTask(job.company_id, node.config, context);
            break;
          case "update_order_status":
            result = await executeUpdateOrderStatus(job.company_id, enrollment.entity_id, node.config, context);
            break;
          case "send_internal_notification":
            result = await executeSendInternalNotification(job.company_id, node.config, context);
            break;
          case "send_email_to_customer":
            result = await executeSendEmailToCustomer(job.company_id, enrollment.entity_id, enrollment.entity_type, node.config, context);
            break;
          case "send_email_to_employee":
            result = await executeSendEmailToEmployee(job.company_id, node.config, context);
            break;
          case "create_ticket":
            result = await executeCreateTicket(job.company_id, node.config, context);
            break;
          case "create_calendar_event":
            result = await executeCreateCalendarEvent(job.company_id, node.config, context);
            break;
          case "wait_delay":
            result = await executeWaitDelay(job.id, node.config);
            break;
          case "webhook_call":
            result = await executeWebhookCall(node.config, context);
            break;
          case "if_condition":
            result = await executeCondition(job, node.config, context);
            break;
          default:
            result = { success: false, error: `Unknown action: ${node.node_subtype}` };
        }

        const durationMs = Date.now() - actionStartTime;

        if (result.success) {
          await markJobDone(job.id);
          await logExecution(job, node.node_subtype, "success", result.details || {}, durationMs);

          // If node_subtype is not a delay, advance to next node
          if (node.node_subtype !== "wait_delay") {
            await advanceEnrollment(job, result.details?.next_branch as string | undefined);
          }
        } else {
          if (result.error === "DELAY_SCHEDULED") {
            // Job rescheduled, do nothing
          } else {
            await markJobFailed(job.id, result.error || "Unknown error");
            await logExecution(job, node.node_subtype, "error", { error: result.error }, durationMs);
          }
        }

        processedCount++;
      } catch (err) {
        const errorMsg = err instanceof Error ? err.message : String(err);
        await markJobFailed(job.id, errorMsg);
        await logExecution(job, node.node_subtype, "error", { error: errorMsg }, 0);
      }
    }

    return new Response(
      JSON.stringify({ processed: processedCount, duration_ms: Date.now() - startTime }),
      { headers: { "Content-Type": "application/json" } }
    );
  } catch (err) {
    console.error("Fatal error:", err);
    return new Response(JSON.stringify({ error: String(err) }), {
      status: 500,
      headers: { "Content-Type": "application/json" },
    });
  }

// HELPER: Advance enrollment to next node
// ============================================================
async function advanceEnrollment(job: Record<string, unknown>, branch?: string) {
  // Find the next node via connections
  const { data: connection } = await supabase
    .from("internal_automation_connections")
    .select("target_node_id")
    .eq("flow_id", job.flow_id)
    .eq("source_node_id", job.node_id)
    .eq(branch ? "condition_branch" : "id", branch || job.id) // if branch, filter by it
    .limit(1)
    .maybeSingle();

  if (!connection) {
    // No more nodes: mark enrollment as completed
    await supabase
      .from("internal_automation_enrollments")
      .update({ status: "completed", completed_at: new Date().toISOString() })
      .eq("id", job.enrollment_id);

    await supabase
      .from("internal_automation_flows")
      .update({ successful_runs: supabase.rpc("increment_column") }) // handled by trigger
      .eq("id", job.flow_id);

    return;
  }

  // Get the target node to check if it's a delay
  const { data: nextNode } = await supabase
    .from("internal_automation_nodes")
    .select("node_subtype, config")
    .eq("id", connection.target_node_id)
    .single();

  let scheduledAt = new Date().toISOString();
  if (nextNode?.node_subtype === "wait_delay") {
    const delayMs = calculateDelayMs(nextNode.config);
    scheduledAt = new Date(Date.now() + delayMs).toISOString();
  }

  // Queue the next node
  await supabase.from("internal_automation_queue").insert({
    enrollment_id: job.enrollment_id,
    flow_id: job.flow_id,
    node_id: connection.target_node_id,
    company_id: job.company_id,
    scheduled_at: scheduledAt,
    status: "pending",
    payload: job.payload,
  });

  // Update enrollment's current node
  await supabase
    .from("internal_automation_enrollments")
    .update({ current_node_id: connection.target_node_id })
    .eq("id", job.enrollment_id);
}

function calculateDelayMs(config: Record<string, unknown>): number {
  const value = Number(config.delay_value) || 1;
  const unit = config.delay_unit as string;
  const multipliers: Record<string, number> = {
    minutes: 60 * 1000,
    hours: 60 * 60 * 1000,
    days: 24 * 60 * 60 * 1000,
  };
  return value * (multipliers[unit] || multipliers.hours);
}

function interpolate(template: string, context: Record<string, unknown>): string {
  return template.replace(/\{\{([^}]+)\}\}/g, (match, key) => {
    const parts = key.trim().split(".");
    let value: unknown = context;
    for (const part of parts) {
      value = (value as Record<string, unknown>)?.[part];
    }
    return value !== undefined && value !== null ? String(value) : match;
  });
}

async function executeCreateTask(companyId: string, config: Record<string, unknown>, context: Record<string, unknown>) {
  const dueDate = config.due_days
    ? new Date(Date.now() + Number(config.due_days) * 86400000).toISOString()
    : null;

  const { error } = await supabase.from("tasks").insert({
    company_id: companyId,
    title: interpolate(String(config.title || ""), context),
    description: config.description ? interpolate(String(config.description), context) : null,
    priority: config.priority || "medium",
    due_date: dueDate,
    assigned_to: config.assign_to || null,
    status: "todo",
    created_by_automation: true,
  });

  if (error) return { success: false, error: error.message };
  return { success: true, details: { action: "create_task" } };
}

async function executeUpdateOrderStatus(
  companyId: string,
  entityId: string,
  config: Record<string, unknown>,
  _context: Record<string, unknown>
) {
  if (!config.new_status) return { success: false, error: "new_status not configured" };

  const { error } = await supabase
    .from("orders")
    .update({ status: config.new_status })
    .eq("id", entityId)
    .eq("company_id", companyId);

  if (error) return { success: false, error: error.message };
  return { success: true, details: { action: "update_order_status", new_status: config.new_status } };
}

    const { data: admins } = await supabase
      .from("company_users")
      .select("user_id")
      .eq("company_id", companyId)
      .in("role", ["owner", "admin"]);
    userIds = (admins || []).map((u: Record<string, string>) => u.user_id);
  }

  // Insert notifications
  const notifications = userIds.map((userId) => ({
    company_id: companyId,
    user_id: userId,
    message,
    link: config.link_to ? interpolate(String(config.link_to), context) : null,
    type: "automation",
    read: false,
    created_at: new Date().toISOString(),
  }));

  if (notifications.length > 0) {
    const { error } = await supabase.from("notifications").insert(notifications);
    if (error) return { success: false, error: error.message };
  }

  return { success: true, details: { action: "send_notification", recipients: userIds.length } };
}

async function executeSendEmailToCustomer(
  companyId: string,
  entityId: string,
  entityType: string,
  config: Record<string, unknown>,
  context: Record<string, unknown>
) {
  if (!resendApiKey) return { success: false, error: "Email service not configured" };

  // Get customer email based on entity type
  let customerEmail: string | null = null;
  if (entityType === "order") {
    const { data: order } = await supabase
      .from("orders")
      .select("client_email")
      .eq("id", entityId)
      .eq("company_id", companyId)
      .single();
    customerEmail = order?.client_email;
  }

  if (!customerEmail) return { success: false, error: "Customer email not found" };

  const subject = interpolate(String(config.subject || ""), context);
  const body = interpolate(String(config.body || ""), context);

  // Get company info for from address
  const { data: company } = await supabase
    .from("companies")
    .select("name, email")
    .eq("id", companyId)
    .single();

  const response = await fetch("https://api.resend.com/emails", {
    method: "POST",
    headers: {
      Authorization: `Bearer ${resendApiKey}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      from: `${company?.name || "Gestionale"} <noreply@${Deno.env.get("APP_DOMAIN") || "gestionale.app"}>`,
      to: [customerEmail],
      subject,
      html: `<div style="font-family: sans-serif; max-width: 600px; margin: 0 auto;">${body.replace(/\n/g, "<br>")}</div>`,
    }),
  });

  if (!response.ok) {
    const err = await response.text();
    return { success: false, error: err };
  }

  return { success: true, details: { action: "send_email_customer", to: customerEmail } };
}

async function executeWaitDelay(jobId: string, config: Record<string, unknown>) {

async function executeWebhookCall(config: Record<string, unknown>, context: Record<string, unknown>) {
  if (!config.url) return { success: false, error: "URL not configured" };

  let payload = {};
  if (config.payload_template) {
    try {
      const interpolated = interpolate(String(config.payload_template), context);
      payload = JSON.parse(interpolated);
    } catch {
      payload = { data: context };
    }
  } else {
    payload = context;
  }

  const response = await fetch(String(config.url), {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payload),
  });

  if (!response.ok) return { success: false, error: `HTTP ${response.status}` };
  return { success: true, details: { action: "webhook", status: response.status } };
}

async function executeCondition(
  job: Record<string, unknown>,
  config: Record<string, unknown>,
  context: Record<string, unknown>
): Promise<{ success: boolean; details?: Record<string, unknown>; error?: string }> {
  // Simple condition evaluation: field operator value
  const { field, operator, value } = config as { field: string; operator: string; value: unknown };
  const contextValue = context[field];

  let conditionMet = false;
  switch (operator) {
    case "equals": conditionMet = String(contextValue) === String(value); break;
    case "not_equals": conditionMet = String(contextValue) !== String(value); break;
    case "contains": conditionMet = String(contextValue).toLowerCase().includes(String(value).toLowerCase()); break;
    case "greater_than": conditionMet = Number(contextValue) > Number(value); break;
    case "less_than": conditionMet = Number(contextValue) < Number(value); break;
    default: conditionMet = false;
  }

  // Find branch connections
  const branch = conditionMet ? "yes" : "no";

  const { data: connection } = await supabase
    .from("internal_automation_connections")
    .select("target_node_id")
    .eq("flow_id", job.flow_id)
    .eq("source_node_id", job.node_id)
    .eq("condition_branch", branch)
    .limit(1)
    .maybeSingle();

  if (connection) {
    await supabase.from("internal_automation_queue").insert({
      enrollment_id: job.enrollment_id,
      flow_id: job.flow_id,
      node_id: connection.target_node_id,
      company_id: job.company_id,
      scheduled_at: new Date().toISOString(),
      status: "pending",
      payload: job.payload,
    });
  }

  return { success: true, details: { action: "condition", branch, condition_met: conditionMet } };
}
```

   ```typescript
   // Hook principale per CRUD delle automazioni interne
   import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
   import { supabase } from "@/integrations/supabase/client";

   export function useInternalAutomations(companyId: string) {
     const queryClient = useQueryClient();

     const flowsQuery = useQuery({
       queryKey: ["internal_automation_flows", companyId],
       queryFn: async () => {
         const { data, error } = await supabase
           .from("internal_automation_flows")
           .select(`
             *,
             internal_automation_nodes(*),
             internal_automation_connections(*)
           `)
           .eq("company_id", companyId)
           .order("created_at", { ascending: false });
         if (error) throw error;
         return data;
       },
     });

     const createFlow = useMutation({
       mutationFn: async (flow: {
         name: string;
         triggerType: string;
         triggerConfig: Record<string, unknown>;
       }) => {
         const { data, error } = await supabase
           .from("internal_automation_flows")
           .insert({
             company_id: companyId,
             name: flow.name,
             trigger_type: flow.triggerType,
             trigger_config: flow.triggerConfig,
             is_active: false,
           })
           .select()
           .single();
         if (error) throw error;

         // Create trigger node
         await supabase.from("internal_automation_nodes").insert({
           flow_id: data.id,
           node_type: "trigger",
           node_subtype: flow.triggerType,
           config: flow.triggerConfig,
           position_x: 300,
           position_y: 50,
           label: `Trigger: ${flow.triggerType}`,
         });

         return data;
       },
       onSuccess: () => queryClient.invalidateQueries({ queryKey: ["internal_automation_flows", companyId] }),
     });

     const saveFlow = useMutation({
       mutationFn: async (params: {
         flowId: string;
         nodes: Array<{ id?: string; node_type: string; node_subtype: string; config: Record<string, unknown>; position_x: number; position_y: number }>;
         connections: Array<{ source_node_id: string; target_node_id: string; condition_branch?: string }>;
       }) => {
         // Upsert nodes
         for (const node of params.nodes) {
           if (node.id) {
             await supabase
               .from("internal_automation_nodes")
               .update({ config: node.config, position_x: node.position_x, position_y: node.position_y })
               .eq("id", node.id);
           } else {
             await supabase.from("internal_automation_nodes").insert({ ...node, flow_id: params.flowId });
           }
         }

         // Replace connections
         await supabase.from("internal_automation_connections").delete().eq("flow_id", params.flowId);
         if (params.connections.length > 0) {
           await supabase.from("internal_automation_connections").insert(
             params.connections.map((c) => ({ ...c, flow_id: params.flowId }))
           );
         }
       },
       onSuccess: () => queryClient.invalidateQueries({ queryKey: ["internal_automation_flows", companyId] }),
     });

           .from("internal_automation_flows")
           .update({ is_active: isActive })
           .eq("id", flowId);
         if (error) throw error;
       },
       onSuccess: () => queryClient.invalidateQueries({ queryKey: ["internal_automation_flows", companyId] }),
     });

     const deleteFlow = useMutation({
       mutationFn: async (flowId: string) => {
         const { error } = await supabase
           .from("internal_automation_flows")
           .delete()
           .eq("id", flowId);
         if (error) throw error;
       },
       onSuccess: () => queryClient.invalidateQueries({ queryKey: ["internal_automation_flows", companyId] }),
     });

     return { flowsQuery, createFlow, saveFlow, toggleFlow, deleteFlow };
   }
   ```

```typescript
                        <h3 className="font-semibold">{flow.name}</h3>
                        {flow.is_active ? (
                          <Badge variant="default" className="bg-green-600 text-xs">Attiva</Badge>
                        ) : (
                          <Badge variant="secondary" className="text-xs">Inattiva</Badge>
                        )}
                        {triggerDef && (
                          <Badge variant="outline" className="text-xs">{triggerDef.category}</Badge>
                        )}
                      </div>
                      <p className="text-sm text-muted-foreground mt-0.5">
                        Trigger: {triggerDef?.label || flow.trigger_type} → {nodeCount} {nodeCount === 1 ? "azione" : "azioni"}
                      </p>
                      <p className="text-xs text-muted-foreground mt-1">
                        Eseguita {flow.total_runs} volte
                        {flow.last_triggered_at && ` · Ultima: ${formatDistanceToNow(new Date(flow.last_triggered_at), { addSuffix: true, locale: it })}`}
                        {flow.total_runs > 0 && ` · ${Math.round((flow.successful_runs / flow.total_runs) * 100)}% successo`}
                      </p>
                    </div>
                  </div>
                  <Switch
                    checked={flow.is_active}
                    onCheckedChange={(checked) => toggleFlow.mutate({ flowId: flow.id, isActive: checked })}
                  />
                </div>
                <div className="flex gap-2 mt-3 pt-3 border-t">
                  <Button
                    variant="outline" size="sm"
                    onClick={() => { setSelectedFlowId(flow.id); setView("builder"); }}
                  >
                    <Edit className="h-3 w-3 mr-1" /> Modifica
                  </Button>
                  <Button variant="outline" size="sm">
                    <List className="h-3 w-3 mr-1" /> Log
                  </Button>
                  <Button
                    variant="outline" size="sm"
                    className="text-destructive hover:text-destructive"
                    onClick={() => {
                      if (confirm("Eliminare questa automazione?")) deleteFlow.mutate(flow.id);
                    }}
                  >
                    <Trash2 className="h-3 w-3 mr-1" /> Elimina
                  </Button>
                </div>
              </div>
            );
          })}
        </div>
      )}

      {/* Trigger Selector Modal */}
      {showTriggerSelector && (
        <InternalTriggerSelector
          onSelect={async (triggerType, triggerConfig) => {
            const flowName = `Nuova automazione ${new Date().toLocaleDateString("it-IT")}`;
            const newFlow = await createFlow.mutateAsync({
              name: flowName,
              triggerType,
              triggerConfig,
            });
            setSelectedFlowId(newFlow.id);
            setView("builder");
            setShowTriggerSelector(false);
          }}
          onClose={() => setShowTriggerSelector(false)}
        />
      )}
    </div>
  );
}
```

**PARTE A: Migrazione dati dalle vecchie automazioni**

Crea una migrazione Supabase per trasferire le automazioni esistenti dalla tabella legacy
`automations` alle nuove tabelle `internal_automation_flows` e `internal_automation_nodes`:

```sql
-- Migrazione one-time: importa automazioni esistenti nella nuova struttura
DO $$
DECLARE
  v_auto RECORD;
  v_flow_id UUID;
  v_trigger_node_id UUID;
  v_action_config JSONB;
  v_action JSONB;
  v_action_node_id UUID;
  v_prev_node_id UUID;
  v_action_subtype TEXT;
  v_pos_y FLOAT := 150;
BEGIN
  FOR v_auto IN SELECT * FROM automations WHERE is_active IS NOT NULL
  LOOP
    -- Crea il flow corrispondente
    INSERT INTO internal_automation_flows (
      company_id, name, trigger_type, trigger_config, is_active, created_at
    ) VALUES (
      v_auto.company_id,
      v_auto.name,
      v_auto.trigger_type,
      COALESCE(v_auto.trigger_config, '{}'),
      COALESCE(v_auto.is_active, false),
      v_auto.created_at
    ) RETURNING id INTO v_flow_id;

    -- Crea nodo trigger
    INSERT INTO internal_automation_nodes (
      flow_id, node_type, node_subtype, config, position_x, position_y, label
    ) VALUES (
      v_flow_id, 'trigger', v_auto.trigger_type,
      COALESCE(v_auto.trigger_config, '{}'),
      300, 50,
      'Trigger: ' || v_auto.trigger_type
    ) RETURNING id INTO v_trigger_node_id;

    v_prev_node_id := v_trigger_node_id;
    v_pos_y := 150;

    -- Crea nodi azione per ogni azione nel JSONB actions array
    FOR v_action IN SELECT * FROM jsonb_array_elements(COALESCE(v_auto.actions, '[]'))
    LOOP
      -- Mappa vecchi tipi ai nuovi
      v_action_subtype := CASE v_action->>'type'
        WHEN 'create_task' THEN 'create_task'
        WHEN 'change_order_status' THEN 'update_order_status'
        WHEN 'create_reminder' THEN 'send_internal_notification'
        ELSE v_action->>'type'
      END;

      INSERT INTO internal_automation_nodes (
        flow_id, node_type, node_subtype, config, position_x, position_y
      ) VALUES (
        v_flow_id, 'action', v_action_subtype,
        COALESCE(v_action->'config', '{}'),
        300, v_pos_y
      ) RETURNING id INTO v_action_node_id;

      -- Connetti al nodo precedente
      INSERT INTO internal_automation_connections (
        flow_id, source_node_id, target_node_id
      ) VALUES (
        v_flow_id, v_prev_node_id, v_action_node_id
      );

      v_prev_node_id := v_action_node_id;
      v_pos_y := v_pos_y + 120;
    END LOOP;

  END LOOP;
END;
$$;
```

```typescript
import { useQuery } from "@tanstack/react-query";
import { supabase } from "@/integrations/supabase/client";
import { Sheet, SheetContent, SheetHeader, SheetTitle } from "@/components/ui/sheet";
import { Badge } from "@/components/ui/badge";
import { formatDistanceToNow } from "date-fns";
import { it } from "date-fns/locale";

interface Props {
  flowId: string;
  flowName: string;
  open: boolean;
  onClose: () => void;
}

export function InternalAutomationLogDrawer({ flowId, flowName, open, onClose }: Props) {
  const { data: logs, isLoading } = useQuery({
    queryKey: ["internal_automation_logs", flowId],
    queryFn: async () => {
      const { data, error } = await supabase
        .from("internal_automation_execution_log")
        .select("*")
        .eq("flow_id", flowId)
        .order("executed_at", { ascending: false })
        .limit(100);
      if (error) throw error;
      return data;
    },
    enabled: open,
  });

  return (
    <Sheet open={open} onOpenChange={onClose}>
      <SheetContent side="right" className="w-full sm:max-w-2xl overflow-y-auto">
        <SheetHeader>
          <SheetTitle>Log Esecuzioni — {flowName}</SheetTitle>
        </SheetHeader>

        <div className="mt-4 space-y-2">
          {isLoading ? (
            <div className="space-y-2">
              {[1,2,3,4,5].map(i => (
                <div key={i} className="h-16 bg-muted animate-pulse rounded" />
              ))}
            </div>
          ) : !logs || logs.length === 0 ? (
            <p className="text-center text-muted-foreground py-8">Nessun log disponibile</p>
          ) : (
            logs.map((log) => (
              <div key={log.id} className="border rounded p-3 flex items-start justify-between gap-3">
                <div className="flex-1 min-w-0">
                  <div className="flex items-center gap-2 flex-wrap">
                    <Badge variant={log.status === "success" ? "default" : "destructive"} className="text-xs">
                      {log.status === "success" ? "✅ Successo" : "❌ Errore"}
                    </Badge>
                    <span className="text-sm font-medium truncate">{log.action}</span>
                    {log.entity_type && (
                      <Badge variant="outline" className="text-xs">{log.entity_type}</Badge>
                    )}
                  </div>
                  {log.details?.error && (
                    <p className="text-xs text-destructive mt-1 truncate">{String(log.details.error)}</p>
                  )}
                  <p className="text-xs text-muted-foreground mt-1">
                    {formatDistanceToNow(new Date(log.executed_at), { addSuffix: true, locale: it })}
                    {log.duration_ms && ` · ${log.duration_ms}ms`}
                  </p>
                </div>
              </div>
            ))
          )}
        </div>
      </SheetContent>
    </Sheet>
  );
```
