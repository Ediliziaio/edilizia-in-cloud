const { data: todayData, isLoading: todayLoading } = useQuery<TodayData>({

      .eq("order.company_id", companyId)
      .eq("is_paid", false)
      .lt("expected_date", todayStr);

    const overdueCount  = overdueRows?.length ?? 0;
    const overdueAmount = overdueRows?.reduce((s, r) => s + (r.amount ?? 0), 0) ?? 0;

    // Fornitori da pagare nei prossimi 7 giorni (tabella: company_costs)
    const { data: suppliersDue } = await supabase

      .select("id, name, amount, due_date, category")
      .eq("company_id", companyId)
      .eq("is_paid", false)
      .gte("due_date", todayStr)
      .lte("due_date", in7Days)
      .order("due_date", { ascending: true });

    const suppliersDueAmount = suppliersDue?.reduce((s, c) => s + (c.amount ?? 0), 0) ?? 0;

    return {
      leadsToday:         leadsToday ?? 0,
      appointmentsToday:  appointmentsToday ?? 0,
      overdueAmount,
      overdueCount,
      suppliersDueAmount,
      suppliersDue: suppliersDue ?? [],

const { data: cashFlowForecast, isLoading: forecastLoading } = useQuery<CashFlowForecastData>({
  queryKey: ["cruscotto-cashflow-forecast", companyId],
  enabled: !!companyId,
  staleTime: 300_000, // 5 minuti
  queryFn: async () => {
    const now      = new Date();
    const todayStr = now.toISOString().split("T")[0];

    const in30 = new Date(now.getTime() + 30 * 86_400_000).toISOString().split("T")[0];
    const in60 = new Date(now.getTime() + 60 * 86_400_000).toISOString().split("T")[0];
    const in90 = new Date(now.getTime() + 90 * 86_400_000).toISOString().split("T")[0];

    // Incassi attesi dai piani di pagamento degli ordini (order_installments)
    const { data: futureInstallments } = await supabase
      .from("order_installments")
      .select("amount, expected_date, order:orders!inner(company_id)")
      .eq("order.company_id", companyId)
      .eq("is_paid", false)
      .gte("expected_date", todayStr)
      .lte("expected_date", in90);

    const all = futureInstallments ?? [];
    const incoming30 = all.filter(i => i.expected_date <= in30).reduce((s, i) => s + (i.amount ?? 0), 0);
    const incoming60 = all.filter(i => i.expected_date <= in60).reduce((s, i) => s + (i.amount ?? 0), 0);
    const incoming90 = all.reduce((s, i) => s + (i.amount ?? 0), 0);

    // Fatturato per periodo (da orders.total_amount)
    const yearStart    = `${now.getFullYear()}-01-01T00:00:00`;
    const monthStart   = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}-01T00:00:00`;
    const quarter      = Math.floor(now.getMonth() / 3);
    const quarterStart = `${now.getFullYear()}-${String(quarter * 3 + 1).padStart(2, "0")}-01T00:00:00`;

    const [{ data: ytdOrders }, { data: monthOrders }, { data: qOrders }] = await Promise.all([
      supabase.from("orders").select("total_amount").eq("company_id", companyId).gte("created_at", yearStart),
      supabase.from("orders").select("total_amount").eq("company_id", companyId).gte("created_at", monthStart),
      supabase.from("orders").select("total_amount").eq("company_id", companyId).gte("created_at", quarterStart),
    ]);

    const sum = (rows: Array<{ total_amount?: number }> | null) =>
      (rows ?? []).reduce((s, r) => s + (r.total_amount ?? 0), 0);

    return {
      incoming30,
      incoming60,
      incoming90,
      ytdRevenue:     sum(ytdOrders),
      monthRevenue:   sum(monthOrders),
      quarterRevenue: sum(qOrders),
    };
  },
});

function fmtEur(n: number): string {
  if (n >= 1_000_000) return `€${(n / 1_000_000).toFixed(1)}M`;
  if (n >= 1_000)     return `€${(n / 1_000).toFixed(0)}k`;
  return `€${Math.round(n)}`;
}

function pctDelta(curr: number, prev: number): number | null {
  if (!prev) return null;
  return ((curr - prev) / prev) * 100;
}

// ─── Health Score (stessa logica di CompanyHealthScore.tsx) ─────────────────

function calcHealth(kpi: KpiData | undefined, finance: FinanceData, ops: OperationsData) {
  const margin    = finance.marginThisMonth ?? 0;
  const cf        = finance.cashFlowNet ?? 0;
  const closeRate = kpi?.close_rate ?? 0;
  const showRate  = kpi?.show_rate  ?? 0;
  const late      = (ops.lateOrders ?? 0) + (ops.overduePayments ?? 0);

  const scores = [
    margin >= 30 ? 100 : margin >= 15 ? 60 : margin >= 0 ? 30 : 0,
    cf > 5000    ? 100 : cf >= 0      ? 70 : cf > -5000  ? 30 : 0,
    closeRate >= 30 ? 100 : closeRate >= 15 ? 60 : closeRate > 0 ? 30 : 0,
    showRate  >= 70 ? 100 : showRate  >= 50 ? 60 : showRate  > 0 ? 20 : 0,
    late === 0 ? 100 : late <= 2 ? 60 : late <= 5 ? 30 : 0,
  ];
  const weights = [0.25, 0.25, 0.20, 0.15, 0.15];
  const total   = Math.round(scores.reduce((s, v, i) => s + v * weights[i], 0));
  const status  = total >= 70 ? "green" : total >= 40 ? "yellow" : "red";
  return { score: total, status } as const;
}

  // ── FINANCE ALERTS ───────────────────────────────────────────────────────
  if (fin.cashFlowNet < 0)
    alerts.push({ id: "negative-cashflow", level: "critical",
      title: `Cash flow negativo questo mese: ${fmtEur(fin.cashFlowNet)}`,
      body:  "Le uscite superano le entrate. Sollecita incassi o posticipa uscite non urgenti.",
      cta: "Analizza costi", link: "/azienda/costi" });

  // ── NUOVI ALERT DA TODAYDATA ──────────────────────────────────────────────
  if ((today?.overdueCount ?? 0) > 0)
    alerts.push({ id: "overdue-receivables", level: "critical",
      title: `${today!.overdueCount} rate non incassate — ${fmtEur(today!.overdueAmount)}`,
      body:  "Rate di pagamento con data già scaduta non ancora incassate. Contatta i clienti.",
      cta: "Gestisci ordini", link: "/azienda/ordini" });

  if ((today?.suppliersDue?.length ?? 0) > 0)
    alerts.push({ id: "suppliers-due", level: "warning",
      title: `${today!.suppliersDue!.length} fornitori da pagare nei prossimi 7 giorni`,
      body:  `${fmtEur(today!.suppliersDueAmount)} di uscite programmate. Verifica la liquidità disponibile.`,
      cta: "Vedi costi", link: "/azienda/costi" });

  // Ordina: critical → warning → info
  return alerts.sort((a, b) =>
    a.level === "critical" && b.level !== "critical" ? -1 :
    b.level === "critical" && a.level !== "critical" ?  1 :
    a.level === "warning"  && b.level === "info"     ? -1 :
    b.level === "warning"  && a.level === "info"     ?  1 : 0
  );
}

function fmtEur(n: number) {
  if (n >= 1_000_000) return `€${(n / 1_000_000).toFixed(1)}M`;
  if (n >= 1_000)     return `€${(n / 1_000).toFixed(1)}k`;
  return `€${Math.round(n)}`;
}

interface Props {

  if (isLoading) return (

VERIFICA B — KpiData esportato
─────────────────────────────────────────────────────────────────────────────
Il tipo KpiData (usato in CruscottoHero e nelle kpi card) viene definito
nel file DashboardStats o simile. Verificane il path e aggiorna l'import
in CruscottoHero.tsx se necessario:

```typescript
// Se KpiData è in un file separato tipo src/types/dashboard.ts:
import type { KpiData } from "@/types/dashboard";

// Se è inline in useCruscottoData.ts già come export:
import type { KpiData } from "@/hooks/useCruscottoData";
```

VERIFICA C — AlertsData interface in AlertPanel.tsx
─────────────────────────────────────────────────────────────────────────────
La prop `marketingAlerts` in AlertPanel.tsx usa l'interfaccia AlertsData.
Questa corrisponde al tipo `marketing?.alerts` della RPC. Verifica che
i campi combacino con ciò che ritorna la RPC `get_marketing_dashboard_stats`.
Se la RPC ritorna il campo con nome diverso, adatta AlertsData o usa `as any`.
