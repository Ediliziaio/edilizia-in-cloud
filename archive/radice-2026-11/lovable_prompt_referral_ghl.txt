```sql
-- ============================================================
-- FUNCTION: calcola e aggiorna tier automaticamente
-- ============================================================
CREATE OR REPLACE FUNCTION update_referrer_tier(p_referrer_id UUID)
RETURNS VOID AS $$
DECLARE
  v_active_count INTEGER;
  v_tier_id UUID;
BEGIN
  -- Conta aziende attive portate dal referrer
  SELECT COUNT(*) INTO v_active_count
  FROM referral_companies rc
  JOIN companies c ON rc.company_id = c.id
  WHERE rc.referrer_id = p_referrer_id
    AND rc.is_active = true
    AND c.status = 'active';

  -- Trova il tier appropriato (il più alto che soddisfa la soglia)
  SELECT id INTO v_tier_id
  FROM referral_tiers
  WHERE min_active_companies <= v_active_count
  ORDER BY min_active_companies DESC
  LIMIT 1;

  -- Aggiorna il referrer
  UPDATE referrers SET
    tier_id = v_tier_id,
    tier_updated_at = now(),
    total_conversions = v_active_count,
    conversion_rate = CASE
      WHEN total_clicks > 0 THEN (v_active_count::NUMERIC / total_clicks * 100)
      ELSE 0
    END
  WHERE id = p_referrer_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================================
-- FUNCTION: calcola commissioni mensili
-- ============================================================
CREATE OR REPLACE FUNCTION calculate_monthly_commissions(
  p_month INTEGER,
  p_year INTEGER
)
RETURNS INTEGER AS $$
DECLARE
  v_referrer RECORD;
  v_company RECORD;
  v_plan_mrr NUMERIC;
  v_commission NUMERIC;
  v_multiplier NUMERIC;
  v_count INTEGER := 0;
BEGIN
  -- Per ogni referrer attivo
  FOR v_referrer IN
    SELECT r.*, rt.commission_multiplier
    FROM referrers r
    LEFT JOIN referral_tiers rt ON r.tier_id = rt.id
    WHERE r.is_active = true
  LOOP
    -- Per ogni azienda attiva portata da questo referrer
    FOR v_company IN
      SELECT rc.company_id, sp.price_monthly, sp.name AS plan_name
      FROM referral_companies rc
      JOIN companies c ON rc.company_id = c.id
      JOIN subscription_plans sp ON c.subscription_plan_id = sp.id
      WHERE rc.referrer_id = v_referrer.id
        AND rc.is_active = true
        AND c.status = 'active'
    LOOP
      v_plan_mrr := COALESCE(v_company.price_monthly, 0);
      v_multiplier := COALESCE(v_referrer.commission_multiplier, 1.00);

      -- Calcola commissione
      IF v_referrer.commission_type = 'percentage' THEN
        v_commission := v_plan_mrr * (v_referrer.commission_value / 100) * v_multiplier;
      ELSE
        v_commission := v_referrer.commission_value * v_multiplier;
      END IF;

      -- Inserisci o aggiorna il ledger
      INSERT INTO referral_commission_ledger (
        referrer_id, company_id, period_month, period_year,
        subscription_plan_name, plan_mrr,
        commission_type, commission_rate, tier_multiplier, commission_amount,
        status
      ) VALUES (
        v_referrer.id, v_company.company_id, p_month, p_year,
        v_company.plan_name, v_plan_mrr,
        v_referrer.commission_type, v_referrer.commission_value,
        v_multiplier, v_commission,
        'pending'
      )
      ON CONFLICT (referrer_id, company_id, period_month, period_year)
      DO UPDATE SET
        plan_mrr = EXCLUDED.plan_mrr,
        commission_amount = EXCLUDED.commission_amount,
        tier_multiplier = EXCLUDED.tier_multiplier,
        calculated_at = now();

      v_count := v_count + 1;
    END LOOP;

    -- Aggiorna total_earned del referrer
    UPDATE referrers SET
      total_earned = (
        SELECT COALESCE(SUM(commission_amount), 0)
        FROM referral_commission_ledger
        WHERE referrer_id = v_referrer.id AND status != 'cancelled'
      )
    WHERE id = v_referrer.id;
  END LOOP;

  RETURN v_count;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

### A — EDGE FUNCTION: track-referral-click

Crea: supabase/functions/track-referral-click/index.ts

```typescript
// Endpoint pubblico (no auth) chiamato quando qualcuno visita un link referral
// POST { referral_code, ip_address, user_agent, utm_source, utm_medium,
//         utm_campaign, landing_page }
// Salva in referral_clicks e incrementa total_clicks sul referrer

serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders });

  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  );

  const body = await req.json();
  const { referral_code, utm_source, utm_medium, utm_campaign, landing_page } = body;
  const ip_address = req.headers.get('x-forwarded-for') ?? req.headers.get('cf-connecting-ip');
  const user_agent = req.headers.get('user-agent');

  // Trova il referrer
  const { data: referrer } = await supabase
    .from('referrers')
    .select('id')
    .eq('referral_code', referral_code)
    .eq('is_active', true)
    .single();

  if (!referrer) {
    return new Response(JSON.stringify({ error: 'Invalid referral code' }), {
      status: 404, headers: corsHeaders
    });
  }

  // Registra il click
  const { data: click } = await supabase
    .from('referral_clicks')
    .insert({
      referrer_id: referrer.id,
      referral_code,
      ip_address,
      user_agent,
      utm_source,
      utm_medium,
      utm_campaign,
      landing_page,
    })
    .select('id')
    .single();

  // Incrementa total_clicks
  await supabase.rpc('increment_referrer_clicks', { p_referrer_id: referrer.id });

  return new Response(JSON.stringify({ success: true, click_id: click?.id }), {
    headers: { ...corsHeaders, 'Content-Type': 'application/json' }
  });
});
```

Aggiungi la funzione SQL:
```sql
CREATE OR REPLACE FUNCTION increment_referrer_clicks(p_referrer_id UUID)
RETURNS VOID AS $$
  UPDATE referrers SET total_clicks = total_clicks + 1 WHERE id = p_referrer_id;
$$ LANGUAGE SQL SECURITY DEFINER;
```

### B — LOGIN PAGE: cattura il codice referral dall'URL

Modifica: src/pages/Login.tsx (o LoginForm.tsx)

```typescript
import { useSearchParams } from 'react-router-dom';
import { useEffect } from 'react';

// Nel componente:
const [searchParams] = useSearchParams();

useEffect(() => {
  const refCode = searchParams.get('ref');
  if (refCode) {
    // Salva in sessionStorage per usarlo dopo la registrazione
    sessionStorage.setItem('referral_code', refCode);

    // Traccia il click (fire and forget)
    fetch('/api/track-referral-click', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        referral_code: refCode,
        landing_page: window.location.pathname,
        utm_source: searchParams.get('utm_source'),
        utm_medium: searchParams.get('utm_medium'),
        utm_campaign: searchParams.get('utm_campaign'),
      }),
    }).catch(() => {}); // Ignora errori di tracking
  }
}, [searchParams]);
```

### C — CREATE COMPANY: usa il codice salvato in sessione

Modifica: src/pages/admin/CreateCompany.tsx

Quando il SuperAdmin crea una nuova azienda, controlla se c'è un codice
referral salvato in sessione OPPURE usa il campo "Referrer" già presente nel form.

La logica di assegnazione referral già esiste — assicurati che dopo la
creazione dell'azienda venga anche:
1. Creato il record in `referral_clicks` con converted=true
2. Aggiornato l'eventuale click esistente con converted=true e converted_company_id
3. Chiamata `update_referrer_tier(referrer_id)` per rivalutare il tier

```typescript
// Dopo la creazione dell'azienda con referrer:
if (selectedReferrerId && companyId) {
  // Già esistente: insert in referral_companies + update companies.referred_by

  // NUOVO: marca il click come convertito
  const savedClickId = sessionStorage.getItem('referral_click_id');
  if (savedClickId) {
    await supabase.from('referral_clicks').update({
      converted: true,
      converted_company_id: companyId,
      converted_at: new Date().toISOString(),
    }).eq('id', savedClickId);
  }

  // NUOVO: aggiorna tier del referrer
  await supabase.rpc('update_referrer_tier', { p_referrer_id: selectedReferrerId });

  sessionStorage.removeItem('referral_code');
  sessionStorage.removeItem('referral_click_id');
}
```

PROMPT 5 — NOTIFICHE EMAIL AI PARTNER
================================================================================

Crea: supabase/functions/send-partner-notification/index.ts

Questa function viene chiamata nei momenti chiave del lifecycle partner.

```typescript
// Parametri accettati:
// { type, referrer_id, data? }
// type: 'welcome' | 'conversion' | 'commission_calculated' |
//       'payout_approved' | 'payout_paid' | 'tier_upgrade'

serve(async (req) => {
  const { type, referrer_id, data } = await req.json();
  const supabase = createClient(/* service role */);

  const { data: referrer } = await supabase
    .from('referrers')
    .select('*, referral_tiers(*)')
    .eq('id', referrer_id)
    .single();

  if (!referrer) return new Response('Not found', { status: 404 });

  let subject = '';
  let htmlBody = '';

  const baseStyle = `font-family: sans-serif; max-width: 600px; margin: 0 auto;`;

  switch (type) {
    case 'welcome':
      subject = `🎉 Benvenuto nel Programma Partner!`;
      htmlBody = `
        <div style="${baseStyle}">
          <h2>Ciao ${referrer.name}!</h2>
          <p>Sei ufficialmente un partner <strong>${referrer.referral_tiers?.name ?? 'Bronze'}</strong>.</p>
          <p>Il tuo codice referral personale è: <strong>${referrer.referral_code}</strong></p>
          <p>Il tuo link:</p>
          <a href="${data?.signup_link}" style="background:#1E40AF;color:white;padding:12px 24px;
             text-decoration:none;border-radius:6px;display:inline-block;">
            Copia il tuo link
          </a>
          <p>Ogni volta che un'azienda si registra con il tuo link, guadagnerai
             ${referrer.commission_type === 'percentage'
               ? referrer.commission_value + '% del piano mensile'
               : '€' + referrer.commission_value + ' fissi al mese'}.</p>
        </div>`;
      break;

    case 'conversion':
      subject = `🚀 Nuova conversione! ${data?.company_name} si è registrata`;
      htmlBody = `
        <div style="${baseStyle}">
          <h2>Ottimo lavoro, ${referrer.name}!</h2>
          <p><strong>${data?.company_name}</strong> si è appena registrata usando il tuo link.</p>
          <p>Inizierai a guadagnare commissioni a partire dal prossimo mese.</p>
          <p>Vai alla tua dashboard per vedere i dettagli.</p>
        </div>`;
      break;

    case 'commission_calculated':
      subject = `💰 Le tue commissioni di ${data?.month_name} sono pronte`;
      htmlBody = `
        <div style="${baseStyle}">
          <h2>Commissioni ${data?.month_name} ${data?.year}</h2>
          <p>Ciao ${referrer.name}, abbiamo calcolato le tue commissioni per ${data?.month_name}:</p>
          <div style="background:#F0FDF4;padding:20px;border-radius:8px;text-align:center;">
            <div style="font-size:32px;font-weight:bold;color:#166534;">€${data?.total_amount}</div>
            <div style="color:#166534;">${data?.company_count} aziende attive</div>
          </div>
          <p>Vai alla tua dashboard per richiedere il pagamento.</p>
        </div>`;
      break;

    case 'payout_approved':
      subject = `✅ Pagamento di €${data?.amount} approvato`;
      htmlBody = `
        <div style="${baseStyle}">
          <h2>Pagamento approvato!</h2>
          <p>Il tuo pagamento di <strong>€${data?.amount}</strong> è stato approvato.</p>
          <p>Riceverai il bonifico entro 3-5 giorni lavorativi.</p>
          <p>Riferimento: <strong>${data?.reference ?? 'In elaborazione'}</strong></p>
        </div>`;
      break;

    case 'tier_upgrade':
      subject = `🎉 Sei salito a ${data?.new_tier}!`;
      htmlBody = `
        <div style="${baseStyle}">
          <h2>Complimenti ${referrer.name}! 🎊</h2>
          <p>Hai raggiunto il tier <strong>${data?.new_tier} ${data?.tier_icon}</strong>!</p>
          <p>Ora guadagni il <strong>${data?.multiplier}x</strong> su ogni commissione.</p>
          <div style="background:#FEF9C3;padding:20px;border-radius:8px;">
            <strong>Nuovi vantaggi sbloccati:</strong>
            <ul>${(data?.perks ?? []).map(p => `<li>${p}</li>`).join('')}</ul>
          </div>
        </div>`;
      break;
  }

  // Invia usando il sistema email della piattaforma (getPlatformSetting)
  // ... logica invio email esistente
});
```

Chiama questa function nei momenti giusti:
- Alla creazione del referrer → type='welcome'
- Dopo update_referrer_tier quando il tier cambia → type='tier_upgrade'
- Dopo calculate_monthly_commissions → type='commission_calculated' per ogni referrer
- Dopo approvazione payout → type='payout_approved'
- Dopo conversione nuova azienda → type='conversion'

VERIFICA FINALE END-TO-END
================================================================================

1. **Flusso referral completo:**
   - Il partner copia il suo link /login?ref=ABC123
   - Un cliente visita il link → track-referral-click registra il click
   - Il SuperAdmin crea l'azienda e assegna il referrer ABC123
   - Il sistema marca il click come convertito
   - update_referrer_tier aggiorna il tier se necessario
   - Email di conversione inviata al partner ✓

2. **Calcolo commissioni:**
   - Il 1° del mese, check-lifecycle-events chiama calculate_monthly_commissions
   - Il ledger viene popolato per tutte le aziende attive
   - Il partner riceve email con il totale ✓

3. **Richiesta payout:**
   - Il partner va su /partner/payout e richiede €1.240
   - L'admin vede la notifica nel tab "Payout (1)"
   - L'admin approva → email di conferma al partner ✓

4. **Upgrade tier:**
   - Il partner raggiunge 10 aziende attive
   - update_referrer_tier lo porta a Gold
   - Email "Sei salito a Gold! +25% commission!" ✓

5. **Portale partner:**
   - Il partner accede a /partner
   - Vede la sua dashboard con KPI, grafico, aziende portate ✓
   - Genera link UTM per Facebook ✓
   - Scarica materiali marketing (se Gold) ✓

NOTE IMPORTANTI
================================================================================

- Il portale /partner è SEPARATO da /azienda e /admin — ha il suo layout
- Un utente può essere contemporaneamente admin di un'azienda E partner
  (stesso email in auth.users, ma record in referrers.email)
- La pagina /partner usa useAuth per l'autenticazione ma useBrandSettings
  della piattaforma (non dell'azienda — il partner vede il brand piattaforma)
- NON modificare la struttura esistente di referrers, referral_companies,
  referral_payouts — solo AGGIUNGI colonne e comportamento
- I tier vengono assegnati automaticamente — il SuperAdmin NON li imposta manualmente
- Tutte le commissioni nel ledger sono in status='pending' fino all'approvazione
  manuale del SuperAdmin (sicurezza contro errori di calcolo)
