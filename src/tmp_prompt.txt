CONTESTO FONDAMENTALE
─────────────────────
Questo software è un SaaS multi-tenant chiamato "Edilizia in Cloud".
Ogni azienda cliente ha la propria area marketing con: contatti, opportunità,
calendario appuntamenti, email marketing, automazioni.

Il superadmin (questo prompt) ha bisogno di una sezione Marketing & Vendita
PROPRIA a livello di piattaforma, per gestire i propri lead/prospect
(potenziali clienti del SaaS stesso), non quelli delle aziende clienti.

Le stesse tabelle marketing_contacts, marketing_opportunities, appointments,
automation_flows ecc. vengono RIUSATE ma con un company_id dedicato al
superadmin (o con un flag is_platform = true). In questo modo si riusa
tutta la logica esistente senza duplicare codice.

Ogni prompt è AUTONOMO e va eseguito in ordine. Non saltare passaggi.

import { PLATFORM_ADMIN_COMPANY_ID } from "@/lib/adminConstants";
import { useSuperAdminPermissions } from "@/hooks/useSuperAdminPermissions";

/**
 * Hook base per le pagine marketing del superadmin.
 * Fornisce il companyId fisso della platform admin company
 * e verifica il permesso can_manage_marketing.

export function useAdminMarketing() {
  const { permissions, isLoading: permLoading } = useSuperAdminPermissions();

    companyId: PLATFORM_ADMIN_COMPANY_ID,
    hasAccess: permissions.can_manage_marketing,
    permLoading,
  };
}

STEP AGGIUNTIVO — Mostra demo richieste automaticamente nel calendario
────────────────────────────────────────────────────────────────────────────

Quando un'opportunità viene spostata allo stage "Demo Richiesta" nella
pipeline SaaS (AM5), crea automaticamente un appuntamento nel calendario
"Demo & Sales" con un trigger PostgreSQL:

```sql
-- Migration: auto_appointment_on_demo_stage

CREATE OR REPLACE FUNCTION create_appointment_on_demo_stage()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_stage_name text;
  v_calendar_id uuid;
BEGIN
  -- Recupera il nome dello stage
  SELECT name INTO v_stage_name
  FROM marketing_pipeline_stages
  WHERE id = NEW.stage_id;

  -- Solo per stage "Demo Richiesta" nella platform admin company
  IF NEW.company_id = '00000000-0000-0000-0000-000000000001'
     AND v_stage_name ILIKE '%demo%'
     AND (OLD.stage_id IS NULL OR OLD.stage_id != NEW.stage_id) THEN

    -- Recupera il calendario di default
    SELECT id INTO v_calendar_id
    FROM marketing_calendars
    WHERE company_id = '00000000-0000-0000-0000-000000000001'
      AND is_default = true
    LIMIT 1;

    -- Crea l'appuntamento (da confermare/spostare manualmente)
    INSERT INTO appointments (
      company_id,
      calendar_id,
      contact_id,
      title,
      appointment_date,
      appointment_time,
      appointment_end_time,
      appointment_type,
      status,
      description
    ) VALUES (
      '00000000-0000-0000-0000-000000000001',
      v_calendar_id,
      NEW.contact_id,
      'Demo - ' || COALESCE(NEW.name, 'Opportunità'),
      (now() + interval '2 days')::date,   -- suggerisce tra 2 giorni
      '10:00',
      '11:00',
      'demo',
      'pending',
      'Appuntamento demo creato automaticamente dal cambio stage.'
    );
  END IF;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_auto_appointment_demo ON marketing_opportunities;
CREATE TRIGGER trg_auto_appointment_demo
  AFTER UPDATE OF stage_id ON marketing_opportunities
  FOR EACH ROW
  EXECUTE FUNCTION create_appointment_on_demo_stage();
```

```sql
-- Migration: auto_contact_on_trial_start

CREATE OR REPLACE FUNCTION create_crm_contact_on_trial()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_pipeline_id uuid;
  v_first_stage_id uuid;
  v_contact_id uuid;
BEGIN
  -- Solo per nuove aziende in stato trial
  IF NEW.status = 'trial' AND (TG_OP = 'INSERT' OR OLD.status != 'trial') THEN

    -- Recupera pipeline di default del superadmin
    SELECT mp.id, mps.id
    INTO v_pipeline_id, v_first_stage_id
    FROM marketing_pipelines mp
    JOIN marketing_pipeline_stages mps ON mps.pipeline_id = mp.id
    WHERE mp.company_id = '00000000-0000-0000-0000-000000000001'
      AND mp.is_default = true
    ORDER BY mps.position ASC
    LIMIT 1;

    -- Crea contatto CRM per il superadmin
    INSERT INTO marketing_contacts (
      company_id,
      first_name,
      last_name,
      email,
      company_name,
      source,
      tags,
      notes
    ) VALUES (
      '00000000-0000-0000-0000-000000000001',
      SPLIT_PART(NEW.name, ' ', 1),                -- nome azienda come first_name
      COALESCE(NULLIF(SPLIT_PART(NEW.name, ' ', 2), ''), 'SRL'),
      NEW.email,
      NEW.name,
      'trial_signup',
      ARRAY['Trial Attivo'],
      'Contatto creato automaticamente all''avvio del trial. Company ID: ' || NEW.id::text
    )
    RETURNING id INTO v_contact_id;

    -- Crea opportunità collegata nel CRM del superadmin
    IF v_pipeline_id IS NOT NULL AND v_first_stage_id IS NOT NULL THEN
      INSERT INTO marketing_opportunities (
        company_id,
        pipeline_id,
        stage_id,
        contact_id,
        name,
        source,
        linked_company_id,   -- campo aggiunto in AM5
        status
      ) VALUES (
        '00000000-0000-0000-0000-000000000001',
        v_pipeline_id,
        v_first_stage_id,
        v_contact_id,
        'Trial - ' || NEW.name,
        'trial_signup',
        NEW.id,              -- collega alla company reale
        'open'
      );
    END IF;

  END IF;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_crm_contact_on_trial ON companies;
CREATE TRIGGER trg_crm_contact_on_trial
  AFTER INSERT OR UPDATE OF status ON companies
  FOR EACH ROW

-- Migration: sync_company_status_to_crm_opportunity

CREATE OR REPLACE FUNCTION sync_company_status_to_opportunity()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_opp_status text;
BEGIN
  -- Mappa lo stato azienda allo stato opportunità
  CASE NEW.status
    WHEN 'active'    THEN v_opp_status := 'won';
    WHEN 'suspended' THEN v_opp_status := 'lost';
    WHEN 'canceled'  THEN v_opp_status := 'lost';
    ELSE v_opp_status := 'open';
  END CASE;

  -- Aggiorna l'opportunità collegata nel CRM del superadmin
  UPDATE marketing_opportunities
  SET
    status = v_opp_status,
    updated_at = now()
  WHERE linked_company_id = NEW.id
    AND company_id = '00000000-0000-0000-0000-000000000001';

  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_sync_status_to_opportunity ON companies;
CREATE TRIGGER trg_sync_status_to_opportunity
  AFTER UPDATE OF status ON companies
  FOR EACH ROW
  WHEN (OLD.status IS DISTINCT FROM NEW.status)
  EXECUTE FUNCTION sync_company_status_to_opportunity();
```
