Nel componente KpiChip esistente, aggiungi il supporto per soglie:

```tsx
// Aggiorna la firma di KpiChip aggiungendo thresholds opzionale:
function KpiChip({
  label,
  value,
  prevValue,
  format = 'number',
  thresholds,       // NUOVO: { warning: number; critical: number; higherIsBetter?: boolean }
}: {
  label: string;
  value: number;
  prevValue?: number;
  format?: 'number' | 'currency' | 'percent';
  thresholds?: { warning: number; critical: number; higherIsBetter?: boolean };
}) {
  // ... codice esistente ...

  // Calcola stato basato su soglie
  const chipStatus = (() => {
    if (!thresholds) return 'neutral';
    const { warning, critical, higherIsBetter = true } = thresholds;
    if (higherIsBetter) {
      if (value >= warning) return 'good';
      if (value >= critical) return 'warning';
      return 'critical';
    } else {
      // Per metriche dove valori alti sono negativi
      if (value <= warning) return 'good';
      if (value <= critical) return 'warning';
      return 'critical';
    }
  })();

  const statusBg = {
    good: '',                                    // nessun colore: normale
    neutral: '',
    warning: 'bg-yellow-50 border-yellow-200',
    critical: 'bg-red-50 border-red-200',
  }[chipStatus];

  const statusText = {
    good: '',
    neutral: '',
    warning: 'text-yellow-700',
    critical: 'text-red-700 font-bold',
  }[chipStatus];

  return (
    <div className={`flex items-center gap-2 rounded-md border px-3 py-1.5 flex-1 min-w-0 ${statusBg}`}>
      <div className="min-w-0">
        <div className={`text-xs truncate ${chipStatus === 'critical' ? 'text-red-600' : 'text-muted-foreground'}`}>
          {label}
        </div>
        <div className={`text-sm font-semibold leading-tight ${statusText}`}>
          {displayValue}
        </div>
      </div>
      {delta && (
        <DeltaIcon className={`h-3.5 w-3.5 shrink-0 ${deltaColor}`} />
      )}
    </div>
  );
}
```

```tsx
// Aggiungi import se non già presente:

// Se formatDuration non esiste in utils, usa questa funzione inline:
const formatDuration = (seconds: number) => {
  if (!seconds || seconds === 0) return '—';
  const m = Math.floor(seconds / 60);
  const s = seconds % 60;
  return m > 0 ? `${m}m ${s}s` : `${s}s`;
};

// Calcola statistiche aggregate del call center dai dati per operatore:
const callCenterSummary = data.call_center && data.call_center.length > 0
  ? {
      totalCalls: data.call_center.reduce((s, r) => s + r.calls_total, 0),
      totalAnswered: data.call_center.reduce((s, r) => s + r.calls_answered, 0),
      totalAppSet: data.call_center.reduce((s, r) => s + r.appointments_set, 0),
      avgDuration: data.call_center.reduce((s, r) => s + r.avg_duration_sec, 0) /
                   data.call_center.filter((r) => r.avg_duration_sec > 0).length || 0,
      operatorsCount: data.call_center.length,
      contactRate: data.call_center.reduce((s, r) => s + r.calls_total, 0) > 0
        ? (data.call_center.reduce((s, r) => s + r.calls_answered, 0) /
           data.call_center.reduce((s, r) => s + r.calls_total, 0)) * 100
        : 0,
    }
  : null;
```
