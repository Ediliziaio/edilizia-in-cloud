```typescript
import { useState, useCallback, useMemo } from 'react';

// ============================================================
// TYPES
// ============================================================

export type TabId =
  | 'panoramica'
  | 'pipeline'
  | 'attivita'
  | 'team'
  | 'fonti'
  | 'trend';

export interface TabDefinition {
  id: TabId;
  label: string;
  iconName: string; // nome lucide-react (es: 'LayoutDashboard')
  description: string;
  defaultVisible: boolean;
}

export interface DashboardLayout {
  visibleTabs: TabId[];
  activeTab: TabId;
}

// ============================================================
// CONFIGURAZIONE TAB (immutabile)
// ============================================================

export const ALL_TABS: TabDefinition[] = [
  {
    id: 'panoramica',
    label: 'Panoramica',
    iconName: 'LayoutDashboard',
    description: 'Vista executive con i KPI più importanti',
    defaultVisible: true,
  },
  {
    id: 'pipeline',
    label: 'Pipeline',
    iconName: 'GitBranch',
    description: 'Funnel opportunità e forecast dettagliato',
    defaultVisible: true,
  },
  {
    id: 'attivita',
    label: 'Attività',
    iconName: 'Phone',
    description: 'Chiamate, appuntamenti e call center',
    defaultVisible: true,
  },
  {
    id: 'team',
    label: 'Team',
    iconName: 'Users',
    description: 'Performance commerciali per venditore',
    defaultVisible: true,
  },
  {
    id: 'fonti',
    label: 'Fonti & ROI',
    iconName: 'BarChart2',
    description: 'Analisi fonti lead e ritorno investimento',
    defaultVisible: true,
  },
  {
    id: 'trend',
    label: 'Trend',
    iconName: 'TrendingUp',
    description: 'Andamento temporale lead, appuntamenti, vinti',
    defaultVisible: true,
  },
];

const STORAGE_KEY = 'mktg_dashboard_layout_v1';

const DEFAULT_LAYOUT: DashboardLayout = {
  visibleTabs: ALL_TABS.filter((t) => t.defaultVisible).map((t) => t.id),
  activeTab: 'panoramica',
};

// ============================================================
// HELPER: legge localStorage in modo sicuro
// ============================================================

function readStorage(): DashboardLayout {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return DEFAULT_LAYOUT;
    const parsed = JSON.parse(raw) as Partial<DashboardLayout>;

    // Valida che i tab salvati esistano ancora
    const validIds = new Set<TabId>(ALL_TABS.map((t) => t.id));
    const visibleTabs = (parsed.visibleTabs ?? DEFAULT_LAYOUT.visibleTabs).filter(
      (id) => validIds.has(id)
    );
    // Assicura che panoramica sia sempre visibile
    if (!visibleTabs.includes('panoramica')) {
      visibleTabs.unshift('panoramica');
    }

    const activeTab =
      parsed.activeTab && validIds.has(parsed.activeTab) && visibleTabs.includes(parsed.activeTab)
        ? parsed.activeTab
        : visibleTabs[0] ?? 'panoramica';

    return { visibleTabs, activeTab };
  } catch {
    return DEFAULT_LAYOUT;
  }
}

function writeStorage(layout: DashboardLayout): void {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(layout));
  } catch {
    // Ignora errori localStorage (private browsing, etc.)
  }
}

// ============================================================
// HOOK
// ============================================================

export function useDashboardLayout() {
  const [layout, setLayout] = useState<DashboardLayout>(() => readStorage());

  const setActiveTab = useCallback((id: TabId) => {
    setLayout((prev) => {
      // Non attivare tab non visibili
      if (!prev.visibleTabs.includes(id)) return prev;
      const next = { ...prev, activeTab: id };
      writeStorage(next);
      return next;
    });
  }, []);

  const toggleTabVisibility = useCallback((id: TabId) => {
    setLayout((prev) => {
      // 'panoramica' non si può nascondere
      if (id === 'panoramica') return prev;

      const isCurrentlyVisible = prev.visibleTabs.includes(id);
      let visibleTabs: TabId[];

      if (isCurrentlyVisible) {
        // Rimuove il tab dalla lista visibili
        visibleTabs = prev.visibleTabs.filter((t) => t !== id);
        // Assicura almeno 1 tab visibile oltre panoramica
        if (visibleTabs.length === 0) visibleTabs = ['panoramica'];
      } else {
        // Aggiunge il tab nella posizione originale (basata su ALL_TABS)
        const allOrder = ALL_TABS.map((t) => t.id);
        visibleTabs = allOrder.filter(
          (tabId) => prev.visibleTabs.includes(tabId) || tabId === id
        );
      }

      // Se il tab attivo viene nascosto, attiva il primo disponibile
      const activeTab =
        visibleTabs.includes(prev.activeTab)
          ? prev.activeTab
          : visibleTabs[0] ?? 'panoramica';

      const next = { visibleTabs, activeTab };
      writeStorage(next);
      return next;
    });
  }, []);

  const resetToDefaults = useCallback(() => {
    writeStorage(DEFAULT_LAYOUT);
    setLayout(DEFAULT_LAYOUT);
  }, []);

  // Tab definitions nella giusta order (basata su ALL_TABS)
  const visibleTabDefs = useMemo(
    () =>
      ALL_TABS.filter((t) => layout.visibleTabs.includes(t.id)),
    [layout.visibleTabs]
  );

  return {
    layout,
    visibleTabDefs,
    activeTab: layout.activeTab,
    setActiveTab,
    toggleTabVisibility,
    resetToDefaults,
  };
}
```

```tsx
export default function MarketingDashboard() {
  const { data, isLoading, error, refetch, filters, updateFilters, permissions } =

  const [showTargets, setShowTargets] = useState(false);

    if (isLoading || !data) {

        <div className="flex items-center justify-center h-32 text-destructive text-sm">
          Errore nel caricamento dei dati. Riprova.
        </div>
      );

    const commonProps = { data, filters };

      case 'panoramica':

      case 'fonti':
        return <TabFonti {...commonProps} />;

          <h1 className="text-base font-semibold leading-tight">

              variant="outline"
              size="sm"
              className="h-7 text-xs gap-1"
              onClick={() => setShowTargets(true)}
            >
              <Target className="h-3 w-3" />
              Target

          <Button
            variant="ghost"
```

```tsx
// AlertBanner.tsx — banner fisso collassabile per gli alert operativi
import React, { useState } from 'react';
import { AlertsData } from '@/hooks/useMarketingDashboard'; // importa il tipo esistente
import { AlertTriangle, Clock, CalendarX, Target, TrendingDown, X, ChevronDown, ChevronUp } from 'lucide-react';
import { useNavigate } from 'react-router-dom';

interface AlertBannerProps {
  alerts: AlertsData;
  alertCount: number;
}

interface AlertItem {
  id: string;
  message: string;
  count?: number;
  severity: 'critical' | 'warning';
  icon: React.ComponentType<{ className?: string }>;
  path?: string;
}

export function AlertBanner({ alerts, alertCount }: AlertBannerProps) {
  const [collapsed, setCollapsed] = useState(false);
  const [dismissed, setDismissed] = useState(false);
  const navigate = useNavigate();

  if (dismissed) return null;

  const items: AlertItem[] = [
    alerts.stale_leads > 0 && {
      id: 'stale-leads',
      message: `${alerts.stale_leads} lead non contattati da 48h+`,
      count: alerts.stale_leads,
      severity: 'critical' as const,
      icon: Clock,
      path: '/azienda/marketing/contatti',
    },
    alerts.stale_opportunities > 0 && {
      id: 'stale-opps',
      message: `${alerts.stale_opportunities} opportunità ferme da 7+ giorni`,
      count: alerts.stale_opportunities,
      severity: 'critical' as const,
      icon: AlertTriangle,
      path: '/azienda/marketing/opportunita',
    },
    alerts.stale_leads_2h > 0 && {
      id: 'stale-leads-2h',
      message: `${alerts.stale_leads_2h} lead non contattati entro 2h`,
      count: alerts.stale_leads_2h,
      severity: 'warning' as const,
      icon: Clock,
      path: '/azienda/marketing/contatti',
    },
    alerts.pending_appointments > 0 && {
      id: 'pending-app',
      message: `${alerts.pending_appointments} appuntamenti passati senza esito`,
      count: alerts.pending_appointments,
      severity: 'warning' as const,
      icon: CalendarX,
      path: '/azienda/marketing/appuntamenti',
    },
    alerts.show_rate_below_threshold && {
      id: 'show-rate',
      message: 'Show rate sotto soglia (< 60%)',
      severity: 'warning' as const,
      icon: Target,
    },
    alerts.pipeline_declining && {
      id: 'pipeline',
      message: 'Pipeline in calo vs periodo precedente',
      severity: 'warning' as const,
      icon: TrendingDown,
    },
  ].filter(Boolean) as AlertItem[];

  const hasCritical = items.some((i) => i.severity === 'critical');
  const bannerColor = hasCritical
    ? 'bg-red-50 border-red-200 text-red-800'
    : 'bg-yellow-50 border-yellow-200 text-yellow-800';
  const dotColor = hasCritical ? 'bg-red-500' : 'bg-yellow-500';

  return (
    <div className={`border-b shrink-0 ${bannerColor}`}>
      {/* Header del banner — sempre visibile */}
      <div className="flex items-center gap-2 px-4 py-1.5">
        <span className={`inline-block h-2 w-2 rounded-full ${dotColor} animate-pulse shrink-0`} />
        <span className="text-xs font-medium flex-1">
          {collapsed
            ? `${alertCount} alert operation${alertCount === 1 ? 'e' : 'i'} — clicca per espandere`
            : items.slice(0, 2).map((item) => item.message).join('  ·  ')
              + (items.length > 2 ? `  +${items.length - 2} altri` : '')}
        </span>
        <div className="flex items-center gap-1">
          <button
            onClick={() => setCollapsed((c) => !c)}
            className="p-0.5 hover:opacity-70 transition-opacity"
          >
            {collapsed
              ? <ChevronDown className="h-3.5 w-3.5" />
              : <ChevronUp className="h-3.5 w-3.5" />}
          </button>
          <button
            onClick={() => setDismissed(true)}
            className="p-0.5 hover:opacity-70 transition-opacity"
            title="Nascondi alert (si rimostra al prossimo aggiornamento)"
          >
            <X className="h-3.5 w-3.5" />
          </button>
        </div>
      </div>

      {/* Lista alert espansa */}
      {!collapsed && items.length > 2 && (
        <div className="px-4 pb-2 flex flex-wrap gap-2">
          {items.map((item) => {
            const Icon = item.icon;
            return (
              <button
                key={item.id}
                onClick={() => item.path && navigate(item.path)}
                className={`flex items-center gap-1.5 text-xs px-2 py-1 rounded-full border
                  transition-opacity hover:opacity-80
                  ${item.severity === 'critical'
                    ? 'bg-red-100 border-red-300 text-red-800'
                    : 'bg-yellow-100 border-yellow-300 text-yellow-800'}
                  ${item.path ? 'cursor-pointer' : 'cursor-default'}`}
              >
                <Icon className="h-3 w-3" />
                {item.message}
              </button>
            );
          })}
        </div>
      )}
    </div>
  );
}
```

```tsx
// SOTTO-COMPONENTE: HeroCard (una delle 3 card principali)

function HeroCard({
  label,
  value,
  formattedValue,
  prevValue,
  target,
  format = 'currency',
  isMain = false,
}: {
  label: string;
  value: number;
  formattedValue: string;
  prevValue?: number;
  target?: number;
  format?: 'currency' | 'number' | 'percent';
  isMain?: boolean;
}) {
  const delta = prevValue !== undefined ? calcDelta(value, prevValue) : null;
  const targetPct = target && target > 0 ? Math.min((value / target) * 100, 100) : null;

  const DeltaIcon =
    delta?.direction === 'up'
      ? TrendingUp
      : delta?.direction === 'down'
      ? TrendingDown
      : Minus;

  const deltaColor =
    delta?.direction === 'up'
      ? 'text-green-600'
      : delta?.direction === 'down'
      ? 'text-red-500'
      : 'text-muted-foreground';

  return (
    <div className={`rounded-lg border bg-card p-3 flex flex-col gap-1.5 ${isMain ? 'ring-1 ring-primary/20' : ''}`}>
      <div className="text-xs font-medium text-muted-foreground uppercase tracking-wide">
        {label}
      </div>
      <div className={`font-bold leading-none ${isMain ? 'text-3xl' : 'text-2xl'}`}>
        {formattedValue}
      </div>
      {/* Delta vs periodo precedente */}
      {delta && (
        <div className={`flex items-center gap-1 text-xs ${deltaColor}`}>
          <DeltaIcon className="h-3 w-3" />
          <span>{delta.value > 0 ? '+' : ''}{delta.value.toFixed(0)}% vs prec.</span>
        </div>
      )}
      {/* Barra progresso target (solo se target definito) */}
      {targetPct !== null && (
        <div className="space-y-0.5 mt-0.5">
          <div className="flex justify-between text-xs text-muted-foreground">
            <span>{targetPct.toFixed(0)}% del target</span>
            <span>{fmtCur(target!)}</span>
          </div>
          <div className="h-1.5 rounded-full bg-muted overflow-hidden">
            <div
              className={`h-full rounded-full transition-all ${
                targetPct >= 100 ? 'bg-green-500' : targetPct >= 50 ? 'bg-primary' : 'bg-orange-400'
              }`}
              style={{ width: `${targetPct}%` }}
            />
          </div>
        </div>
      )}
    </div>
  );
}

// ──────────────────────────────────────────────────────────────
// SOTTO-COMPONENTE: KpiChip (chip compatto per la strip)
// ──────────────────────────────────────────────────────────────
function KpiChip({
  label,
  value,
  prevValue,
  format = 'number',
}: {
  label: string;
  value: number;
  prevValue?: number;
  format?: 'number' | 'currency' | 'percent';
}) {
  const delta = prevValue !== undefined ? calcDelta(value, prevValue) : null;

  const displayValue =
    format === 'currency'
      ? fmtCur(value)
      : format === 'percent'
      ? `${value.toFixed(1)}%`
      : fmt(value);

  const deltaColor =
    delta?.direction === 'up'
      ? 'text-green-600'
      : delta?.direction === 'down'
      ? 'text-red-500'
      : 'text-muted-foreground';

  const DeltaIcon =
    delta?.direction === 'up' ? TrendingUp : delta?.direction === 'down' ? TrendingDown : Minus;

  return (
    <div className="flex items-center gap-2 rounded-md border bg-card px-3 py-1.5 flex-1 min-w-0">
      <div className="min-w-0">
        <div className="text-xs text-muted-foreground truncate">{label}</div>
        <div className="text-sm font-semibold leading-tight">{displayValue}</div>
      </div>
      {delta && (
        <DeltaIcon className={`h-3.5 w-3.5 shrink-0 ${deltaColor}`} />
      )}
    </div>
  );
}
```

```tsx
import React from 'react';
import {
  ALL_TABS,
  TabDefinition,
  TabId,
} from '@/hooks/useDashboardLayout';
import {
  Sheet,
  SheetContent,
  SheetHeader,
  SheetTitle,
  SheetDescription,
} from '@/components/ui/sheet';
import { Switch } from '@/components/ui/switch';
import { Button } from '@/components/ui/button';
import {
  LayoutDashboard,
  GitBranch,
  Phone,
  Users,
  BarChart2,
  TrendingUp,
  RotateCcw,
} from 'lucide-react';

const TAB_ICONS: Record<string, React.ComponentType<{ className?: string }>> = {
  LayoutDashboard,
  GitBranch,
  Phone,
  Users,
  BarChart2,
  TrendingUp,
};

interface DashboardCustomizePanelProps {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  visibleTabDefs: TabDefinition[];
  onToggleTab: (id: TabId) => void;
  onReset: () => void;
}

export function DashboardCustomizePanel({
  open,
  onOpenChange,
  visibleTabDefs,
  onToggleTab,
  onReset,
}: DashboardCustomizePanelProps) {
  const visibleIds = new Set(visibleTabDefs.map((t) => t.id));

  return (
    <Sheet open={open} onOpenChange={onOpenChange}>
      <SheetContent side="right" className="w-80">
        <SheetHeader>
          <SheetTitle>Personalizza Dashboard</SheetTitle>
          <SheetDescription>
            Scegli quali tab visualizzare nella barra di navigazione.
            La tab Panoramica è sempre visibile.
          </SheetDescription>
        </SheetHeader>

        <div className="mt-6 space-y-1">
          {ALL_TABS.map((tab) => {
            const Icon = TAB_ICONS[tab.iconName];
            const isVisible = visibleIds.has(tab.id);
            const isLocked = tab.id === 'panoramica'; // non disabilitabile

            return (
              <div
                key={tab.id}
                className={`flex items-center gap-3 rounded-lg p-3 transition-colors ${
                  isVisible ? 'bg-accent/50' : 'opacity-50'
                }`}
              >
                <div className="flex items-center justify-center h-8 w-8 rounded-md bg-background border shrink-0">
                  {Icon && <Icon className="h-4 w-4 text-muted-foreground" />}
                </div>
                <div className="flex-1 min-w-0">
                  <div className="text-sm font-medium">{tab.label}</div>
                  <div className="text-xs text-muted-foreground truncate">
                    {tab.description}
                  </div>
                </div>
                <Switch
                  checked={isVisible}
                  disabled={isLocked}
                  onCheckedChange={() => onToggleTab(tab.id as TabId)}
                />
              </div>
            );
          })}
        </div>

        <div className="mt-6 pt-4 border-t">
          <Button
            variant="outline"
            size="sm"
            className="w-full gap-2"
            onClick={() => {
              onReset();
              onOpenChange(false);
            }}
          >
            <RotateCcw className="h-3.5 w-3.5" />
            Ripristina impostazioni predefinite
          </Button>
          <p className="text-xs text-center text-muted-foreground mt-2">
            Le preferenze vengono salvate automaticamente
          </p>
        </div>
      </SheetContent>
    </Sheet>
  );
}
```
