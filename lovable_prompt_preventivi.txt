-- ============================================================
-- FUNCTION: genera numero progressivo offerta
-- ============================================================
CREATE OR REPLACE FUNCTION generate_quote_number(p_company_id UUID)
RETURNS TEXT AS $$
DECLARE
  v_year TEXT := to_char(now(), 'YYYY');
  v_count INTEGER;
  v_number TEXT;
BEGIN
  SELECT COUNT(*) + 1 INTO v_count
  FROM quotes
  WHERE company_id = p_company_id
    AND to_char(created_at, 'YYYY') = v_year;

  v_number := 'OFF-' || v_year || '-' || LPAD(v_count::TEXT, 3, '0');
  RETURN v_number;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================================
-- TRIGGER: aggiorna updated_at
-- ============================================================
CREATE OR REPLACE FUNCTION quotes_set_updated_at()
RETURNS TRIGGER AS $$
BEGIN NEW.updated_at = now(); RETURN NEW; END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER quotes_updated_at
BEFORE UPDATE ON quotes
FOR EACH ROW EXECUTE FUNCTION quotes_set_updated_at();

CREATE TRIGGER quote_pdf_materials_updated_at
BEFORE UPDATE ON quote_pdf_materials
FOR EACH ROW EXECUTE FUNCTION quotes_set_updated_at();

-- ============================================================
-- TRIGGER: ricalcola totale quote automaticamente
-- ============================================================
CREATE OR REPLACE FUNCTION recalculate_quote_totals()
RETURNS TRIGGER AS $$
DECLARE
  v_subtotal NUMERIC(12,2);
  v_discount NUMERIC(12,2);
  v_tax_rate NUMERIC(5,2);
  v_tax_amount NUMERIC(12,2);
  v_total NUMERIC(12,2);
  v_quote_id UUID;
BEGIN
  -- Determina quale quote aggiornare
  IF TG_OP = 'DELETE' THEN
    v_quote_id := OLD.quote_id;
  ELSE
    v_quote_id := NEW.quote_id;
  END IF;

  -- Calcola subtotale dalle righe
  SELECT COALESCE(SUM(total_price), 0) INTO v_subtotal
  FROM quote_items
  WHERE quote_id = v_quote_id AND item_type != 'text';

  -- Prendi lo sconto e aliquota IVA dalla quote
  SELECT
    global_discount_percentage,
    tax_rate
  INTO v_discount, v_tax_rate
  FROM quotes WHERE id = v_quote_id;

  v_discount := v_subtotal * COALESCE(v_discount, 0) / 100;
  v_tax_amount := (v_subtotal - v_discount) * COALESCE(v_tax_rate, 22) / 100;
  v_total := v_subtotal - v_discount + v_tax_amount;

  UPDATE quotes SET
    subtotal = v_subtotal,
    global_discount_amount = v_discount,
    tax_amount = v_tax_amount,
    total = v_total,
    updated_at = now()
  WHERE id = v_quote_id;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER quote_items_recalculate
AFTER INSERT OR UPDATE OR DELETE ON quote_items
FOR EACH ROW EXECUTE FUNCTION recalculate_quote_totals();

import { serve } from "https://deno.land/std@0.177.0/http/server.ts";
import { createClient } from "npm:@supabase/supabase-js@2";
import { PDFDocument, rgb, StandardFonts } from "npm:pdf-lib@1.17.1";
import { corsHeaders } from "../_shared/cors.ts";

serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders });

  try {
    const authHeader = req.headers.get('Authorization');
    if (!authHeader) return new Response(JSON.stringify({ error: 'Unauthorized' }), { status: 401 });

    const supabase = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
    );

    // Verifica autenticazione
    const { data: { user }, error: authError } = await createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_ANON_KEY')!,
      { global: { headers: { Authorization: authHeader } } }
    ).auth.getUser();

    if (authError || !user) {
      return new Response(JSON.stringify({ error: 'Unauthorized' }), { status: 401 });
    }

    const { quote_id } = await req.json();

    // Carica tutti i dati
    const { data: quote, error: quoteError } = await supabase
      .from('quotes')
      .select(`
        *,
        quote_items(*),
        quote_pdf_attachments(
          *,
          quote_pdf_materials(*)
        )
      `)
      .eq('id', quote_id)
      .single();

    if (quoteError || !quote) {
      return new Response(JSON.stringify({ error: 'Quote not found' }), { status: 404 });
    }

    // Verifica company ownership
    const { data: profile } = await supabase
      .from('profiles')
      .select('company_id')
      .eq('id', user.id)
      .single();

    if (profile?.company_id !== quote.company_id) {
      return new Response(JSON.stringify({ error: 'Forbidden' }), { status: 403 });
    }

    // Carica dati azienda
    const { data: company } = await supabase
      .from('companies')
      .select('name, logo_url, address, city, vat_number, phone, email')
      .eq('id', quote.company_id)
      .single();

    // GENERA PDF con pdf-lib
    const pdfDoc = await PDFDocument.create();
    const helvetica = await pdfDoc.embedFont(StandardFonts.Helvetica);
    const helveticaBold = await pdfDoc.embedFont(StandardFonts.HelveticaBold);

    const A4_WIDTH = 595.28;
    const A4_HEIGHT = 841.89;
    const MARGIN = 50;

    // --- PAGINA 1: COPERTINA ---
    const coverPage = pdfDoc.addPage([A4_WIDTH, A4_HEIGHT]);
    // [Implementa il layout della copertina come descritto sopra]
    // Usa coverPage.drawText(), coverPage.drawLine(), coverPage.drawRectangle()

    // --- PAGINA 2: DETTAGLIO PRODOTTI ---
    const detailPage = pdfDoc.addPage([A4_WIDTH, A4_HEIGHT]);
    // [Implementa la tabella prodotti e il riepilogo importi]

    // Ordina items per position
    const sortedItems = (quote.quote_items || []).sort((a, b) => a.position - b.position);
    // Disegna ogni riga...

    // --- PAGINE PDF ALLEGATI ---
    const attachments = (quote.quote_pdf_attachments || [])
      .sort((a, b) => a.position - b.position);

    for (const attachment of attachments) {
      const material = attachment.quote_pdf_materials;
      if (!material?.file_url) continue;

      // Scarica il PDF da storage
      const { data: fileData } = await supabase.storage
        .from('quote-materials')
        .download(material.file_url.split('quote-materials/')[1]);

      if (!fileData) continue;

      const pdfBytes = await fileData.arrayBuffer();
      const attachedPdf = await PDFDocument.load(pdfBytes);
      const pages = await pdfDoc.copyPages(attachedPdf, attachedPdf.getPageIndices());
      pages.forEach(page => pdfDoc.addPage(page));
    }

    // Salva PDF
    const finalPdfBytes = await pdfDoc.save();
    const fileName = `${quote.company_id}/${quote.quote_number.replace(/\//g, '-')}.pdf`;

    // Upload su storage (bucket quotes-pdf, crea il bucket se non esiste)
    await supabase.storage
      .from('quote-pdfs')
      .upload(fileName, finalPdfBytes, {
        contentType: 'application/pdf',
        upsert: true
      });

    // Crea signed URL valida 24h
    const { data: signedUrlData } = await supabase.storage
      .from('quote-pdfs')
      .createSignedUrl(fileName, 86400);

    // Aggiorna quote con pdf_url e timestamp
    await supabase.from('quotes').update({
      pdf_url: fileName,
      pdf_generated_at: new Date().toISOString()
    }).eq('id', quote_id);

    return new Response(JSON.stringify({
      success: true,
      pdf_url: signedUrlData?.signedUrl,
      file_name: `${quote.quote_number} - ${quote.title}.pdf`
    }), {
      headers: { ...corsHeaders, 'Content-Type': 'application/json' }
    });

  } catch (error) {
    console.error('Error generating quote PDF:', error);
    return new Response(JSON.stringify({ error: error.message }), {
      status: 500,
      headers: corsHeaders
    });
  }
});

### EDGE FUNCTION 3: quote-sign (PUBBLICA — no auth richiesta)

File: supabase/functions/quote-sign/index.ts

COSA FA:
1. Riceve: { token, action: 'view' | 'sign' | 'refuse', signer_name?, refuse_reason? }
2. NON richiede autenticazione (è pubblica)
3. Cerca la quote per signature_token
4. Se action='view': aggiorna viewed_at (solo se non già visto), ritorna i dati della quote
5. Se action='sign':
   - Verifica che la quote sia in status 'inviata' e non scaduta
   - Salva: signed_at=now(), signed_by_name, signed_by_ip, status='accettata'
   - Genera PDF finale con timbro "FIRMATO DIGITALMENTE da [nome] il [data]"
   - Invia email di conferma all'azienda e al cliente
6. Se action='refuse':
   - Salva: refused_at=now(), refused_reason, status='rifiutata'
   - Notifica l'azienda

SICUREZZA:
- Rate limiting: max 10 richieste per IP per 5 minuti
- Valida che signature_token sia UUID valido (regex)
- Non esporre mai dati sensibili dell'azienda (solo quelli del preventivo)
- Verifica expires_at: se scaduto, restituisci errore 410 Gone
